  199     def _run_locked(
  200         self,
  201         *,
  202         initial_context: dict[str, Any],
  203         initial_candidate: dict[str, Any] | None,
  204     ) -> SearchResult:
  205         start_time = self.owner._begin_native_run(initial_context)
  206         stopping_reason: str | None = None
  207         candidate_to_seed = initial_candidate
  208 
  209         while (
  210             self.owner._run_state.evaluation_iterations
  211             < self.owner._config.max_iterations_hard_limit
  212         ):
  213             stopping_reason = self._stopping_reason(initial_context)
  214             if stopping_reason is not None:
  215                 self._stop(stopping_reason)
  216                 break
  217 
  218             batch, generated, stopping_reason = self._prepare_batch(
  219                 initial_context=initial_context,
  220                 initial_candidate=candidate_to_seed,
  221             )
  222             candidate_to_seed = None
  223             if stopping_reason is not None:
  224                 self._stop(stopping_reason)
  225                 break
  226 
  227             if not batch:
  228                 stopping_reason = self._handle_empty_generation()
  229                 if stopping_reason is not None:
  230                     break
  231                 continue
  232 
  233             self._mark_nonempty_generation()
  234             stopping_reason = self._evaluate_batch(
  235                 batch=batch,
  236                 generated=generated,
  237                 initial_context=initial_context,
  238             )
  239             if stopping_reason is not None:
  240                 break
  241 
  242         return self.owner._finish_native_run(
  243             start_time=start_time,
  244             stopping_reason=stopping_reason,
  245         )

  234     def _prepare_service_run(self) -> None:
  235         """Ensure the ask/tell bridge has one explicit run-state owner.
  236 
  237         The compatibility ``run()`` path always starts a fresh state.  The
  238         contract adapter, however, may receive several ask/tell transitions
  239         before a full loop is invoked.  This small owner operation gives that
  240         path a stable search identity and makes a later ask an explicit
  241         continuation of the same state rather than an accidental new ledger.
  242         """
  243         if self._run_state.status in {None, SearchStatus.NOT_STARTED}:
  244             self._run_state = SearchRunState(
  245                 search_id=str(uuid4())[:8],
  246                 status=SearchStatus.RUNNING,
  247             )
  248             self._config.stopping.reset()
  249             self._reset_diversity_tracker(operation="prepare_service_run")
  250             return
  251         if self._run_state.status is not SearchStatus.RUNNING:
  252             self._run_state.status = SearchStatus.RUNNING
