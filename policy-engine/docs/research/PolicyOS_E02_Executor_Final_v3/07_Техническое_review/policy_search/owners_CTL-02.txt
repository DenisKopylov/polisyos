  440         raise ValueError("Python random state must be a three-item tuple")

  442     if isinstance(version, bool) or not isinstance(version, int):
  443         raise ValueError("Python random state version is missing or invalid")
  444     if version != _PYTHON_RANDOM_STATE_VERSION:
  445         raise ValueError(

  448         )

  450         raise ValueError("Python random state vector has an incompatible shape")
  451     if any(
  452         isinstance(item, bool) or not isinstance(item, int) or not 0 <= item < 2**32
  453         for item in internal_state
  454     ):
  455         raise ValueError("Python random state vector contains an invalid value")
  456     if gauss_next is not None and (
  457         isinstance(gauss_next, bool) or not isinstance(gauss_next, (int, float))
  458     ):
  459         raise ValueError("Python random gaussian state must be numeric or null")
