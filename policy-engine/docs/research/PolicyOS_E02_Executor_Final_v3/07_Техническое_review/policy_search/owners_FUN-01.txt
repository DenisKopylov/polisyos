  648     def advance(
  649         self,
  650         ticket: FunnelTicket | str,
  651         target_level: int | None = None,
  652         policy: AdvancePolicy | None = None,
  653     ) -> FunnelOutcome:
  654         """Advance an existing ticket to the requested target level or policy."""
  655 
  656         resolved_ticket = self._resolve_ticket(ticket)
  657         execution_target = self._resolve_target_level(target_level, policy)
  658         resolved_ticket.degradation_mode = self._routing_mode(resolved_ticket.context)
  659         burn_in_calibration = (
  660             policy == "burn_in"
  661             and str(resolved_ticket.context.get("burn_in_cohort", "")) == "calibration"
  662         )
  663 
  664         while not resolved_ticket.is_terminal:
  665             next_level = self._resolve_next_level(resolved_ticket)
  666             if next_level is None:
  667                 self._mark_terminal(resolved_ticket)
  668                 break
  669             if next_level > execution_target:
  670                 break
  671             if resolved_ticket.degradation_mode == "freeze_frontier":
  672                 resolved_ticket.final_action = "defer"
  673                 resolved_ticket.is_terminal = True
  674                 break
  675 
  676             if resolved_ticket.current_level in (2, 3) and resolved_ticket.next_level == next_level:
  677                 scheduling_action = self._maybe_schedule_transition(
  678                     resolved_ticket,
  679                     execution_target=execution_target,
  680                     policy=policy,
  681                     trace_step=None,
  682                 )
  683                 if scheduling_action in {"reject", "defer", "retry_cheaper"}:
  684                     break
  685 
  686             stage = self._stages_by_level[next_level]
  687             stage_candidate = deepcopy(resolved_ticket.candidate)
  688             result = stage.evaluate(stage_candidate, self._build_stage_context(resolved_ticket))
  689             try:
  690                 subject_unchanged = candidate_payloads_equal(
  691                     stage_candidate, resolved_ticket.candidate
  692                 )
  693             except (TypeError, ValueError):
  694                 subject_unchanged = False
  695             if not subject_unchanged:
  696                 # Preserve the physical attempt and cost, but admit no score,
  697                 # work packet, routing decision or changed subject as the input.
  698                 resolved_ticket.trace.append(
  699                     FunnelTraceStep(
  700                         fidelity_level=next_level,
  701                         stage_name=result.stage_name,
  702                         objective_value=None,
  703                         is_promising=False,
  704                         duration_seconds=result.duration_seconds,
