  191 def statistical_uncertainty_from_ci_width(
  192     simulation_results: dict[str, Any],
  193     *,
  194     source: str,
  195     quantification_method: str,
  196     recommended_action: str | None = None,
  197 ) -> UncertaintyEstimate:
  198     """Build a statistical estimate without laundering missing CI data.
  199 
  200     A missing, malformed, non-finite, or negative confidence-interval width is
  201     unassessed uncertainty.  A finite zero width remains a measured zero-width
  202     interval, including when the effect itself is zero.
  203     """
  204 
  205     bootstrap = simulation_results.get("bootstrap")
  206     if not isinstance(bootstrap, Mapping) or "ci_width" not in bootstrap:
  207         return UncertaintyEstimate(
  208             level=1.0,
  209             source=f"{source}; confidence-interval width missing",
  210             quantification_method="ci_width_missing",
  211             is_reducible=True,
  212             recommended_action=recommended_action,
  213         )
  214 
  215     raw_width = bootstrap.get("ci_width")
  216     if raw_width is None:
  217         return UncertaintyEstimate(
  218             level=1.0,
  219             source=f"{source}; confidence-interval width missing",
  220             quantification_method="ci_width_missing",
  221             is_reducible=True,
  222             recommended_action=recommended_action,
  223         )
  224     if isinstance(raw_width, bool):
  225         return UncertaintyEstimate(
  226             level=1.0,
  227             source=f"{source}; confidence-interval width invalid",
  228             quantification_method="ci_width_invalid",
  229             is_reducible=True,
  230             recommended_action=recommended_action,
  231         )
  232     try:
  233         ci_width = float(raw_width)
  234     except (TypeError, ValueError, OverflowError):
  235         return UncertaintyEstimate(
  236             level=1.0,
  237             source=f"{source}; confidence-interval width invalid",
  238             quantification_method="ci_width_invalid",
  239             is_reducible=True,
  240             recommended_action=recommended_action,
  241         )
  242     if not isfinite(ci_width) or ci_width < 0.0:
  243         return UncertaintyEstimate(
  244             level=1.0,
  245             source=f"{source}; confidence-interval width invalid",
  246             quantification_method="ci_width_invalid",
  247             is_reducible=True,
  248             recommended_action=recommended_action,
  249         )
  250 
  251     if "ate" not in simulation_results:
  252         return UncertaintyEstimate(
  253             level=1.0,
  254             source=f"{source}; effect scale invalid",
  255             quantification_method="effect_invalid",
  256             is_reducible=True,
  257             recommended_action=recommended_action,
  258         )
  259     raw_effect = simulation_results["ate"]
  260     if raw_effect is None or isinstance(raw_effect, bool):
  261         return UncertaintyEstimate(
  262             level=1.0,
  263             source=f"{source}; effect scale invalid",
  264             quantification_method="effect_invalid",
  265             is_reducible=True,
  266             recommended_action=recommended_action,
  267         )
  268     try:
  269         effect = abs(float(raw_effect))
  270     except (TypeError, ValueError, OverflowError):
  271         return UncertaintyEstimate(
  272             level=1.0,
  273             source=f"{source}; effect scale invalid",
  274             quantification_method="effect_invalid",
  275             is_reducible=True,
  276             recommended_action=recommended_action,
  277         )
  278     if not isfinite(effect):
  279         return UncertaintyEstimate(
  280             level=1.0,
  281             source=f"{source}; effect scale invalid",
  282             quantification_method="effect_invalid",
  283             is_reducible=True,
  284             recommended_action=recommended_action,
  285         )
  286 
  287     if effect == 0.0:
  288         level = 0.0 if ci_width == 0.0 else 1.0
  289     else:
  290         level = min(1.0, ci_width / (2.0 * effect))
  291     return UncertaintyEstimate(
  292         level=level,
  293         source=source,
  294         quantification_method=quantification_method,
  295         is_reducible=True,
  296         recommended_action=recommended_action,
  297     )
