 1197     def _select_training_subset(self, evaluations: list[Evaluation]) -> list[Evaluation]:
 1198         filtered = self._effective_training_corpus(evaluations)
 1199         if len(filtered) <= self._config.max_train_size:
 1200             return filtered
 1201         # Keep recent half + uniformly sampled remainder for diversity.
 1202         recent_n = self._config.max_train_size // 2
 1203         recent = filtered[-recent_n:]
 1204         older = filtered[:-recent_n]
 1205         step = max(1, len(older) // max(1, self._config.max_train_size - recent_n))
 1206         sampled = older[::step][: self._config.max_train_size - recent_n]
 1207         return sampled + recent
