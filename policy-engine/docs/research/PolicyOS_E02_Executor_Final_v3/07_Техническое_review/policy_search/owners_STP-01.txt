  151     """Stop if no significant improvement for N consecutive iterations."""

  153     def __init__(

  164         self._objective_key = objective_key
  165         if tolerance_profile is not None and type(tolerance_profile) is not PlateauToleranceProfile:
  166             raise TypeError("tolerance_profile must be a validated PlateauToleranceProfile")
  167         self._tolerance_profile = tolerance_profile

  171         return "improvement_plateau"

  173     def check(self, history: list[dict[str, Any]], state: dict[str, Any]) -> StoppingCondition:
  174         if len(history) < self._patience + 1:
  175             return StoppingCondition(should_stop=False)
  176 
  177         # The baseline and patience window are one evidence set. Do not drop
  178         # absent or malformed rows, or let NaN/infinity become a convergence
  179         # signal through min() and the signed-gain comparison.
  180         values: list[float] = []
  181         for record in history:
  182             if not isinstance(record, Mapping) or self._objective_key not in record:
  183                 return StoppingCondition(should_stop=False)
  184             raw_value = record[self._objective_key]
  185             if isinstance(raw_value, bool) or not isinstance(raw_value, (int, float, Decimal)):
  186                 return StoppingCondition(should_stop=False)
  187             try:
  188                 value = float(raw_value)
  189             except (OverflowError, TypeError, ValueError):
  190                 return StoppingCondition(should_stop=False)
  191             if not isfinite(value):
  192                 return StoppingCondition(should_stop=False)
  193             values.append(value)
  194 
  195         recent_values = values[-self._patience :]
  196         historical_values = values[: -self._patience]
  197 
  198         profile = self._tolerance_profile
  199         select_best = max if profile is not None and profile.direction == "maximize" else min
  200         best_recent = select_best(recent_values)
  201         best_historical = select_best(historical_values)
  202         if profile is not None:
  203             signed_improvement = (
  204                 best_recent - best_historical
  205                 if profile.direction == "maximize"
  206                 else best_historical - best_recent
  207             )
