  222     def get_warm_start_evaluations(
  223         self,
  224         similar_runs: list[RunFingerprint],
  225         max_evals: int = 50,
  226         *,
  227         target_fingerprint: RunFingerprint,
  228     ) -> list[Evaluation]:
  229         """Reuse compatible history only after a complete target identity is supplied.
  230 
  231         Collects the best evaluations from each binding-compatible run, up to
  232         ``max_evals`` total. Incomplete or malformed target identities are
  233         refused before any history read. The result remains a list so rejected
  234         history rows stay visible to compatible callers.
  235         """
  236         if max_evals <= 0 or not similar_runs:
  237             return []
  238         if not isinstance(
  239             target_fingerprint, RunFingerprint
  240         ) or not self._has_complete_binding_basis(target_fingerprint):
  241             return []
  242 
  243         prepared: list[list[Evaluation]] = []
  244         for fp in similar_runs:
  245             if not self._binding_matches(target_fingerprint, fp):
  246                 continue
  247 
  248             rows = self._load_run_evaluations(fp)
  249             evaluations = [
  250                 self._deserialize_evaluation(row, fp.run_id, expected_fingerprint=fp)
  251                 for row in rows
  252             ]
  253             evaluations.sort(
  254                 key=lambda evaluation: self._evaluation_sort_key(evaluation, target_fingerprint)
  255             )
  256             if evaluations:
  257                 prepared.append(evaluations)
  258 
  259         # Give every compatible source its first observation, then redistribute
  260         # the remaining capacity round-robin.  This avoids the old floor quota
  261         # dropping unused capacity from short histories while retaining source
  262         # diversity for the initial warm-start corpus.
  263         all_evals: list[Evaluation] = []
  264         offsets = [0] * len(prepared)
  265         while len(all_evals) < max_evals:
  266             progressed = False
  267             for index, evaluations in enumerate(prepared):
  268                 offset = offsets[index]
  269                 if offset >= len(evaluations):
  270                     continue
  271                 all_evals.append(evaluations[offset])
  272                 offsets[index] += 1
  273                 progressed = True
  274                 if len(all_evals) >= max_evals:
  275                     break
  276             if not progressed:
  277                 break
  278 
  279         return all_evals
