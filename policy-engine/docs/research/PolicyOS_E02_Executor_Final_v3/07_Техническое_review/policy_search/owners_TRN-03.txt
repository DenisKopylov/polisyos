  308 def compute_provenance_weight(
  309     source: TransferContext,
  310     target: TransferContext,
  311     *,
  312     created_at: datetime | None = None,
  313     policy: TransferPolicy | None = None,
  314 ) -> float:
  315     """Weight transferred knowledge by scope distance and age."""
  316 
  317     active_policy = policy or TransferPolicy()
  318     if source.task_family != target.task_family:
  319         return 0.0
  320 
  321     if source.tenant_scope_status is not target.tenant_scope_status:
  322         return 0.0
  323     same_scope = source.tenant_hash == target.tenant_hash
  324     same_domain = source.domain == target.domain
  325 
  326     if same_scope and same_domain:
  327         base = 1.0
  328     elif same_scope and active_policy.allow_cross_domain:
  329         base = 0.65
  330     elif (
  331         active_policy.allow_cross_tenant
  332         and active_policy.allow_cross_domain
  333         and target.cross_tenant_opt_in
  334         and source.tenant_scope_status is TransferTenantScopeStatus.AUTHENTICATED
  335         and target.tenant_scope_status is TransferTenantScopeStatus.AUTHENTICATED
  336     ):
  337         base = 0.35
  338     else:
  339         return 0.0
  340 
  341     if created_at is None:
  342         return base
  343 
  344     age = max(timedelta(), target.timestamp - created_at)
  345     if age > timedelta(days=active_policy.ttl_days * 2):
  346         return 0.0
  347     if age > timedelta(days=active_policy.ttl_days):
  348         return base * 0.5
  349     return base
