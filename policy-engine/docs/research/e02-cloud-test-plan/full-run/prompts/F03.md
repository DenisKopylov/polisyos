## Результат для локальных архитектурных и кодовых агентов

Сформируй один переносимый пакет. Нужны следующие небольшие текстовые файлы плюс raw archive:

1. result.json: job/run, exact source/tree, manifest hash, selected profile/lock hashes, environment/setup segments, start/end, все assigned terminal rows и gate rows. Одна primary row на cell ID; все retry attempts рядом, никогда не переписывать. Для каждого cell — repo path, expected/actual blob, точный argv/cwd, collected IDs, call IDs, setup/call/teardown counts отдельно, module-collection errors/skips, PASS/FAILED/ERROR/SKIP/XFAIL/XPASS, elapsed/CPU/RSS с units и boundary, receipt_mode/receipt_status, hashes/raw refs. outcome и качество receipt — разные поля. UNRUN/timeout/source unavailable имеют null неизвестных counts, не 0. Прямые исключения call-phase записывай как failed pytest call с actual exception/traceback, не определяй AssertionError по ключевому слову.
2. cells.tsv: одна строка на назначенный cell ID: id, state, collected, calls, call_pass, call_failed, setup_error, teardown_error, skip, xfail, xpass, wall_s, rss_kib, receipt_status. Отдельные колонки collection_skip/error; failure на teardown не прибавляется к call_failed. Path берётся из исходного manifest по ID. Unknown=null. Если назначены property states, их rows/attempts/verdicts включи в result.json и отдельный properties.tsv, не смешивая с native denominator. Это удобный полный текстовый fallback; не обрезать список до top-N.
3. failures.jsonl: каждый failed/error/skip/xfailed/xpassed node или collection event: cell/attempt, exact nodeid и phase, actual message + полный traceback/longrepr, expected/actual конкретные значения из вывода, список реально missing inputs, raw hashes. Не прикладывай весь successful stdout. Причина по умолчанию cause_status=not_established; гипотеза допускается только отдельно candidate с ref на наблюдение и следующей опровергающей пробой. Не чинить failure. Не объявлять inherited без replay правильной slice base и полного input intersection.
4. handoff.md: короткое фактическое резюме counts/times, список blockers/UNRUN, точные команды воспроизведения красных, какие реальные backend/service surfaces исполнялись, что осталось fixture/mocked/неизмеренным. Связь B/LA локальный потребитель присоединит по immutable path через finding-routes.json; не генерируй 282 художественных объяснения. Раздел «измерено» отдели от «не установлено» и кандидатных направлений проверки. Все counts выведи скриптом из полного result, не вручную.
5. environment.json, source-inputs.json, per-attempt observer outputs/origins, canonical execution.json где есть, JUnit, resources, timing, raw stdout/stderr и supervisor events. Observer captures только session-finish snapshot loaded owned modules; это не полная import history/все nested backend bytes. Внеcheckout project origins — явный input/custody blocker, не молчаливый PASS. Отдельно продолжение для всех UNRUN/budget, с тем же SHA/path/blob/profile и reason.

В raw archive включи только deciding outputs, manifests и временный instrumentation source; не копии tracked source, venv/node_modules/production_data/caches/tmp databases. ```python
"""Observe an E02 pytest child without modifying tests or pytest reports."""

from __future__ import annotations

import hashlib
import json
import os
import sys
import time
from pathlib import Path
from typing import TYPE_CHECKING

if TYPE_CHECKING:
    import pytest

_root = Path(os.environ["E02_OBSERVER_DIR"])
_out = _root / str(os.getpid())
_out.mkdir(parents=True, exist_ok=False)


def _save(name: str, value: object) -> None:
    (_out / name).write_text(json.dumps(value, ensure_ascii=False) + "\n", encoding="utf-8")


def _event(kind: str, **fields: object) -> None:
    row = {"kind": kind, "pid": os.getpid(), "monotonic_ns": time.monotonic_ns(), **fields}
    with (_out / "events.jsonl").open("a", encoding="utf-8") as stream:
        stream.write(json.dumps(row, ensure_ascii=False) + "\n")


_control_names = (
    "POLISYOS_R1_REMOVE_V3_ARTIFACTID_SCALAR",
    "POLISYOS_R6_FOREIGN_CONTEXT_IDENTITY_REMOVAL",
    "POLISYOS_R7_REMOVE_ACTIVE_PROBE_SUPPRESSION",
)
_save("process.json", {"pid": os.getpid(), "ppid": os.getppid(), "argv": sys.argv,
                       "property_controls": {key: os.environ.get(key) for key in _control_names}})


def pytest_collection_finish(session: pytest.Session) -> None:
    """Record the final selected nodes in this process."""
    _save("collection.json", {
        "pid": os.getpid(),
        "nodes": [{"id": item.nodeid, "path": str(item.path)} for item in session.items],
        "config": str(session.config.inipath),
        "collect_only": bool(session.config.option.collectonly),
    })


def pytest_collectreport(report: pytest.CollectReport) -> None:
    """Preserve collection failure and module-level skip details."""
    if report.outcome != "passed":
        _event("collection_report", nodeid=report.nodeid, outcome=report.outcome,
               longrepr=str(report.longrepr))


def pytest_runtest_logreport(report: pytest.TestReport) -> None:
    """Keep each setup, call and teardown report separately."""
    _event("test_report", nodeid=report.nodeid, phase=report.when,
           outcome=report.outcome, duration_seconds=report.duration,
           wasxfail=getattr(report, "wasxfail", None),
           longrepr=str(report.longrepr) if report.longrepr is not None else None)


def pytest_sessionfinish(session: pytest.Session, exitstatus: int | pytest.ExitCode) -> None:
    """Snapshot loaded owned modules, without importing additional modules."""
    modules = []
    files = {}
    for name, module in sorted(sys.modules.items()):
        if name.split(".")[0] not in {"polisyos", "tools", "tests"}:
            continue
        origin = getattr(module, "__file__", None)
        if not origin:
            continue
        path = Path(origin).resolve()
        key = str(path)
        if key not in files:
            try:
                files[key] = {"sha256": hashlib.sha256(path.read_bytes()).hexdigest()}
            except OSError as exc:
                files[key] = {"read_error": type(exc).__name__}
        modules.append({"module": name, "origin": str(origin), "resolved": key})
    _save("origins.json", {"pid": os.getpid(), "scope": "session_finish_snapshot",
                           "modules": modules, "files": files})
    _save("session.json", {"pid": os.getpid(), "exitstatus": int(exitstatus),
                           "collected": int(session.testscollected),
                           "config": str(session.config.inipath)})
