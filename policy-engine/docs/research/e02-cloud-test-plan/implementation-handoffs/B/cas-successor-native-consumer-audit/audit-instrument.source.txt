def sha256(b):
 return hashlib.sha256(b).hexdigest()


def git(*argv, cwd=ROOT):
 return subprocess.check_output(['git', *argv], cwd=cwd)


def ident_bytes(b):
 return {'bytes': len(b), 'sha256': sha256(b)}


def source(path):
 b = git('show', SHA + ':' + path)
 return b, {'source_sha': SHA, 'path': path, 'git_blob': git('rev-parse', SHA + ':' + path).decode().strip(), **ident_bytes(b)}


def artifact(path):
 if not path.exists():
  return {'path': str(path), 'state': 'absent'}
 with path.open('rb') as f:
  before = os.fstat(f.fileno())
  b = f.read()
  after = os.fstat(f.fileno())
 return {'path': str(path), **ident_bytes(b), 'stable_size_mtime_during_read': (before.st_size, before.st_mtime_ns) == (after.st_size, after.st_mtime_ns)}

def xml_key(item):
 path=item['location'][0]
 tail=item['nodeid'].removeprefix(path+'::')
 base, bracket, parameter=tail.partition('[')
 parts=base.split('::')
 classname=path.removesuffix('.py').replace('/', '.')
 if len(parts)>1:
  classname+='.'+'.'.join(parts[:-1])
 return classname, parts[-1]+bracket+parameter


def observed(phases):
 if any(p['outcome']=='failed' for p in phases if p['when'] in {'setup','teardown'}):
  return 'ERROR'
 call=[p for p in phases if p['when']=='call']
 if call:
  cls=call[-1].get('classification',call[-1]['outcome'])
  return {'passed':'PASS','failed':'FAIL','skipped':'SKIP','xfail':'XFAIL','xpass':'XPASS','strict_xpass':'STRICT_XPASS'}.get(cls, 'UNCLASSIFIED')
 if any(p['outcome']=='skipped' for p in phases):
  return 'SKIP'
 return 'UNRUN'
