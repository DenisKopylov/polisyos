import datetime
import hashlib
import json
import shutil
import subprocess
import xml.etree.ElementTree as ET
from pathlib import Path

ROOT = Path('/workspace/e02-B-current-cas-generation')
BASE = Path('policy-engine/docs/research/e02-cloud-test-plan/implementation-handoffs/B/compiled-plan-consumer-evidence')
DEST = ROOT / BASE
HERE = Path(__file__).resolve().parent
ORIGINAL = Path('/tmp/e02-B-reopened-compiled-plan-ba48c7c5fe41')
LABEL = 'final-6d7'
transfers = []


def custody(path):
    content = path.read_bytes()
    return {'path': str(path.relative_to(ROOT)), 'bytes': len(content), 'sha256': hashlib.sha256(content).hexdigest()}


for path in sorted(ORIGINAL.rglob('*')):
    relative = path.relative_to(ORIGINAL)
    if not path.is_file() or path.is_symlink() or 'cache' in relative.parts or path.suffix == '.pyc':
        continue
    target = DEST / LABEL / relative
    target.parent.mkdir(parents=True, exist_ok=True)
    shutil.copy2(path, target)
    assert target.read_bytes() == path.read_bytes()
    transfers.append({'original_path': str(path), 'source_mode': oct(path.stat().st_mode), **custody(target)})
for name in ('plan_run_final.py', 'plan_origins_final.py', 'publish-plan-final.py'):
    target = DEST / 'literal-instruments' / (name + '.txt')
    shutil.copy2(HERE / name, target)
    transfers.append({'original_path': str(HERE / name), 'source_mode': oct((HERE / name).stat().st_mode), **custody(target)})

result = json.loads((ORIGINAL / 'result.json').read_text())

    cases.append({'classname': case.get('classname'), 'name': case.get('name'), 'duration_seconds': case.get('time'), 'outcome': outcome})
