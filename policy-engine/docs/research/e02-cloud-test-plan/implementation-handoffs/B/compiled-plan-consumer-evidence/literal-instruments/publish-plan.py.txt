def custody(path):
    content = path.read_bytes()
    return {'path': str(path.relative_to(ROOT)), 'bytes': len(content), 'sha256': hashlib.sha256(content).hexdigest()}


for label, suffix in RUNS:
    original = Path('/tmp/e02-B-reopened-compiled-plan-' + suffix)
    for path in sorted(original.rglob('*')):
        relative = path.relative_to(original)
        if not path.is_file() or path.is_symlink() or 'cache' in relative.parts or path.suffix == '.pyc':
            continue
        target = DEST / label / relative
        target.parent.mkdir(parents=True, exist_ok=True)
        shutil.copy2(path, target)
        assert target.read_bytes() == path.read_bytes()
        transfers.append({'original_path': str(path), 'source_mode': oct(path.stat().st_mode), **custody(target)})
    profile = json.loads((original / 'profile.json').read_text())
    result = json.loads((original / 'result.json').read_text())
    cases = []
    for case in ET.parse(original / 'native.xml').getroot().iter('testcase'):
        error = case.find('error')
        failure = case.find('failure')
        skip = case.find('skipped')
        outcome = 'ERROR' if error is not None else 'FAIL' if failure is not None else 'SKIP' if skip is not None else 'PASS'
        item = {'classname': case.get('classname'), 'name': case.get('name'), 'duration_seconds': case.get('time'), 'outcome': outcome}
        if failure is not None or error is not None:
            element = failure if failure is not None else error
            item['message'] = element.get('message')
        cases.append(item)
    phases = [json.loads(line) for line in (original / 'phases.jsonl').read_text().splitlines()]
    checks.append({'label': label, 'source_checkout': profile['source_checkout_before'], 'source_after': result['source_checkout_after'], 'test_checkout': profile['test_checkout_before'], 'test_after': result['test_checkout_after'], 'argv': profile['argv'], 'cwd': profile['cwd'], 'counts': result['counts'], 'returncode': result['returncode'], 'elapsed_seconds': result['elapsed_seconds'], 'actual_phase_count': len(phases), 'actual_cases': cases, 'profile': str(BASE / label / 'profile.json'), 'stdout': str(BASE / label / 'native-stdout.txt'), 'xml': str(BASE / label / 'native.xml'), 'result': str(BASE / label / 'result.json'), 'module_origins': str(BASE / label / 'module-origins.jsonl'), 'phases': str(BASE / label / 'phases.jsonl')})
