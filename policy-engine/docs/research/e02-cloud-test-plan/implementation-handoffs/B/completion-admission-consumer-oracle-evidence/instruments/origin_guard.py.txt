import hashlib, json, os, sys
from pathlib import Path

ROOT = Path('/workspace/e02-B-current-adapters/policy-engine/src')
OUT = Path(os.environ['COMPLETION_ORIGIN_OBSERVATIONS'])

def record(stage):
    rows=[]
    for name, module in sorted(sys.modules.items()):
        if name != 'polisyos' and not name.startswith('polisyos.'):
            continue
        raw=getattr(module, '__file__', None)
        if not raw:
            continue
        path=Path(raw).resolve()
        if not path.is_relative_to(ROOT):
            raise AssertionError(f'Foreign production import: {name}: {path}')
        rows.append({'module':name,'path':str(path),'sha256':hashlib.sha256(path.read_bytes()).hexdigest()})
    with OUT.open('a') as output:
        output.write(json.dumps({'stage':stage,'modules':rows},sort_keys=True)+'\n')

def pytest_collection_finish(session):
    record('collection_finish')

def pytest_sessionfinish(session, exitstatus):
    record('session_finish')
