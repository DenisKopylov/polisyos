import json, os
from pathlib import Path

MUTATION = '''
from polisyos.scientist.orchestration.engine.budget_ledger import FileBudgetLedger
from polisyos.scientist.orchestration.engine.budget_middleware import BudgetMiddleware
_original_list = BudgetMiddleware.list_completion_obligations_safe
def _ignore_pending(self, keys):
    _original_list(self, keys)
    return ()
BudgetMiddleware.list_completion_obligations_safe = _ignore_pending
FileBudgetLedger._assert_runtime_admission = lambda self, keys: None
FileBudgetLedger._intersecting_completions = staticmethod(lambda snapshot, keys, owner_epoch=None: ())
'''

def pytest_collection_modifyitems(session, config, items):
    assert len(items)==1
    target=items[0].module
    original=target._CHILD
    prefix="import runpy,sys; ns=runpy.run_path(sys.argv[1],run_name='completion_consumer_driver'); "
    assert original.startswith(prefix)
    target._CHILD=prefix+'exec('+repr(MUTATION)+"); ns['_driver'](sys.argv[2:])"
    Path(os.environ['COMPLETION_ORIGIN_OBSERVATIONS']).with_name('removal-admission.json').write_text(json.dumps({'original_child':original,'mutation':MUTATION,'mutated_child':target._CHILD,'selector':items[0].nodeid,'canonical_source_files_mutated':False},indent=2)+'\n')
