def git(*args):
    return subprocess.check_output(['git', '--git-dir=' + BARE, *args], env=ENV)

def meta(data):
    return {'bytes': len(data), 'sha256': hashlib.sha256(data).hexdigest()}

def tree(ref):
    result = {}
    for raw in git('ls-tree', '-rz', ref).split(b'\0'):
        if not raw:
            continue
        header, name = raw.split(b'\t', 1)
        mode, kind, oid = header.decode().split()
        result[name.decode()] = {'git_blob_sha1': oid, 'mode': mode, 'type': kind}
    return result

trees = {name: tree(sha) for name, sha in PINS.items()}

def delta(base, candidate):
    raw = git('diff', '--name-status', '--no-renames', '-z', base, candidate).split(b'\0')
    return [(raw[i].decode(), raw[i + 1].decode()) for i in range(0, len(raw) - 1, 2)]

def category(path):
    for start, label in [('policy-engine/docs/research/', 'research'),
                         ('policy-engine/src/', 'source'),
                         ('policy-engine/tests/', 'tests'),
                         ('policy-engine/release-fragments/', 'release')]:
        if path.startswith(start):
            return label
    return 'other'

    assert actual == oid and kind == 'blob'

proc.stdin.close()
assert proc.wait() == 0

def verify_projection(rows, base_name, candidate_name):
    expected = delta(PINS[base_name], PINS[candidate_name])
    actual_paths = [row['path'] for row in rows]
    if len(actual_paths) != len(set(actual_paths)):
        errors.append('duplicate footprint paths:' + base_name)
    if sorted(actual_paths) != sorted(path for _, path in expected):
        errors.append('footprint path set differs from exact Git diff:' + base_name)
    for row in rows:
        path = row['path']
        for role, name in [('base', base_name), ('candidate', candidate_name)]:
            declared, actual = row[role], trees[name].get(path)
            if declared is None:
                if actual is not None:
                    errors.append('unexpected existing ' + role + ':' + path)
                continue
            if actual is None or actual['git_blob_sha1'] != declared['git_blob_sha1']:
                errors.append('blob mismatch:' + role + ':' + path)
                continue
            if any(declared[key] != blobmeta[actual['git_blob_sha1']][key]
                   for key in ('bytes', 'sha256')):
                errors.append('byte/hash mismatch:' + role + ':' + path)
    return summarize(expected, candidate_name)

def summarize(changes, candidate_name):
    groups = collections.Counter(category(path) for _, path in changes)
    extensions = collections.Counter(Path(path).suffix or '<none>' for _, path in changes)
    return {'path_count': len(changes), 'statuses': dict(collections.Counter(s for s, _ in changes)),
            'categories': dict(sorted(groups.items())), 'extensions': dict(sorted(extensions.items())),
            'candidate_bytes': sum(blobmeta[trees[candidate_name][path]['git_blob_sha1']]['bytes']
                                   for _, path in changes if path in trees[candidate_name])}

def closure(base_name, candidate_name):
    changes = delta(PINS[base_name], PINS[candidate_name])
    rows = []
    for status, path in changes:
        if not path.startswith(('policy-engine/src/', 'policy-engine/tests/')):
            continue
        item = {'status': status, 'path': path}
        for role, name in [('base', base_name), ('candidate', candidate_name)]:
            entry = trees[name].get(path)
            item[role] = None if entry is None else {**entry, **blobmeta[entry['git_blob_sha1']]}
        rows.append(item)
    return rows
