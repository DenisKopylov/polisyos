import hashlib

B = 'policy-engine/docs/research/e02-cloud-test-plan/implementation-handoffs/B/'
PACKET = B + 'continuation-disposition-delta-20261007.json'
checked = set()

def git_bytes(sha, path):
    return subprocess.check_output(['git', 'show', sha + ':' + path])

def reference_bytes(ref):
    sha = ref.get('git_sha') or ref.get('commit') or ref.get('source_sha')
    assert sha and ref.get('path'), ref
    data = git_bytes(sha, ref['path'])
    if 'sha256' in ref:
        assert hashlib.sha256(data).hexdigest() == ref['sha256'], ref
    if 'bytes' in ref:
        assert len(data) == ref['bytes'], ref
    checked.add((sha, ref['path']))
    return data

def refs_in(value):
    if isinstance(value, dict):
        sha = value.get('git_sha') or value.get('commit') or value.get('source_sha')
        if sha and value.get('path') and value.get('sha256'):
            reference_bytes(value)
        for child in value.values():
            refs_in(child)
    elif isinstance(value, list):
        for child in value:
            refs_in(child)

def decode_text(data):
    if data.lstrip().startswith(b'{'):
        wrapper = json.loads(data)
        if isinstance(wrapper.get('text'), str):
            return wrapper['text']
    return data.decode()

def xml_counts(ref):
    root = ET.fromstring(decode_text(reference_bytes(ref)))
    counts = Counter()
    for case in root.iter('testcase'):
        state = ('ERROR' if case.find('error') is not None else
                 'FAIL' if case.find('failure') is not None else
                 'SKIP' if case.find('skipped') is not None else 'PASS')
        counts[state] += 1
    return dict(counts)
