def git(*args: str, data: bytes | None = None) -> bytes:
    return subprocess.check_output(['git', *args], cwd=ROOT, input=data)


def sgit(*args: str, data: bytes | None = None) -> str:
    return git(*args, data=data).decode().strip()


def js(value: object) -> bytes:
    return (json.dumps(value, ensure_ascii=False, indent=2) + '\n').encode()


def sha(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def blob(parent: str, path: str) -> dict | None:
    proc = subprocess.run(['git', 'show', parent + ':' + path], cwd=ROOT, capture_output=True)
    if proc.returncode:
        return None
    return {'git_blob_sha1': sgit('rev-parse', parent + ':' + path),
            'bytes': len(proc.stdout), 'sha256': sha(proc.stdout)}


def stage(path: str, data: bytes) -> dict:
    object_id = sgit('hash-object', '-w', '--stdin', data=data)
    sgit('update-index', '--add', '--cacheinfo', '100644', object_id, path)
    return {'path': path, 'bytes': len(data), 'sha256': sha(data), 'git_blob_sha1': object_id}


def read(parent: str, path: str) -> dict:
    return json.loads(git('show', parent + ':' + path))
