def digest(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def save(name: str, data: bytes) -> dict:
    target = OUT / name
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_bytes(data)
    return {'path': f'{REL}/{name}', 'bytes': len(data), 'sha256': digest(data)}


def js(value: object) -> bytes:
    return (json.dumps(value, ensure_ascii=False, indent=2) + '\n').encode()


def copy(name: str, path: Path, *, lossless_text: bool = False) -> dict:
    data = path.read_bytes()
    identity = {'cloud_path': str(path), 'bytes': len(data), 'sha256': digest(data)}
    if lossless_text:
        text = data.decode('utf-8')
        assert text.encode('utf-8') == data
        result = save(name, js({'schema': 'policyos.e02.lossless_text.v1', 'source': identity,
                               'encoding': 'utf-8', 'text': text}))
        result['native_bytes'] = identity
        assert json.loads((OUT / name).read_text())['text'].encode() == data
        return result
    result = save(name, data)
    result['native_bytes'] = identity
    return result
