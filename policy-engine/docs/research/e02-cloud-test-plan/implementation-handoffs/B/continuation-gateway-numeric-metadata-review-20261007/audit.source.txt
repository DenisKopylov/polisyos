def git(*args):
    return subprocess.check_output(['git', *args], cwd=ROOT)

def raw(sha, path):
    key = sha, path
    if key not in cache:
        cache[key] = git('show', sha + ':' + path)
    return cache[key]

def obj(sha, path):
    return json.loads(raw(sha, path))

def check(ok, label, detail=None):
    checks.append({'check': label, 'pass': bool(ok), 'detail': detail})

def content_ref(x, where):
    sha = x.get('git_sha') or x.get('commit') or x.get('source_sha')
    path = x.get('path')
    if isinstance(sha, str) and isinstance(path, str) and not path.startswith('/') and (x.get('bytes') is not None or x.get('sha256')):
        b = raw(sha, path)
        check(x.get('bytes') in (None, len(b)) and x.get('sha256') in (None, hashlib.sha256(b).hexdigest()), 'content-reference', {'where': where, 'sha': sha, 'path': path})

def recurse(x, where):
    if isinstance(x, dict):
        content_ref(x, where)
        for k, v in x.items():
            recurse(v, where + '/' + str(k))
    elif isinstance(x, list):
        for i, v in enumerate(x):
            recurse(v, where + '/' + str(i))

def decoded(sha, path):
    b = raw(sha, path)
    if b.lstrip().startswith(b'{'):
        x = json.loads(b)
        if 'text' in x and 'original_sha256' in x:
            d = x['text'].encode('utf-8')
            check(len(d) == x['original_bytes'] and hashlib.sha256(d).hexdigest() == x['original_sha256'], 'lossless-decode', {'path': path, 'decoded_bytes': len(d)})
            return d
    return b

def xml_counts(b):
    cases = ET.fromstring(b).findall('.//testcase')
    c = collections.Counter()
    for case in cases:
        c['ERROR' if case.find('error') is not None else 'FAIL' if case.find('failure') is not None else 'SKIP' if case.find('skipped') is not None else 'PASS'] += 1
    return dict(c), len(cases)
