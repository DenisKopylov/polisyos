import datetime, hashlib, importlib.metadata, json, os, platform, resource, subprocess, sys, time
from pathlib import Path

TASK = Path('/tmp/e02-dur-getter-oracle-0903e31d1b90')
ROOT = Path('/workspace/e02-B-current-composition')
SOURCE = '0903e31d1b9090560280c2e15aea0575da97ec20'
TREE = '67f0998c3da674ab0a3836b22336016b82b348ed'
CAPS = ['OMP_NUM_THREADS','OPENBLAS_NUM_THREADS','MKL_NUM_THREADS','NUMEXPR_NUM_THREADS','VECLIB_MAXIMUM_THREADS','BLIS_NUM_THREADS']

def git(*args):
    return subprocess.check_output(['git','-C',str(ROOT),*args])

def identity():
    return {'sha':git('rev-parse','HEAD').decode().strip(),
            'tree':git('rev-parse','HEAD^{tree}').decode().strip(),
            'status':git('status','--porcelain').decode()}

def child():
    import pytest
    before = identity()
    assert before == {'sha':SOURCE,'tree':TREE,'status':''}, before
    observed = {'source_before':before,'started_utc':datetime.datetime.now(datetime.UTC).isoformat(),
                'python':sys.version,'executable':sys.executable,'platform':platform.platform(),
                'environment':{name:os.environ.get(name) for name in CAPS+['PYTHONPATH','GIT_OBJECT_DIRECTORY','GIT_ALTERNATE_OBJECT_DIRECTORIES','GIT_INDEX_FILE','PYTEST_DISABLE_PLUGIN_AUTOLOAD']},
                'versions':{name:importlib.metadata.version(name) for name in ['pytest','jax','jaxlib','numpy','pydantic']}}
    class Custody:
        def pytest_sessionfinish(self, session, exitstatus):
            origins=[]
            for name, module in sorted(sys.modules.items()):
                path=getattr(module,'__file__',None)
                if not name.startswith('polisyos') or not path:
                    continue
                file=Path(path).resolve()
                assert file.is_relative_to(ROOT), (name,str(file))
                relative=str(file.relative_to(ROOT))
                actual=file.read_bytes();expected=git('show',SOURCE+':'+relative)
                assert actual==expected,(name,relative)
                origins.append({'module':name,'path':relative,'bytes':len(actual),
                                'sha256':hashlib.sha256(actual).hexdigest()})
            after=identity()
            assert after==before,after
            observed.update({'source_after':after,'actual_source_origins':origins,
                             'origin_count':len(origins),'origin_mismatches':[],
                             'pytest_exit':int(exitstatus)})
            (TASK/'profile.json').write_text(json.dumps(observed,indent=2)+'\n')
            print('GETTER_ORIGINS',len(origins),SOURCE)
    return pytest.main(['-c','/dev/null','--noconftest','-vv','-s','--tb=short',
                        '-p','no:cacheprovider','--basetemp',str(TASK/'fixtures-final'),
                        '--junitxml',str(TASK/'native.xml'),
                        str(TASK/'test_independent_getter_consumer.py')],plugins=[Custody()])

if len(sys.argv)>1 and sys.argv[1]=='child':
    sys.exit(child())
env=os.environ.copy()
for name in CAPS:
    env.pop(name,None)
env.update({'GIT_OBJECT_DIRECTORY':'/tmp/e02-B-jit-data-getattr-20261007/git-objects',
            'GIT_ALTERNATE_OBJECT_DIRECTORIES':'/workspace/polisyos/.git/objects',
            'GIT_INDEX_FILE':'/tmp/e02-B-jit-data-getattr-20261007/index',
            'PYTHONPATH':str(ROOT/'policy-engine/src')+':'+str(ROOT/'policy-engine/product'),
            'PYTEST_DISABLE_PLUGIN_AUTOLOAD':'1','PYTHONDONTWRITEBYTECODE':'1',
            'POLISYOS_METRICS_PORT':'0','JAX_PLATFORMS':'cpu'})
env.pop('PYTEST_ADDOPTS',None)
argv=[sys.executable,str(Path(__file__)), 'child']
start=time.monotonic()
with (TASK/'native.txt').open('wb') as stdout:
    process=subprocess.Popen(argv,cwd=ROOT/'policy-engine',env=env,stdout=stdout,stderr=subprocess.STDOUT)
    _,status,usage=os.wait4(process.pid,0)
    code=os.waitstatus_to_exitcode(status)
    process.returncode=code
wrapper={'argv':argv,'cwd':str(ROOT/'policy-engine'),'actual_exit':code,
         'wall_s':time.monotonic()-start,'max_rss_kib':usage.ru_maxrss,
         'source_expected':SOURCE,'tree_expected':TREE,
         'input_files':[{'path':str(p),'bytes':p.stat().st_size,'sha256':hashlib.sha256(p.read_bytes()).hexdigest()}
                        for p in [Path(__file__),TASK/'test_independent_getter_consumer.py']],
         'numeric_caps':{name:env.get(name) for name in CAPS},
         'scope':'independent3cases real JAX/compiler; source0903 held; no canonical writes, service, setup, quota or cleanup'}
(TASK/'wrapper.json').write_text(json.dumps(wrapper,indent=2)+'\n')
print(json.dumps(wrapper,indent=2))
sys.exit(code)
