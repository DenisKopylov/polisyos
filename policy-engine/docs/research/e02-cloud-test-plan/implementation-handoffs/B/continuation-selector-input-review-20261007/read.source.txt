def git(*args):
    return subprocess.check_output(['git', *args], cwd=ROOT)


def raw(sha, path):
    return git('show', sha + ':' + path)


def ref(sha, path):
    b = raw(sha, path)
    return {'git_sha': sha, 'path': path, 'git_blob': git('rev-parse', sha + ':' + path).decode().strip(), 'bytes': len(b), 'sha256': hashlib.sha256(b).hexdigest()}
