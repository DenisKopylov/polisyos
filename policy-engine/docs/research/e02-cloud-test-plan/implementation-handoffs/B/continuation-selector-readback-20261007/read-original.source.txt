def git(*args):
    return subprocess.check_output(['git', *args], cwd=ROOT)


def raw(sha, path):
    return git('show', sha + ':' + path)


def describe(content):
    return {'bytes': len(content), 'sha256': hashlib.sha256(content).hexdigest()}


def ref(sha, path):
    content = raw(sha, path)
    return {'git_sha': sha, 'path': path, 'git_blob': git('rev-parse', sha + ':' + path).decode().strip(), **describe(content)}


def stable(path):
    first = path.read_bytes()
    second = path.read_bytes()
    assert first == second, str(path)
    return first


def local_ref(path, content):
    return {'path': str(path), **describe(content), 'custody': 'Actual ignored root raw; portable root publication not established by this leaf. Complete relevant input projection and Git source refs are retained in this review.'}
