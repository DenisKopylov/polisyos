def git(*args):
    return subprocess.check_output(['git', *args], cwd=REPO)
def inventory():
    paths = git('ls-files', '--', 'policy-engine/src', 'policy-engine/pytest.ini',
                'policy-engine/pyproject.toml', 'policy-engine/uv.lock', 'policy-engine/'+TEST).decode().splitlines()
    rows = []
    for name in paths:

import ast
import polisyos.scientist.orchestration.engine.async_executor as executor_module
original_source = git('show', SHA+':policy-engine/src/polisyos/scientist/orchestration/engine/async_executor.py')
tree = ast.parse(original_source)
changes = []
class RemoveCacheOffLoop(ast.NodeTransformer):
    def __init__(self):
        self.parents = []
    def visit_AsyncFunctionDef(self, node):
        self.parents.append(node.name)
        result = self.generic_visit(node)
        self.parents.pop()
        return result
    def visit_Call(self, node):
        if isinstance(node.func, ast.Name) and node.func.id == 'run_blocking_async' and any(x in {'_recover_cache','_put_cache_entry','_execute_node'} for x in self.parents):
            changes.append({'line':node.lineno,'function':self.parents[-1],'before':node.func.id,'after':'_e02_onloop_cache_io'})
            node.func.id = '_e02_onloop_cache_io'
        return self.generic_visit(node)
RemoveCacheOffLoop().visit(tree)
assert len(changes) == 4, changes
async def _e02_onloop_cache_io(func, *args, **kwargs):
    kwargs.pop('timeout_seconds', None)
    kwargs.pop('unbounded', None)
    return func(*args, **kwargs)
executor_module.__dict__['_e02_onloop_cache_io'] = _e02_onloop_cache_io
exec(compile(ast.fix_missing_locations(tree),executor_module.__file__,'exec'),executor_module.__dict__)
print('B52_LITERAL_REMOVAL '+json.dumps({'source_sha256':hashlib.sha256(original_source).hexdigest(),'changes':changes,'replacement':'async bridge runs unchanged real cache function on caller loop; only helper timeout/unbounded transport options omitted','source_files_unchanged':True}),flush=True)

import pytest

origins = []
for name, module in sorted(sys.modules.items()):
    if not name.startswith('polisyos'):
        continue
    filename = getattr(module,'__file__',None)
    if filename is None:
        continue
    path = pathlib.Path(filename).resolve()
    rel = path.relative_to(REPO.resolve()).as_posix()
    data = path.read_bytes()
    assert data == git('show', SHA+':'+rel), name
    origins.append({'module':name,'path':str(path),'source_path':rel,'sha256':hashlib.sha256(data).hexdigest(),'bytes':len(data)})
(OUT/'module-origins.json').write_text(json.dumps(origins,indent=2)+'\n')
context.update({'source_equal_after':True,'all_loaded_canonical_modules_git_equal':True,
                'loaded_canonical_count':len(origins),'source_head_after':git('rev-parse','HEAD').decode().strip(),
                'pytest_argv':args,'pytest_return_code':code})
(OUT/'source-context.json').write_text(json.dumps(context,indent=2)+'\n')
sys.exit(code)
