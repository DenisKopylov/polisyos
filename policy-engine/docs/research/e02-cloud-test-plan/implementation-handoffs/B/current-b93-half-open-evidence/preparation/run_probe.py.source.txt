"""Capture an independent external consumer against an immutable canonical source tree."""

from __future__ import annotations

import argparse
import datetime
import hashlib
import importlib.metadata
import json
import os
import subprocess
import sys
import time
from pathlib import Path

ROOT = Path("/workspace/e02-B-current-coordination")
SOURCE = "1862c021df74a5c5815bc5cd21a20e61b30132b6"
BASE = Path(__file__).parent
CAPS = (
    "OMP_NUM_THREADS",
    "OPENBLAS_NUM_THREADS",
    "MKL_NUM_THREADS",
    "NUMEXPR_NUM_THREADS",
    "VECLIB_MAXIMUM_THREADS",
    "BLIS_NUM_THREADS",
)
SCOPED = [
    "policy-engine/src/polisyos/fabric/connectors/resilience/_bounded_registry.py",
    "policy-engine/src/polisyos/fabric/connectors/resilience/circuit_breaker.py",
    "policy-engine/src/polisyos/fabric/connectors/resilience/rate_limiter.py",
    "policy-engine/tests/unit/fabric/connectors/test_resilience.py",
    "policy-engine/tests/unit/fabric/connectors/test_resilience_e02.py",
    "policy-engine/pyproject.toml",
    "policy-engine/uv.lock",
    "policy-engine/pytest.ini",
]


def git(*args):
    return subprocess.check_output(["git", *args], cwd=ROOT).decode().strip()


def identity(path):
    data = path.read_bytes()
    return {
        "path": str(path),
        "sha256": hashlib.sha256(data).hexdigest(),
        "bytes": len(data),
    }


def source_check(loaded=False):
    paths = set(SCOPED)
    if loaded:
        for name, module in sorted(sys.modules.items()):
            if name == "polisyos" or name.startswith("polisyos."):
                file = getattr(module, "__file__", None)
                if file is None:
                    continue
                path = Path(file)
                assert path.is_relative_to(ROOT / "policy-engine/src"), (name, file)
                assert path.suffix == ".py", (name, file)
                paths.add(str(path.relative_to(ROOT)))
    refs = []
    for relative in sorted(paths):
        path = ROOT / relative
        expected = subprocess.check_output(
            ["git", "show", SOURCE + ":" + relative], cwd=ROOT
        )
        assert path.read_bytes() == expected, relative
        refs.append(
            {
                **identity(path),
                "source_commit": SOURCE,
                "relative_path": relative,
                "git_blob": git("rev-parse", SOURCE + ":" + relative),
            }
        )
    return {
        "executed_head": git("rev-parse", "HEAD"),
        "executed_tree": git("rev-parse", "HEAD^{tree}"),
        "semantic_source_sha": SOURCE,
        "semantic_source_tree": git("rev-parse", SOURCE + "^{tree}"),
        "scope": "actual loaded source bytes; metadata HEAD may advance",
        "refs": refs,
    }


parser = argparse.ArgumentParser()
parser.add_argument("--mode", choices=["native", "remove"], required=True)
parser.add_argument("--child", action="store_true")
args = parser.parse_args()
out = BASE / args.mode
probe = BASE / "test_half_open_consumer.py"
if args.child:
    import pytest

    before = source_check()
    select = str(probe) + (
        "::test_actual_half_open_provider_retains_regulator[ttl-and-pressure]"
        if args.mode == "remove"
        else ""
    )
    code = pytest.main(
        [
            "-c",
            "/dev/null",
            "--confcutdir=" + str(BASE),
            "-o",
            "addopts=",
            "-o",
            "cache_dir=" + str(out / "cache"),
            "-q",
            "-s",
            select,
            "--basetemp=" + str(out / "pytest-tmp"),
            "--junitxml=" + str(out / "native.xml"),
        ]
    )
    packet = {
        "before": before,
        "after": source_check(loaded=True),
        "pytest_exit": code,
        "probe": identity(probe),
        "driver": identity(Path(__file__)),
        "installed_distributions": {
            name: importlib.metadata.version(name)
            for name in ["pytest", "pytest-asyncio", "pydantic", "opentelemetry-api"]
        },
    }
    (out / "source-context.json").write_text(json.dumps(packet, indent=2) + "\n")
    raise SystemExit(code)
out.mkdir(exist_ok=False)
env = os.environ.copy()
for key in CAPS:
    env.pop(key, None)
env.update(
    PYTHONPATH=str(ROOT / "policy-engine/src"),
    PYTHONDONTWRITEBYTECODE="1",
    PYTEST_DISABLE_PLUGIN_AUTOLOAD="1",
    POLISYOS_METRICS_PORT="0",
)
env.pop("PYTEST_ADDOPTS", None)
env["E02_REMOVE_PROTECTION"] = "1" if args.mode == "remove" else "0"
argv = [sys.executable, str(Path(__file__)), "--mode", args.mode, "--child"]
packet = {
    "argv": argv,
    "cwd": str(ROOT / "policy-engine"),
    "source_before": source_check(),
    "source_sha": SOURCE,
    "mode": args.mode,
    "environment": {
        key: env.get(key)
        for key in (
            *CAPS,
            "PYTHONPATH",
            "PYTEST_DISABLE_PLUGIN_AUTOLOAD",
            "PYTEST_ADDOPTS",
            "PYTHONDONTWRITEBYTECODE",
            "POLISYOS_METRICS_PORT",
            "E02_REMOVE_PROTECTION",
        )
    },
    "python": sys.version,
    "interpreter": sys.executable,
    "probe_input": identity(probe),
    "driver_input": identity(Path(__file__)),
    "start_utc": datetime.datetime.now(datetime.UTC).isoformat(),
}
start = time.monotonic()
with (out / "native.txt").open("xb") as log:
    child = subprocess.Popen(
        argv, cwd=packet["cwd"], env=env, stdout=log, stderr=subprocess.STDOUT
    )
    _, status, usage = os.wait4(child.pid, 0)
    child.returncode = os.waitstatus_to_exitcode(status)
packet.update(
    exit_code=child.returncode,
    wall_seconds=time.monotonic() - start,
    max_rss_kib=usage.ru_maxrss,
    resource_method="direct Linux wait4 child rusage",
    source_after=source_check(),
    end_utc=datetime.datetime.now(datetime.UTC).isoformat(),
    output=identity(out / "native.txt"),
)
(out / "wrapper.json").write_text(json.dumps(packet, indent=2) + "\n")
print(
    json.dumps(
        {
            k: packet[k]
            for k in ["mode", "exit_code", "wall_seconds", "max_rss_kib", "output"]
        },
        indent=2,
    )
)
raise SystemExit(child.returncode)
