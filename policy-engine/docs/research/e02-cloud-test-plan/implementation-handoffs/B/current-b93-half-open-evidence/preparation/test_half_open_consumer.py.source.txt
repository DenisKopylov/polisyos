"""Independent actual HALF_OPEN provider retention across TTL and lookup pressure."""

from __future__ import annotations

import asyncio
import json
import os

import pytest
from polisyos.fabric.connectors.resilience import _bounded_registry as registry_module
from polisyos.fabric.connectors.resilience import circuit_breaker as circuit_module
from polisyos.fabric.connectors.resilience._bounded_registry import (
    BoundedResourceRegistryCapacityError,
)


@pytest.mark.parametrize(
    ("ttl", "advance", "capacity"),
    [(1.0, 2.0, 1), (1000.0, 0.0, 1), (1.0, 2.0, 2)],
    ids=["ttl-and-pressure", "pressure-without-expiry", "spare-protected-slot"],
)
def test_actual_half_open_provider_retains_regulator(
    monkeypatch, tmp_path, ttl, advance, capacity
):
    now = [0.0]
    monkeypatch.setattr(registry_module, "_monotonic", lambda: now[0])
    monkeypatch.setattr(circuit_module, "_monotonic", lambda: now[0])
    removed = os.environ.get("E02_REMOVE_PROTECTION") == "1"
    if removed:
        # Counterfactual only: keep registry/decorator/lease markers and real provider intact.
        monkeypatch.setattr(
            registry_module.BoundedResourceRegistry,
            "_entry_is_quiescent",
            lambda self, key, value: True,
        )
    config = circuit_module.CircuitBreakerConfig(
        failure_threshold=1,
        success_threshold=1,
        timeout_seconds=10.0,
        half_open_max_calls=1,
        window_size_seconds=60.0,
        min_throughput=1,
    )
    physical = tmp_path / "physical-provider.jsonl"
    observations = []
    pending = 0
    peak_pending = 0

    async def scenario():
        nonlocal pending, peak_pending
        started, release = asyncio.Event(), asyncio.Event()

        @circuit_module.with_circuit_breaker(config, circuit_id=lambda key, action: key)
        async def provider(key, action):
            nonlocal pending, peak_pending
            pending += 1
            peak_pending = max(peak_pending, pending)
            with physical.open("a") as output:
                output.write(
                    json.dumps({"key": key, "action": action, "pending": pending})
                    + "\n"
                )
                output.flush()
                os.fsync(output.fileno())
            try:
                if action == "trip":
                    raise ConnectionError("actual trip body")
                if action == "probe":
                    started.set()
                    await release.wait()
                return {"key": key, "action": action}
            finally:
                pending -= 1

        registry = provider._circuit_breakers
        registry._max_items = capacity
        registry._ttl_seconds = ttl
        with pytest.raises(ConnectionError, match="actual trip body"):
            await provider("owned", "trip")
        original = registry.snapshot()["owned"]
        assert original.state is circuit_module.CircuitState.OPEN
        now[0] = 10.0
        active = asyncio.create_task(provider("owned", "probe"))
        await asyncio.wait_for(started.wait(), 5.0)
        generation = original._generation
        issued = sorted(original._half_open_leases)
        assert original.state is circuit_module.CircuitState.HALF_OPEN
        assert len(issued) == 1 and original._active_attempts == set(issued)
        assert registry._leases == {"owned": 1} and pending == 1
        now[0] += advance
        try:
            same = registry.get_or_create(
                "owned", lambda: circuit_module.CircuitBreaker("owned", config)
            )
            same_outcome = None
            try:
                same_outcome = await provider("owned", "illegal-second-probe")
            except Exception as error:
                same_outcome = type(error).__name__
            pressure_outcome = None
            try:
                pressure_outcome = await provider("new-key", "pressure")
            except Exception as error:
                pressure_outcome = type(error).__name__
            snapshot = registry.snapshot()
            during = {
                "clock": now[0],
                "ttl": ttl,
                "capacity": capacity,
                "same_object": same is original,
                "same_generation": same._generation == generation,
                "same_outcome": same_outcome,
                "pressure_outcome": pressure_outcome,
                "old_state": original.state.value,
                "issued_tokens": issued,
                "old_tokens": sorted(original._half_open_leases),
                "old_active": sorted(original._active_attempts),
                "registry_keys": sorted(snapshot),
                "registry_leases": dict(registry._leases),
                "protected_count": registry._protected_count_locked(),
                "quiescent_count": registry._quiescent_count_locked(),
                "pending": pending,
                "peak_pending": peak_pending,
                "physical_rows": [
                    json.loads(line) for line in physical.read_text().splitlines()
                ],
                "removed_protection": removed,
            }
            observations.append({"during": during})
        finally:
            release.set()
            result = await active
        observations[-1]["released"] = {
            "result": result,
            "pending": pending,
            "old_state": original.state.value,
            "old_tokens": sorted(original._half_open_leases),
            "old_active": sorted(original._active_attempts),
            "leases": dict(registry._leases),
        }
        # The real completed HALF_OPEN success returns to neutral CLOSED and frees its slot.
        now[0] += ttl + 100.0
        recovery = await provider("recovery-key", "capacity-after-release")
        observations[-1]["recovery"] = {
            "result": recovery,
            "pending": pending,
            "keys": sorted(registry.snapshot()),
            "leases": dict(registry._leases),
            "physical_rows": [
                json.loads(line) for line in physical.read_text().splitlines()
            ],
        }
        packet = {
            "schema": "e02.b93.half_open_observation.v1",
            "observations": observations,
        }
        (tmp_path / "observations.json").write_text(json.dumps(packet, indent=2) + "\n")
        print("B93_ACTUAL " + json.dumps(packet, sort_keys=True))
        # Consumer invariant, evaluated after actual effects and all finally cleanup are recorded.
        assert during["same_object"] and during["same_generation"], during
        assert during["same_outcome"] == "CircuitOpenError", during
        assert during["old_tokens"] == issued and during["old_active"] == issued, during
        assert during["pending"] == 1, during
        assert during["protected_count"] <= capacity, during
        if capacity == 1:
            assert (
                during["pressure_outcome"]
                == BoundedResourceRegistryCapacityError.__name__
            ), during
            assert during["physical_rows"] == [
                {"key": "owned", "action": "trip", "pending": 1},
                {"key": "owned", "action": "probe", "pending": 1},
            ], during
            assert during["peak_pending"] == 1, during
        else:
            assert during["pressure_outcome"] == {
                "key": "new-key",
                "action": "pressure",
            }, during
            assert during["physical_rows"][-1] == {
                "key": "new-key",
                "action": "pressure",
                "pending": 2,
            }, during
        assert observations[-1]["released"]["old_state"] == "CLOSED"
        assert observations[-1]["released"]["old_tokens"] == []
        assert observations[-1]["released"]["old_active"] == []
        assert observations[-1]["released"]["leases"] == {}
        assert recovery == {"key": "recovery-key", "action": "capacity-after-release"}
        assert pending == 0 and registry._leases == {}

    asyncio.run(scenario())
