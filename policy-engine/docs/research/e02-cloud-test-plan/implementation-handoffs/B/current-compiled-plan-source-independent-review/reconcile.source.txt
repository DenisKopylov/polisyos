"""Reconcile immutable B43 source custody, not runtime behavior or closure.

The enumerated changed paths, owner context files and canonical card are the
complete inputs of this metadata check. Git reads are disclosed explicitly;
there are no product imports, pytest runs, gate runs or workspace projections.
Manual source-flow conclusions live in review.json. AST presence/equality is
not an independent scientific/runtime oracle.
"""

import ast
import hashlib
import json
import re
import subprocess
import sys
from pathlib import Path

ROOT = Path(subprocess.check_output(["git", "rev-parse", "--show-toplevel"], text=True).strip())
BASE = "97d0684c1f2765fc67d40331a8a2602a08c1f2e0"
FIRST = "f563fde9238a08b55135f167ddf0d6fa154690c5"
FIX = "e69f25bf273758b10d5b205a28f9349804fdc7d1"
FINAL = "9ce4d3fefd35b4e8d2982ebf8cf12a5e999ca640"
CARD_SHA = "69780761ae091d8fcc6ab8778c7f5f7227eeef0b"
PREFIX = "policy-engine/"
CHAIN = PREFIX + "src/polisyos/foundry/methods/artifacts/_chain.py"
PARTS = PREFIX + "src/polisyos/foundry/methods/artifacts/parts.py"
STUB = PREFIX + "src/polisyos/foundry/methods/components/composer.pyi"
CARD = PREFIX + "docs/plans/active/agent-packages/PolicyOS_E02_Combined_Agent_Package/bundles/CMP-01.md"
READS = []


def git(*args):
    return subprocess.check_output(["git", "-C", str(ROOT), *args])


def read(sha, path):
    data = git("show", f"{sha}:{path}")
    READS.append({"sha": sha, "path": path, "bytes": len(data), "sha256": hashlib.sha256(data).hexdigest()})
    return data


def definition(data, name, container=None):
    nodes = ast.parse(data).body
    if container:
        nodes = next(n for n in nodes if isinstance(n, ast.ClassDef) and n.name == container).body
    return next(n for n in nodes if isinstance(n, (ast.ClassDef, ast.FunctionDef, ast.AsyncFunctionDef)) and n.name == name)


def dump(node):
    return ast.dump(node, include_attributes=False)


changed = git("diff", "--name-only", BASE, FINAL).decode().splitlines()
assert len(changed) == 10, changed
final_inputs = {path: read(FINAL, path) for path in changed}
base_chain = read(BASE, CHAIN)
first_parts = read(FIRST, PARTS)
fix_parts = read(FIX, PARTS)
first_stub = read(FIRST, STUB)
fix_stub = read(FIX, STUB)
fix_chain = read(FIX, CHAIN)
assert dump(definition(base_chain, "ChainArtifact")) == dump(definition(final_inputs[CHAIN], "ChainArtifact"))
assert dump(definition(first_parts, "load_compiled_chain_plan")) == dump(definition(fix_parts, "load_compiled_chain_plan"))
assert fix_parts == final_inputs[PARTS]

context_paths = [
    "src/polisyos/foundry/methods/artifacts/__init__.py",
    "src/polisyos/foundry/methods/base.py",
    "src/polisyos/foundry/methods/selection/registry.py",
    "src/polisyos/foundry/methods/components/linker.py",
    "src/polisyos/foundry/methods/backends/chain_executor.py",
    "src/polisyos/foundry/methods/backends/async_chain_executor.py",
    "src/polisyos/core/artifacts/store.py",
    "src/polisyos/core/artifacts/manifest_profile.py",
    "src/polisyos/core/artifacts/manifest.py",
    "src/polisyos/core/canon/canon_json.py",
    "src/polisyos/common/canonical.py",
]
for path in context_paths:
    read(FINAL, PREFIX + path)
card = read(CARD_SHA, CARD)
raw_criterion = re.search(rb"<!-- SOURCE_BEGIN B:B43 -->(.*?)<!-- SOURCE_END B:B43 -->", card, re.S).group(1)
assert hashlib.sha256(raw_criterion).hexdigest() == "10b7eabc7d65e7ee68eff8d1b3ee523ce960f00f696c34edf1aae65cdd67a8c9"


def has_stub_field(data):
    return any(isinstance(n, ast.AnnAssign) and isinstance(n.target, ast.Name) and n.target.id == "_insertion_order" for n in definition(data, "MethodNode").body)


assert not has_stub_field(first_stub)
assert has_stub_field(fix_stub)
assert has_stub_field(final_inputs[STUB])
fix_from_chain = definition(fix_chain, "from_chain", "CompiledChainPlan")
final_from_chain = definition(final_inputs[CHAIN], "from_chain", "CompiledChainPlan")
# The sole production successor adds a local annotation to the same assignment.
assert isinstance(fix_from_chain.body[1], ast.Assign)
assert isinstance(final_from_chain.body[1], ast.AnnAssign)
assert dump(fix_from_chain.body[1].value) == dump(final_from_chain.body[1].value)
assert [dump(n) for n in fix_from_chain.body[2:]] == [dump(n) for n in final_from_chain.body[2:]]

result = {
    "mode": "immutable Git/AST metadata only; no native/product import/gate execution",
    "reviewer": "/root/dur_probe",
    "reviewer_head": git("rev-parse", "HEAD").decode().strip(),
    "reviewer_tree": git("rev-parse", "HEAD^{tree}").decode().strip(),
    "python": {"executable": sys.executable, "version": sys.version},
    "source": {"base": BASE, "first": FIRST, "fix": FIX, "final": FINAL, "tree": git("rev-parse", FINAL + "^{tree}").decode().strip()},
    "enumerated_changed_paths": changed,
    "successful_Git_reads": READS,
    "failed_reads": [],
    "metadata_outcome": "PASS",
    "recomputed_quantities": {
        "legacy_ChainArtifact_AST_unchanged": True,
        "strict_loader_AST_unchanged_first_to_fix": True,
        "selected_profile_writer_fix_bytes_equal_final": True,
        "first_stub_insertion_field": False,
        "fixed_and_final_stub_insertion_field": True,
        "fix_to_final_from_chain_only_local_annotation": True,
        "canonical_raw_capture_sha256": hashlib.sha256(raw_criterion).hexdigest(),
    },
    "unresolved_by_construction": [
        "No runtime tests/gates are executed by this reviewer; AST comparisons do not prove execution semantics.",
        "Only the ten-path owner delta and eleven named adjacent owner files are selected. This is not an all-repository/backend conformance verdict.",
        "Current registry stability, caller execution inputs and permission remain explicit operational premises.",
        "Native artifacts and execution outcomes are independently owned and must be read under their exact source/input refs before runtime acceptance.",
    ],
    "closure_ids": [],
}
print(json.dumps(result, ensure_ascii=False, indent=2))
