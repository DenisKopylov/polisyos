def digest(data):
    return hashlib.sha256(data).hexdigest()


def git(*args):
    return subprocess.check_output(["git", "-C", str(ROOT), *args])


def identity(data):
    return {"bytes": len(data), "sha256": digest(data)}


def checkpoint():
    result = {
        "sha": git("rev-parse", "HEAD").decode().strip(),
        "tree": git("rev-parse", "HEAD^{tree}").decode().strip(),
        "branch": git("symbolic-ref", "HEAD").decode().strip(),
        "tracked_status": git("status", "--porcelain=v1", "--untracked-files=no").decode(),
    }
    assert result["sha"] == SHA and result["tree"] == TREE
    assert result["branch"] == "refs/heads/codex/e02-B-current-coordination"
    assert result["tracked_status"] == ""
    return result

    first = path.stat()
    data = path.read_bytes()
    last = path.stat()
    assert digest(data) == expected, path
    assert (first.st_dev, first.st_ino, first.st_size, first.st_mtime_ns) == (
        last.st_dev, last.st_ino, last.st_size, last.st_mtime_ns
    ), path
    records.append({"path": str(path), **identity(data), "stable_during_read": True})
    data_by_path[path] = data

            tracked_blobs[path.decode()] = oid
