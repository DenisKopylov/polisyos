if journal!=i['runtime_phase_reports']:errors.append('complete journal/inventory phase mismatch')
if w['output_sha256']!=sha(stdout):errors.append('wrapper stdout hash mismatch')
for point in [i['git_at_start'],i['git_at_finish']]:
    if point['head']!=SOURCE or point['tree']!=TREE or point['tracked_status']!='':errors.append('actual start/finish source mismatch')

    node=item['nodeid']; items[node]=item
    parts=node.split('::'); file=parts[0] or item['location'][0]
    while file.startswith('../'):file=file[3:]
    dotted=file.removesuffix('.py').replace('/','.')
    classname=dotted+('.'+'.'.join(parts[1:-1]) if len(parts)>2 else '')
    keys[(classname,parts[-1])].append(node)

if len(items)!=len(i['session_items_before_execution']):errors.append('duplicate inventory nodeid')

    if len(nodes)!=1:errors.append('unresolved/ambiguous XML identity '+repr(key));continue
    node=nodes[0];seen.add(node);reports=phases[node]
    outcome='ERROR' if t.find('error') is not None else 'FAIL' if t.find('failure') is not None else 'SKIP' if t.find('skipped') is not None else 'PASS'
    inferred='ERROR' if any(r['when']!='call' and r['outcome']=='failed' for r in reports) else 'FAIL' if any(r['when']=='call' and r['outcome']=='failed' for r in reports) else 'SKIP' if any(r['outcome']=='skipped' for r in reports) else 'PASS'
    if outcome!=inferred:mismatch.append(node)
    casecounts[outcome]+=1;perfile[items[node]['location'][0]][outcome]+=1
    if outcome=='FAIL':
        f=t.find('failure');failure_rows.append({'nodeid':node,'source_location':items[node]['location'],'phases':[{'when':r['when'],'outcome':r['outcome'],'longrepr_sha256':r['longrepr_sha256']} for r in reports],'message':f.attrib.get('message'),'full_failure_text':f.text,'body_entered':any(r['when']=='call' for r in reports),'source_cause':'not inferred; canonical owner diagnoses separately.     if outcome=='ERROR':

        errors_by_type[e.attrib.get('message','').split(':',1)[0]]+=1
        signature='literal_ENOSPC_or_errno28' if 'No space left on device' in txt or '[Errno 28]' in txt else 'numbered_tmpdir_refusal_without_literal_errno28' if 'could not create numbered dir' in txt else 'other'
        error_signatures[signature]+=1
if seen!=set(items):errors.append('XML/inventory full identity set mismatch')
if mismatch:errors.append('per-case XML/phase outcome mismatch')
suite=list(x.iter('testsuite')); declared={k:sum(int(s.attrib.get(k,0)) for s in suite) for k in ['tests','failures','errors','skipped']}
