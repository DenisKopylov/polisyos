from pathlib import Path
import collections
import hashlib
import json
import re
import subprocess
import xml.etree.ElementTree as ET

ROOT = Path('/workspace/e02-B-current-coordination')
RAW = ROOT / '.polisyos/e02-B-current/raw'
TMP = Path('/tmp/e02-B-successor-final/raw/review')
OUT = ROOT / 'policy-engine/docs/research/e02-cloud-test-plan/implementation-handoffs/B/current-dd-terminal-evidence'
SOURCE = 'dd9584728bd96f204983615216efdbc58117e713'
TREE = '7bf4be4a117762c8d37d4babcc214a2b28c2113e'

def git(*args):
    return subprocess.check_output(['git', *args], cwd=ROOT, text=True).strip()

def identity(path):
    payload = path.read_bytes()
    return {'path': str(path), 'bytes': len(payload), 'sha256': hashlib.sha256(payload).hexdigest()}

assert git('rev-parse', 'HEAD') == SOURCE
assert git('rev-parse', 'HEAD^{tree}') == TREE
assert git('symbolic-ref', '--short', 'HEAD') == 'codex/e02-B-current-coordination'
assert not git('status', '--porcelain')
partition_path = RAW / 'successor-final-dd958-case-partition.json'
partition = json.loads(partition_path.read_text())
assert partition['source_sha'] == SOURCE and partition['source_tree'] == TREE
assert partition['parse_state'] == 'COMPLETE'
assert len(partition['reconciliation_checks']) == 27
assert all(partition['reconciliation_checks'].values())
assert partition['case_counts'] == {'PASS': 2880, 'FAIL': 140, 'ERROR': 600, 'XFAIL': 0, 'SKIP': 2, 'UNRUN': 0}
assert len(partition['cases']) == 3622
assert partition['denominators']['distinct_collection_files'] == 236

copies = {
    'native.stdout.txt': RAW / 'review/successor-final-B-cohort.txt',
    'native.junit.xml': TMP / 'successor-final-B-cohort.xml',
    'native.execution.json': RAW / 'review/successor-final-B-cohort.json',
    'native.profile.json': TMP / 'successor-final-B-cohort-profile.json',
    'case-partition.execution.json': RAW / 'review/successor-final-case-partition.json',
    'case-partition.stdout.txt': RAW / 'review/successor-final-case-partition.txt',
    'ruff.stdout.txt': RAW / 'review/successor-final-all-changed-ruff.txt',
}
secret_patterns = {
    'credential_uri': rb'(?:postgres(?:ql)?|https?)://[^\s/@:]+:[^\s/@]+@',
    'private_key': rb'-----BEGIN (?:RSA |EC |OPENSSH )?PRIVATE KEY-----',
    'github_token': rb'\b(?:ghp_|github_pat_)[A-Za-z0-9_]{20,}',
    'bearer_value': rb'(?i)bearer\s+[A-Za-z0-9_.-]{30,}',
}
OUT.mkdir(parents=True, exist_ok=False)
copied = []
for name, path in copies.items():
    before = identity(path)
    payload = path.read_bytes()
    scan = {key: len(re.findall(pattern, payload)) for key, pattern in secret_patterns.items()}
    assert not any(scan.values()), (name, scan)
    destination = OUT / name
    destination.write_bytes(payload)
    after = identity(path)
    target = identity(destination)
    assert before == after
    assert target['bytes'] == before['bytes'] and target['sha256'] == before['sha256']
    copied.append({'source': before, 'tracked_destination': str(destination.relative_to(ROOT)), 'secret_pattern_counts': scan})

xml_cases = list(ET.parse(copies['native.junit.xml']).getroot().iter('testcase'))
assert len(xml_cases) == len(partition['cases'])
rows = []
counts = collections.Counter()
for case in partition['cases']:
    if case['state'] not in {'FAIL', 'ERROR', 'SKIP'}:
        continue
    indices = case['junit_representation_indexes']
    assert len(indices) == 1
    element = xml_cases[indices[0]]
    outcomes = [child for child in element if child.tag in {'failure', 'error', 'skipped'}]
    assert len(outcomes) == 1
    outcome = outcomes[0]
    message = outcome.get('message', '')
    trace = outcome.text or ''
    basis_text = message + '\n' + trace
    if case['state'] == 'SKIP':
        category = 'named_catalog_service_or_recompute_input_absent'
    elif 'catalog_fetch_source_unreadable' in basis_text:
        category = 'catalog_fetch_source_unreadable'
    elif 'l6_control_artifact_missing' in basis_text:
        category = 'l6_control_artifact_missing'
    elif 'test_runtime_rego_authorization_parity.py' in case['nodeid'] and 'OPA' in basis_text:
        category = 'required_opa_backend_unavailable'
    elif 'test_restored_frontier_above_new_cap_refuses_before_consumer_effects' in case['nodeid']:
        category = 'C_restored_cap_consumer_effects_before_refusal'
    elif 'test_process_file_stream_cleanup_ownership[process-disconnect-failure]' in case['nodeid']:
        category = 'C_old_handle_cleanup_obligation_not_reconciled'
    else:
        category = 'other_call_failure_requires_exact_consumer_adjudication'
    counts[(case['state'], category)] += 1
    rows.append({
        'nodeid': case['nodeid'], 'state': case['state'],
        'phase_states': case['phase_states'], 'junit_case_index': indices[0],
        'literal_xml_outcome': outcome.tag, 'exception_type': outcome.get('type'),
        'message': message, 'full_trace_sha256': hashlib.sha256(trace.encode()).hexdigest(),
        'full_trace': 'native.junit.xml testcase index above; complete bytes also in native.stdout.txt',
        'reason_category': category,
        'category_grade': 'observed reason partition; no causal introducing-commit or finding closure decision',
        'P41_inherited_red_basis': 'not_established',
    })
assert len(rows) == 742
assert counts[('ERROR', 'catalog_fetch_source_unreadable')] == 559
assert counts[('ERROR', 'l6_control_artifact_missing')] == 41
assert counts[('FAIL', 'catalog_fetch_source_unreadable')] == 110
assert counts[('FAIL', 'l6_control_artifact_missing')] == 2
assert counts[('FAIL', 'required_opa_backend_unavailable')] == 7
assert counts[('FAIL', 'C_restored_cap_consumer_effects_before_refusal')] == 1
assert counts[('FAIL', 'C_old_handle_cleanup_obligation_not_reconciled')] == 1
assert counts[('FAIL', 'other_call_failure_requires_exact_consumer_adjudication')] == 19
assert counts[('SKIP', 'named_catalog_service_or_recompute_input_absent')] == 2
