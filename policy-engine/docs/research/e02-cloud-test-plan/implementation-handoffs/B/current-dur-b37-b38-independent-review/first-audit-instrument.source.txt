cache = {}

def git(*argv):
    return subprocess.check_output(['/usr/bin/git', '-C', REPO, *argv])

def read(ref, path):
    key = (ref, path)
    if key not in cache:
        cache[key] = git('show', ref + ':' + path)
    return cache[key]

def load(ref, path):
    return json.loads(read(ref, path))

def sha(data):
    return hashlib.sha256(data).hexdigest()

def pinned(ref, path):
    data = read(ref, path)
    return {'commit': ref, 'path': path, 'bytes': len(data), 'sha256': sha(data),
            'git_blob': git('rev-parse', ref + ':' + path).decode().strip()}

def check_artifact(ref, item):
    record = pinned(item.get('commit', ref), item['path'])
    assert record['sha256'] == item['sha256'], (item['path'], 'hash')
    if 'bytes' in item:
        assert record['bytes'] == item['bytes'], (item['path'], 'bytes')
    return record

def output_items(check):
    result = []
    for name in ('output', 'outputs', 'wrapper', 'junit'):
        value = check.get(name, [])
        if isinstance(value, dict):
            value = [value]
        for item in value:
            if isinstance(item, dict) and 'path' in item and 'sha256' in item:
                result.append(item)
    return result

def xml_cases(ref, path):
    root = ET.fromstring(read(ref, path))
    cases = list(root.iter('testcase'))
    ids = [(c.get('classname'), c.get('name')) for c in cases]
    assert len(ids) == len(set(ids)), path
    counts = collections.Counter()
    groups = collections.defaultdict(collections.Counter)
    classified = []
    for case in cases:
        tags = [tag for tag in ('error', 'failure', 'skipped') if case.find(tag) is not None]
        assert len(tags) <= 1, (path, case.attrib)
        outcome = {'error': 'ERROR', 'failure': 'FAIL', 'skipped': 'SKIP'}.get(tags[0] if tags else '', 'PASS')
        counts[outcome] += 1
        groups[case.get('classname')][outcome] += 1
        classified.append({'classname': case.get('classname'), 'name': case.get('name'), 'outcome': outcome})
    return {'unique_cases': len(cases), 'counts': dict(counts),
            'class_counts': {name: dict(value) for name, value in groups.items()}}, cases, classified
