def git(*args):
    return subprocess.check_output(['git', '-C', str(ROOT), *args])


def file_record(relative):
    data = (ROOT / relative).read_bytes()
    canonical = git('show', EXPECTED + ':' + relative)
    assert data == canonical, relative
    return {'path': relative, 'bytes': len(data), 'sha256': hashlib.sha256(data).hexdigest(),
            'git_blob': git('rev-parse', EXPECTED + ':' + relative).decode().strip(),
            'actual_disk_git_bytes_match': True}


def snapshot():
    head = git('rev-parse', 'HEAD').decode().strip()
    tree = git('rev-parse', 'HEAD^{tree}').decode().strip()
    status = git('status', '--porcelain=v1').decode()
    branch = git('symbolic-ref', '-q', 'HEAD').decode().strip()
    result = {'head': head, 'tree': tree, 'branch': branch, 'tracked_status': status,
              'expected_clean_attached': head == EXPECTED and tree == EXPECTED_TREE and not status}
    assert result['expected_clean_attached'], 'source checkpoint changed; no source restamp allowed'
    return result


def write(name, value):
    with (OUT / name).open('x') as stream:
        json.dump(value, stream, indent=2)
        stream.write('\n')
