def ident(path):
    data = path.read_bytes()
    return {'path':str(path), 'bytes':len(data), 'sha256':hashlib.sha256(data).hexdigest()}

def lossless_table(rows):
    names = sorted(set().union(*(r.keys() for r in rows)))
    columns = []
    encoded = [[None]*len(names) for _ in rows]
    for column_index,name in enumerate(names):
        values = []
        ids = {}
        for row_index,row in enumerate(rows):
            present = name in row
            value = row.get(name)
            token = json.dumps([present,value],ensure_ascii=False,sort_keys=True,separators=(',',':'))
            if token not in ids:
                ids[token] = len(values);values.append({'present':present,'value':value})
            encoded[row_index][column_index] = ids[token]
        columns.append({'name':name,'values':values})
    result = {'columns':columns,'rows':encoded}
    decoded = [{c['name']:c['values'][index]['value'] for c,index in zip(columns,row,strict=True) if c['values'][index]['present']} for row in encoded]
    assert decoded == rows
    return {'schema':'policyos.e02.lossless_json_row_table.v1','row_count':len(rows),'decoding':'For each row and each column, index column.values; include name:value only if present=true. Exact JSON types and absent keys are preserved.','table':result,'canonical_decoded_rows_sha256':hashlib.sha256(json.dumps(rows,sort_keys=True,separators=(',',':'),ensure_ascii=False).encode()).hexdigest(),'lossless_decode_equal':True}
