"""Adapt literal completed capture fields for the unchanged 3cd data-only parser."""
import hashlib
import json
from pathlib import Path

OUT = Path('/workspace/e02-B-current-coordination/.polisyos/e02-B-current/raw/affected-10e-20261006')

def read(name):
    return json.loads((OUT / name).read_bytes())

def ident(name):
    p = OUT / name
    b = p.read_bytes()
    return {'path': str(p), 'bytes': len(b), 'sha256': hashlib.sha256(b).hexdigest()}

def write(name, value):
    with (OUT / name).open('x') as f:
        json.dump(value, f, indent=2, ensure_ascii=False)
        f.write('\n')

w = read('wrapper.json')
p = read('execution-profile.json')
i = read('inventory.json')
assert w['harness_state'] == 'EXECUTED' and w['setup_failure'] is None
assert i['state'] == 'session_finished'
assert w['source_before'] == w['source_after']
assert w['source_before']['head'] == p['actual_source_head'] == '10e812a98d5f7f3997e7d0aa44de43dc90eb022c'
assert w['argv'] == p['argv']
assert len(i['selectors']) == len(set(i['selectors'])) == w['whole_file_count'] == 35
sources = {name: ident(name) for name in ('wrapper.json', 'execution-profile.json', 'selector-proposal.json', 'inventory.json')}

    'schema': 'policyos.e02.parser_input_aliases.v1',

    'test_paths': i['selectors'],
