import hashlib
import json
import os
import platform
import resource
import subprocess
import sys
import time
from pathlib import Path
repo=Path('/workspace/e02-B-current-runtime')
root=Path('/workspace/e02-B-current-coordination')
def git(repo,*a):return subprocess.check_output(['git','-C',str(repo),*a]).decode().strip()
command=[sys.executable,'/tmp/e02-run-final-footprint-driver.py','--repo',str(repo),'--base','198076863e143dea9f89f02734b13d50dae3eed5','--target','dd9584728bd96f204983615216efdbc58117e713','--heads-json','/tmp/e02-run-final-footprint-heads.json','--output','/tmp/e02-run-final-footprint-dd.json']
before={'head':git(root,'rev-parse','HEAD'),'tree':git(root,'rev-parse','HEAD^{tree}'),'status':git(root,'status','--porcelain')}
start=time.monotonic();p=subprocess.run(command,cwd=repo,stdout=subprocess.PIPE,stderr=subprocess.STDOUT)
wall=time.monotonic()-start

Path('/tmp/e02-run-final-footprint-execution.json').write_text(json.dumps(record,indent=2)+'\n')
print(json.dumps({'exit_code':p.returncode,'wall_seconds':wall,'root_stable':before==after,'outputs':record['outputs']}))
if p.returncode:raise SystemExit(p.returncode)
