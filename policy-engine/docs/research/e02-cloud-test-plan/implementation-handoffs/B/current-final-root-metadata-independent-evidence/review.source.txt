def git(*args):
    return subprocess.check_output(["git", *args])


def raw(sha, path):
    key = sha + ":" + path
    body = git("show", key)
    READS[key] = {"source_ref": key, "bytes": len(body), "sha256": hashlib.sha256(body).hexdigest()}
    return body


def obj(sha, path):
    return json.loads(raw(sha, path))


def paths(a, b):
    return git("diff", "--name-only", a, b).decode().splitlines()


def check(name, condition, details):
    CHECKS.append({"id": name, "outcome": "PASS" if condition else "FAIL", "details": details})


def verify(sha, row, pathkey="path"):
    body = raw(sha, row[pathkey])
    assert len(body) == row["bytes"] and hashlib.sha256(body).hexdigest() == row["sha256"], row[pathkey]
    return body


def handoff(value):
    assert REQUIRED <= set(value)
    assert value["schema"] == "policyos.e02.implementation_handoff.v1"
    assert value["predicate_basis"] in PREDICATES
    assert set("statement runtime_path proxy_divergence negative_controls".split()) <= set(value["property"])
    for item in value["checks"]:
        assert set("command target_sha environment input_closure outcome output".split()) <= set(item)
        assert item["outcome"] in {"PASS", "FAIL", "ERROR", "SKIP", "UNRUN"}


def indexed_rows(value):
    if isinstance(value, dict):
        if set("path bytes sha256".split()) <= set(value):
            yield value
        else:
            for child in value.values():
                yield from indexed_rows(child)
    elif isinstance(value, list):
        for child in value:
            yield from indexed_rows(child)


def verify_bound_refs(value):
    if isinstance(value, dict):
        source = value.get("source_ref")
        if set("bytes sha256".split()) <= set(value):
            if "commit" in value and "path" in value:
                verify(value["commit"], value)
            elif isinstance(source, str) and re.match(r"^[0-9a-f]{40}:", source):
                sha, path = source.split(":", 1)
                verify(sha, {**value, "path": path.split("#", 1)[0]})
        for child in value.values():
            verify_bound_refs(child)
    elif isinstance(value, list):
        for child in value:
            verify_bound_refs(child)


def junit(sha, path):
    root = ET.fromstring(raw(sha, path))
    outcomes = collections.Counter()
    failures = []
    for case in root.iter("testcase"):
        state = "FAIL" if case.find("failure") is not None else "ERROR" if case.find("error") is not None else "SKIP" if case.find("skipped") is not None else "PASS"
        outcomes[state] += 1
        if state != "PASS":
            failures.append({"classname": case.get("classname"), "name": case.get("name"), "outcome": state})
    return {"case_count": sum(outcomes.values()), "outcomes": dict(outcomes), "nonpass": failures}
