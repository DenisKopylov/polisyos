def git(*argv):
    return subprocess.check_output(['git', *argv], cwd=ROOT)

def blob(sha, path):
    return git('show', sha + ':' + path)

def digest(data):
    return hashlib.sha256(data).hexdigest()

def ref(sha, path):
    data = blob(sha, path)
    return {'git_sha': sha, 'path': path, 'bytes': len(data), 'sha256': digest(data)}

def read_ref(item):
    sha = item.get('git_sha', item.get('commit', item.get('source_sha')))
    data = blob(sha, item['path'])
    assert len(data) == item['bytes'] and digest(data) == item['sha256'], item
    return data

def state(case):
    if case.find('failure') is not None:
        return 'FAIL'
    if case.find('error') is not None:
        return 'ERROR'
    if case.find('skipped') is not None:
        return 'SKIP'
    return 'PASS'
