def git(*args):
    return subprocess.check_output(['/usr/bin/git', '--no-optional-locks', '-C', str(ROOT), *args])

def ref(path):
    raw = path.read_bytes()
    return {'path': str(path), 'bytes': len(raw), 'sha256': hashlib.sha256(raw).hexdigest()}

def source(path):
    expected = git('show', SHA + ':' + path)
    actual = (ROOT / path).read_bytes()
    assert actual == expected, path
    return {**ref(ROOT / path), 'repository_path': path, 'source_sha': SHA,
            'git_blob': git('rev-parse', SHA + ':' + path).decode().strip()}

def state():
    return {'head': git('rev-parse', 'HEAD').decode().strip(),
            'tree': git('rev-parse', 'HEAD^{tree}').decode().strip(),
            'status': git('status', '--porcelain').decode()}

def counts(xml):
    cases = ET.parse(xml).getroot().findall('.//testcase')
    rows = []
    for case in cases:
        outcome = 'ERROR' if case.find('error') is not None else 'FAIL' if case.find('failure') is not None else 'SKIP' if case.find('skipped') is not None else 'PASS'
        rows.append({'name': case.attrib['name'], 'outcome': outcome})
    return {'observed_cases': len(rows), 'outcomes': dict(collections.Counter(r['outcome'] for r in rows)), 'cases': rows}

def observations(path, prefix):
    return [json.loads(line[len(prefix):]) for line in path.read_text().splitlines() if line.startswith(prefix)]
