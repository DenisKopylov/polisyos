
from pathlib import Path
import hashlib, importlib, importlib.metadata, json, subprocess, sys
ROOT=Path(sys.argv[1]); OUT=Path(sys.argv[2]); TEST=Path(sys.argv[3]); SOURCE=sys.argv[4]
modules=['polisyos.core.llm.response','polisyos.core.llm.traced_client','polisyos.core.llm.settlement','polisyos.core.security.audit_log_adapter','polisyos.core.security.audit_protocol','polisyos.core.security.audit_sink','polisyos.core.security.audit_models','polisyos.core.security.audit_verifier','polisyos.scientist.orchestration.engine.budget','polisyos.scientist.orchestration.engine.budget_ledger','polisyos.scientist.orchestration.engine.budget_middleware','polisyos.scientist.orchestration.llm.budget_enforcer','polisyos.scientist.orchestration.llm.gateway_client']
def ref(path):
 raw=path.read_bytes(); rel=path.relative_to(ROOT).as_posix(); expected=subprocess.check_output(['git','show',f'{SOURCE}:{rel}'],cwd=ROOT)
 assert raw==expected, rel
 return {'path':rel,'actual_origin':str(path),'bytes':len(raw),'sha256':hashlib.sha256(raw).hexdigest(),'git_blob':subprocess.check_output(['git','rev-parse',f'{SOURCE}:{rel}'],cwd=ROOT,text=True).strip(),'source_sha':SOURCE}
origins=[]
for name in modules:
 m=importlib.import_module(name); row=ref(Path(m.__file__).resolve());row['module']=name;origins.append(row)
print('B65_EXACT_SOURCE_ORIGINS '+json.dumps(origins), flush=True)
import pytest
args=[str(TEST),'-c',str(ROOT/'policy-engine/pyproject.toml'),'--noconftest','-p','no:cacheprovider','-s','-ra',f'--basetemp={OUT}/pytest-tmp',f'--junitxml={OUT}/native.xml']
print('B65_NATIVE_ARGUMENTS '+json.dumps(args),flush=True)
result=pytest.main(args)
all_origins=[]
for name,m in sorted(sys.modules.items()):
 if name=='polisyos' or name.startswith('polisyos.'):
  f=getattr(m,'__file__',None)
  if f:
   path=Path(f).resolve()
   if path.is_relative_to(ROOT/'policy-engine/src'):
    row=ref(path);row['module']=name;all_origins.append(row)
   else:raise AssertionError(f'foreign source module {name}: {f}')
versions={n:importlib.metadata.version(n) for n in ['pytest','pydantic','aiohttp','ruff','numpy']}
(OUT/'source-context.json').write_text(json.dumps({'source_sha':SOURCE,'explicit_source_origins':origins,'complete_loaded_polisyos_module_denominator':len(all_origins),'all_loaded_polisyos_origins':all_origins,'python_executable':sys.executable,'python_version':sys.version,'observed_versions':versions,'pytest_args':args,'pytest_result':int(result)},indent=2)+'\n')
raise SystemExit(int(result))
