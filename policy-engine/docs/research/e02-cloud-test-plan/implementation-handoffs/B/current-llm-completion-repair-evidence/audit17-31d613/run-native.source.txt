    return subprocess.check_output(["git", "-C", str(root), "rev-parse", "HEAD"], text=True).strip()

    ["git", "-C", str(TEST_ROOT), "show", f"{TEST_SHA}:{TEST.relative_to(TEST_ROOT)}"]
)
env = dict(os.environ)
for key in ("PYTEST_PLUGINS", "E02_B_COHORT_INVENTORY_PATH"):
    env.pop(key, None)
