@dataclass(frozen=True, slots=True)
class Unit:
    dimension: str
    symbol: str
    scale: float = ...
    def __post_init__(self) -> None: ...
    def is_compatible(self, other: Unit) -> bool: ...
    def stable_digest(self) -> str: ...
    def _stable_dict(self) -> dict[str, Any]: ...

@dataclass(frozen=True)
class DimVar:
    name: str
    def __post_init__(self) -> None: ...
    def __repr__(self) -> str: ...
    def __str__(self) -> str: ...

DimExpr = int | str | DimVar | None
Shape = tuple[DimExpr, ...]

@dataclass(frozen=True, slots=True)
class SlotSpec:
    name: str
    slot_type: SlotType
    unit: Unit
    contract_id: str | None = ...
    shape: Shape = ...
    description: str = ...
    bounds: tuple[float | int | None, float | int | None] = ...
    contract_capabilities: frozenset[OutputContractCapability] = ...
    contract_owner: str | None = ...
    def __post_init__(self) -> None: ...
    @classmethod
    def for_output_contract(
        cls,
        name: str,
        slot_type: SlotType,
        unit: Unit,
        *,
        output_contract: type[object],
        shape: Shape = (),
        description: str = "",
        bounds: tuple[float | int | None, float | int | None] = (None, None),
    ) -> SlotSpec: ...
    def stable_digest(self) -> str: ...
    def _stable_dict(self) -> dict[str, Any]: ...
    def __hash__(self) -> int: ...
    def __eq__(self, other: object) -> bool: ...
