def digest(raw):
    return hashlib.sha256(raw).hexdigest()


def git(*words):
    return subprocess.check_output(['/usr/bin/git', '--no-optional-locks', '-C', str(ROOT), *words])


def snapshot():
    return {'head':git('rev-parse','HEAD').decode().strip(),
            'tree':git('rev-parse','HEAD^{tree}').decode().strip(),
            'status':git('status','--porcelain=v1','--untracked-files=all').decode()}


def source(path):
    raw = git('show', SHA + ':' + path)
    if (ROOT / path).read_bytes() != raw:
        raise ValueError('Live source differs from supplied Git object: ' + path)
    return raw

def local(path, *, preserve=False):
    raw = path.read_bytes()
    row = {'path':str(path),'bytes':len(raw),'sha256':digest(raw),'complete_bytes_read':True}
    if preserve:
        output = OUT / path.name
        with output.open('xb') as stream:
            stream.write(raw)
        assert output.read_bytes() == raw
        row['preserved_complete_bytes'] = str(output)
    refs.append(row)
    return raw
