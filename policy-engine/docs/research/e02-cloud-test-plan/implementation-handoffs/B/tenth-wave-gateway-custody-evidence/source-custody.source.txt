def git_bytes(sha: str, path: str) -> bytes:
    return subprocess.check_output(['git', '-C', str(ROOT), 'show', f'{sha}:{path}'])


def ref(sha: str, path: str) -> dict:
    data = git_bytes(sha, path)
    return {'source_sha': sha, 'path': path, 'bytes': len(data),
            'sha256': hashlib.sha256(data).hexdigest()}


def observations(data: bytes) -> list:
    text = data.decode('utf-8')
    marker = 'EXACT_MONEY_ACTUAL '
    result = []
    cursor = 0
    while (start := text.find(marker, cursor)) != -1:
        record, consumed = json.JSONDecoder().raw_decode(text[start + len(marker):])
        result.append(record)
        cursor = start + len(marker) + consumed
    return result


def own_path(origin: dict) -> str:
    return origin.get('relative_path') or 'policy-engine/' + origin['path'].split('/policy-engine/')[1]


def xml_counts(data: bytes) -> dict:
    try:
        envelope = json.loads(data)
    except json.JSONDecodeError:
        pass
    else:
        data = envelope['text'].encode('utf-8')
    root = ET.fromstring(data)
    tests = list(root.iter('testcase'))
    return {'cases': len(tests), 'PASS': sum(not len(t) for t in tests),
            'FAIL': sum(t.find('failure') is not None for t in tests),
            'ERROR': sum(t.find('error') is not None for t in tests),
            'SKIP': sum(t.find('skipped') is not None for t in tests)}
