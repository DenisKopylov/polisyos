def convert(spec):
    argv=list(spec.argv)
    assert argv[:2]==[uv,'run'], argv
    payload=argv[2:]
    extras=[]
    while payload and payload[0]=='--extra':
        extras.append(payload[1]);payload=payload[2:]
    executable,*args=payload
    if executable=='python':direct=[python,*args]
    elif executable=='pytest':direct=[python,'-m','pytest',*args]
    elif executable=='polisyos-tools':direct=[python,'-m','tools.cli',*args]
    else:raise AssertionError(payload)
    caps=dict(spec.env or {})
    assert not caps or caps==verify.PYTEST_NUMERICAL_ENV
    return {'label':spec.label,'canonical_argv':argv,'direct_argv':direct,'cwd':str(root),'canonical_env':caps,'removed_only_wrapper_numeric_caps':caps,'required_extra_profiles':extras,'state':'UNRUN pending parent freeze; no executable verdict'}
rows=[{'group':'verify_backend',**convert(s)} for s in verify_specs]+[{'group':'ci_addition',**convert(s)} for s in ci_specs]
assert len(rows)==25
