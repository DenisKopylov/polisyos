def git(*args):
    return subprocess.check_output(['git', *args], cwd=REPO)


def materialize(sha):
    changed = git('diff', '--name-only', BASE, FINAL, '--', 'policy-engine/src').decode().splitlines()
    for rel in changed:
        path = EXPORT / rel
        path.parent.mkdir(parents=True, exist_ok=True)
        path.write_bytes(git('show', f'{sha}:{rel}'))


def run(label, sha, test, *, mode, mutation=None):
    env = dict(os.environ, PYTHONPATH=str(EXPORT / 'policy-engine/src') + ':' + str(PRODUCT), PYTHONDONTWRITEBYTECODE='1')
    argv = [PYTHON, '-m', 'pytest', '-c', str(PRODUCT / 'pytest.ini'), '-o', 'addopts=', '--import-mode=importlib', '--confcutdir=' + str(Path(test.split('::')[0]).parent), '-q', '-rA', test]
    files = ['models','registry','runtime','calibration']
    metadata = {'source_sha':sha, 'source_tree':git('rev-parse', f'{sha}^{{tree}}').decode().strip(), 'command':argv, 'cwd':str(PRODUCT), 'export':str(EXPORT), 'mode':mode, 'mutation':mutation, 'source_file_sha256':{name:hashlib.sha256((EXPORT / f'policy-engine/src/polisyos/scientist/methods/autotune/{name}.py').read_bytes()).hexdigest() for name in files}, 'test_path':test.split('::')[0], 'test_sha256':hashlib.sha256(Path(test.split('::')[0]).read_bytes()).hexdigest()}
    start=time.monotonic()
    path=OUT / (label+'.txt')
    with path.open('w') as output:
        output.write(json.dumps(metadata,sort_keys=True)+'\n');output.flush()
        result=subprocess.run(argv,cwd=PRODUCT,env=env,stdout=output,stderr=subprocess.STDOUT)
        metadata.update(exit_code=result.returncode,wall_seconds=time.monotonic()-start)
        output.write('\nMEASUREMENT='+json.dumps({'exit_code':result.returncode,'wall_seconds':metadata['wall_seconds']})+'\n')
    metadata.update(log=str(path),log_sha256=hashlib.sha256(path.read_bytes()).hexdigest())
    (OUT/(label+'.json')).write_text(json.dumps(metadata,indent=2)+'\n')
    print(json.dumps({'probe':label,'exit_code':result.returncode,'log':str(path)}),flush=True)
    return result.returncode
