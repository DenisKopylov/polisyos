def git(*args):
    return subprocess.check_output(['git',*args],cwd=ROOT,text=True).strip()


def fingerprint(path):
    data=path.read_bytes()
    return {'bytes':len(data),'sha256':hashlib.sha256(data).hexdigest()}


def snapshot():
    result={'head':git('rev-parse','HEAD'),'tree':git('rev-parse','HEAD^{tree}'),
            'branch':git('symbolic-ref','--short','HEAD'),
            'tracked_status':git('status','--porcelain','--untracked-files=no'),
            'product_delta_from_base':git('diff',BASE,'--name-only','--','policy-engine/src'),
            'utc':datetime.datetime.now(datetime.UTC).isoformat()}
    assert result['head']==SOURCE and not result['tracked_status'] and not result['product_delta_from_base'],result
    return result


def run(mode):
    before=snapshot()
    lines=PLAN.read_text().splitlines()
    index=next(i for i,line in enumerate(lines) if line.startswith('### OPT-01'))
    line=next(line for line in lines[index:] if line.startswith('cd policy-engine && python -m pytest '))
    literal=shlex.split(line.split(' && ',1)[1])[3:]
    args=literal if mode=='canonical' else ['-q',*[f'tests/unit/scientist/methods/autotune/test_pareto.py::TestParetoFront::{name}' for name in NAMES],'-ra']
    directory=HERE/mode
    directory.mkdir(exist_ok=False)
    basetemp=Path('/dev/shm')/f'search-opt01-v2-{mode}-{uuid.uuid4().hex}'
    assert not basetemp.exists()
    args=[*args,'-o','addopts=','--import-mode=importlib','--strict-markers','--basetemp',str(basetemp),'--junitxml',str(directory/'junit.xml'),'-rA']
    argv=[PYTHON,'-B','-c',BOOTSTRAP,str(ROOT),SOURCE,str(OBSERVER),str(directory/'events.jsonl.txt'),mode,*args]
    start=time.monotonic()
    with (directory/'complete.txt').open('xb') as stream:
        child=subprocess.run(argv,cwd=PRODUCT,env=ENV,stdout=stream,stderr=subprocess.STDOUT)
    after=snapshot()
    cases=list(ET.parse(directory/'junit.xml').iter('testcase'))
    counts={'tests':len(cases),'failed':sum(c.find('failure') is not None for c in cases),'errors':sum(c.find('error') is not None for c in cases),'skipped':sum(c.find('skipped') is not None for c in cases)}
    counts['passed']=counts['tests']-counts['failed']-counts['errors']-counts['skipped']
    result={'mode':mode,'canonical_command':line,'argv':argv,'equivalent_pytest_argv':[PYTHON,'-B','-m','pytest',*args],
            'basetemp':str(basetemp),'basetemp_existed_before':False,'exit_code':child.returncode,'wall_seconds':time.monotonic()-start,
            'source_before':before,'source_after':after,'junit_counts':counts,
            'outputs':{name:fingerprint(directory/name) for name in ['complete.txt','events.jsonl.txt','junit.xml']}}
    (directory/'result.json').write_text(json.dumps(result,indent=2)+'\n')
    print(json.dumps({'mode':mode,'exit_code':child.returncode,'counts':counts,'wall_seconds':result['wall_seconds']}),flush=True)


if __name__=='__main__':
    mode=sys.argv[1]
    if mode=='metadata':
        result={'python':sys.version,'executable':sys.executable,'source':snapshot(),
                'packages':{name:importlib.metadata.version(name) for name in ['pytest','ruff','torch','botorch','gpytorch','numpy','scipy','pydantic']},
                'environment':{name:ENV.get(name) for name in ['PYTHONPATH','PYTHONDONTWRITEBYTECODE','TIKTOKEN_CACHE_DIR','OMP_NUM_THREADS','MKL_NUM_THREADS','OPENBLAS_NUM_THREADS','NUMEXPR_NUM_THREADS','POLISYOS_PYTEST_WORKERS','PYTEST_ADDOPTS']},
                'observer':fingerprint(OBSERVER),'capture':fingerprint(Path(__file__)),'limits_added':False}
        old=ast.parse(subprocess.check_output(['git','show',BASE+':'+TEST],cwd=ROOT,text=True))
        new=ast.parse((ROOT/TEST).read_text())
        functions=lambda tree:{node.name:node for node in ast.walk(tree) if isinstance(node,ast.FunctionDef)}
        a,b=functions(old),functions(new)
        changes=[]
        for name in a:
            if ast.dump(a[name])!=ast.dump(b[name]): changes.append(name)
            original=[ast.dump(node) for node in ast.walk(a[name]) if isinstance(node,ast.Assert)]
            current=[ast.dump(node) for node in ast.walk(b[name]) if isinstance(node,ast.Assert)]
            assert original==current,name
        assert set(changes)==set(NAMES),changes
        result.update(changed_existing_functions=changes,original_assertion_ast='identical',added_functions=sorted(b.keys()-a.keys()))
        (HERE/'environment.json').write_text(json.dumps(result,indent=2)+'\n')
        print(json.dumps({'source':result['source']['head'],'assertions':'identical','packages':result['packages']}))
    else:
        run(mode)
