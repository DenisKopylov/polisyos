import hashlib,json,os,resource,subprocess,sys,time
from pathlib import Path
OUT=Path(__file__).parent
REPO=Path('/workspace/e02-D-published-funnel')
PROJECT=REPO/'policy-engine'
def git(*args):return subprocess.check_output(['git',*args],cwd=REPO,text=True).strip()
def inputs():
    return {'sha':git('rev-parse','HEAD'),'tree':git('rev-parse','HEAD^{tree}'),'status':git('status','--porcelain'),'paths':{p:{'blob':git('rev-parse','HEAD:'+p),'sha256':hashlib.sha256((REPO/p).read_bytes()).hexdigest()} for p in ['policy-engine/tests/unit/scientist/orchestration/engine/test_budget_ledger.py','policy-engine/src/polisyos/scientist/orchestration/engine/budget_ledger.py','policy-engine/src/polisyos/scientist/orchestration/engine/budget_middleware.py','policy-engine/src/polisyos/scientist/methods/search/funnel/orchestrator.py','policy-engine/tests/integration/scientist/methods/search/funnel/test_orchestrator.py','policy-engine/pytest.ini']}}
def census():return {n:Path('/sys/fs/cgroup/'+n).read_text() for n in ('memory.current','memory.max','memory.events','memory.stat')}
before=inputs();(OUT/'before.json').write_text(json.dumps(before,indent=2)+'\n')
(OUT/'resources-before.json').write_text(json.dumps(census(),indent=2)+'\n')
selectors=['tests/unit/scientist/orchestration/engine/test_budget_ledger.py','tests/integration/scientist/methods/search/funnel/test_orchestrator.py::test_native_policy_callers_full_split_actual_events_and_fresh_reopen','tests/integration/scientist/methods/search/funnel/test_orchestrator.py::test_actual_spend_does_not_resume_but_released_owned_capacity_rechecks_remaining_work']
base=OUT/'fresh-pytest-fixtures'
assert not base.exists()
argv=['/tmp/e02-D-runtime-minimal-20261006/bin/python','-B','-m','pytest',*selectors,'--run-quarantine','--basetemp='+str(base),'--junitxml='+str(OUT/'junit.xml'),'-o','cache_dir='+str(OUT/'pytest-cache'),'-p','capture','-rA']
env={**os.environ,'PYTHONDONTWRITEBYTECODE':'1','PYTHONPATH':str(PROJECT/'src')+':'+str(PROJECT)+':'+str(OUT),'E02_PROJECT':str(PROJECT),'E02_OUTPUT':str(OUT),'TIKTOKEN_CACHE_DIR':'/workspace/e02-D2-receipts/tokenizer'}
manifest={'source':before,'cwd':str(PROJECT),'argv':argv,'env_overrides':{k:env[k] for k in ('PYTHONDONTWRITEBYTECODE','PYTHONPATH','E02_PROJECT','E02_OUTPUT','TIKTOKEN_CACHE_DIR')},'start_time_ns':time.time_ns(),'basetemp_before':'nonexisting','quotas_or_timeout_added':False,'profile_receipt':'/tmp/e02-D-runtime-minimal-profile-receipt.json','profile_receipt_sha256':hashlib.sha256(Path('/tmp/e02-D-runtime-minimal-profile-receipt.json').read_bytes()).hexdigest()}
(OUT/'execution-manifest.json').write_text(json.dumps(manifest,indent=2)+'\n')
started=time.monotonic()
with (OUT/'complete.stdout.txt').open('wb') as stdout,(OUT/'complete.stderr.txt').open('wb') as stderr:
    result=subprocess.run(argv,cwd=PROJECT,env=env,stdout=stdout,stderr=stderr)
after=inputs();(OUT/'after.json').write_text(json.dumps(after,indent=2)+'\n')
(OUT/'resources-after.json').write_text(json.dumps(census(),indent=2)+'\n')
origins=json.loads((OUT/'actual-origins.json').read_text()) if (OUT/'actual-origins.json').exists() else []
for row in origins:row['expected_git_blob']=git('rev-parse',before['sha']+':'+row['repo_path']);row['matches']=row['git_blob']==row['expected_git_blob']
(OUT/'verified-origins.json').write_text(json.dumps(origins,indent=2)+'\n')
summary={'source':before,'after':after,'source_unchanged':before==after,'exitcode':result.returncode,'wall_seconds':time.monotonic()-started,'maxrss_kib':resource.getrusage(resource.RUSAGE_CHILDREN).ru_maxrss,'origins':len(origins),'origin_errors':[o for o in origins if not o['matches']],'end_time_ns':time.time_ns(),'result_dir':str(OUT)}
(OUT/'result.json').write_text(json.dumps(summary,indent=2)+'\n');print(json.dumps(summary))
