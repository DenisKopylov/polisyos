  results=[]
  for name in names:
   module=sys.modules.get(name);p=pathlib.Path(module.__file__).resolve() if module else None
   local=p is not None and p.is_relative_to(lane/'src')
   actual=hashlib.sha256(p.read_bytes()).hexdigest() if p else None
   expected=hashlib.sha256(subprocess.check_output(['git','show',source_sha+':policy-engine/'+str(p.relative_to(lane))])).hexdigest() if local else None
   results.append({'module':name,'file':str(p) if p else None,'sha256':actual,'tracked_frozen_sha256':expected,'PASS':local and actual==expected})
  print('E02_RUNTIME_ORIGINS='+json.dumps({'source_sha':source_sha,'modules':results},sort_keys=True))
  if not all(r['PASS'] for r in results):session.exitstatus=1
raise SystemExit(pytest.main(sys.argv[1:],plugins=[Origins()]))
