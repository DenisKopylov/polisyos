def sha(data):
    return hashlib.sha256(data).hexdigest()


def git(*args):
    return subprocess.check_output(["git", "-C", str(ROOT), *args])


def blob(commit, path):
    return git("show", commit + ":" + path)


def save(name, value):
    (OUT / name).write_text(json.dumps(value, indent=2, ensure_ascii=False) + "\n")


def local_manifest(folder):
    return [
        {"path": str(p.relative_to(ROOT)), "bytes": len(p.read_bytes()), "sha256": sha(p.read_bytes())}
        for p in sorted(folder.iterdir()) if p.is_file()
    ]


def junit(path):
    root = ElementTree.parse(path).getroot()
    suites = [root] if root.tag == "testsuite" else list(root)
    return {k: sum(int(s.attrib.get(k, 0)) for s in suites) for k in ("tests", "failures", "errors", "skipped")}


def origins(path, default_source):
    report = json.loads(path.read_text())
    rows = report["loaded_origins"]
    failures = list(report.get("errors", []))
    for row in rows:
        p = row["path"]
        if not p.startswith("policy-engine/"):
            p = "policy-engine/" + p
        actual = sha(blob(row.get("source", default_source), p))
        if actual != row["sha256"]:
            failures.append({"path": p, "expected": actual, "recorded": row["sha256"]})
    return {"loaded_product_origins": len(rows), "independently_rechecked_git_bytes": len(rows), "errors": failures}
