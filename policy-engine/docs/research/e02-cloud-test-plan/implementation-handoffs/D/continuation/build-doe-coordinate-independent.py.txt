CACHE: dict[tuple[str, str], bytes] = {}


def git(*args: str) -> bytes:
    return subprocess.check_output(["git", *args], cwd=REPO)


def blob(commit: str, path: str) -> bytes:
    key = (commit, path)
    if key not in CACHE:
        CACHE[key] = git("show", f"{commit}:{path}")
    return CACHE[key]


def sha(raw: bytes) -> str:
    return hashlib.sha256(raw).hexdigest()


receipt_path = D + "search-doe-coordinate-order-continuation.json"
receipt_raw = blob(TRANSPORT, receipt_path)
receipt = json.loads(receipt_raw)
header_path = PACK + "deciding-outputs.header.json"
header_raw = blob(TRANSPORT, header_path)
header = json.loads(header_raw)
compressed = blob(TRANSPORT, PACK + "deciding-outputs.json.gz")
assert sha(compressed) == header["compressed_sha256"]
assert len(compressed) == header["compressed_bytes"]
raw_json = gzip.decompress(compressed)
assert sha(raw_json) == header["uncompressed_json_sha256"]
assert len(raw_json) == header["uncompressed_json_bytes"]
bundle = json.loads(raw_json)
assert bundle["candidate_sha"] == SOURCE == header["candidate_sha"]
members: dict[str, bytes] = {}
originals = []
for meta, item in zip(header["members"], bundle["members"], strict=True):
    assert {k: v for k, v in item.items() if k != "data_base64"} == meta
    raw = base64.b64decode(item["data_base64"], validate=True)
    assert len(raw) == meta["raw_bytes"] and sha(raw) == meta["raw_sha256"]
    assert item["name"] not in members
    members[item["name"]] = raw
    original = Path(item["original_path"])
    exists = original.is_file()
    if exists:
        assert original.read_bytes() == raw
    originals.append({"member": item["name"], "original_present_and_exact": exists})
assert len(members) == header["member_count"] == 81
assert sum(map(len, members.values())) == header["raw_member_byte_total"] == 392440


def locator(text: str) -> tuple[str, str]:
    left, right = text.split("!/", 1)
    assert left == PACK + "deciding-outputs.json.gz"
    member, sep, pointer = right.partition("#")
    assert member in members, member
    if sep:
        value = json.loads(members[member])
        for key in pointer.lstrip("/").split("/"):
            key = key.replace("~1", "/").replace("~0", "~")
            value = value[int(key)] if isinstance(value, list) else value[key]
    return member, pointer if sep else ""


locators = []


def walk(value: object) -> None:
    if isinstance(value, dict):
        for item in value.values():
            walk(item)
    elif isinstance(value, list):
        for item in value:
            walk(item)
    elif isinstance(value, str) and value.startswith(PACK) and "!/" in value:
        locators.append(locator(value))


walk(receipt)

walk(json.loads(origin_raw))

checks = []

runtime = []
for check in receipt["checks"]:
    metadata_member, _ = locator(check["metadata"])
    meta = json.loads(members[metadata_member])
    candidate = check["target_sha"]
    actual_tree = git("rev-parse", candidate + "^{tree}").decode().strip()
    assert check.get("target_tree", actual_tree) == actual_tree
    assert meta.get("source_sha", meta.get("target_sha")) == candidate
    assert meta.get("source_tree", meta.get("tree", actual_tree)) == actual_tree
    assert meta["source_after"] == candidate
    assert meta["status_after"] == ""
    assert check["returncode"] == meta["returncode"]
    output_member, _ = locator(check["output"])
    assert sha(members[output_member]) == meta["output_sha256"]
    if "command_ref" in check:
        command_member, command_pointer = locator(check["command_ref"])
        assert command_member == metadata_member and command_pointer == "/argv"
    for path, expected in meta.get("changed_file_sha256", {}).items():
        assert sha(blob(candidate, path)) == expected
        input_bindings.append({"check": check["id"], "path": path, "sha256": expected})
    for selector, expected in meta.get("selector_sha256", {}).items():
        path = "policy-engine/" + selector.split("::", 1)[0]
        assert sha(blob(candidate, path)) == expected
    parent = metadata_member.rsplit("/", 1)[0]
    junit_member = parent + "/junit.xml"
    counts = None
    if junit_member in members:
        testcases = list(ET.fromstring(members[junit_member]).iter("testcase"))
        counts = {"tests": len(testcases), "failure": sum(t.find("failure") is not None for t in testcases), "error": sum(t.find("error") is not None for t in testcases), "skip": sum(t.find("skipped") is not None for t in testcases)}
        counts["pass"] = counts["tests"] - counts["failure"] - counts["error"] - counts["skip"]
        assert counts == check["counts"], (check["id"], counts, check["counts"])
    elif "counts" in check:
        assert check["counts"] is None
    if metadata_member.endswith("/run.json"):
        if meta["returncode"] == -9:
            assert junit_member not in members
            assert parent + "/junit.native.json" not in members
        native_member = parent + "/junit.native.json"
        if native_member in members:
            native = json.loads(members[native_member])
            assert native["exitstatus"] == meta["returncode"]
            assert native["python"] == "3.14.7"
            for module, info in native["modules"].items():
                if module.startswith("polisyos."):
                    path = "policy-engine/" + info["path"].split("/policy-engine/", 1)[1]
                    assert sha(blob(candidate, path)) == info["sha256"]
                    origins.append({"check": check["id"], "module": module, "path": path, "sha256": info["sha256"]})
                else:
                    file = Path(info["path"])
                    assert file.is_file() and sha(file.read_bytes()) == info["sha256"]
                    backend_files.append({"check": check["id"], "module": module, **info})
            native_calls = native["native_calls"]
        else:
            native_calls = None
        for arg in meta["argv"]:
            if arg.startswith("--basetemp="):
                path = arg.split("=", 1)[1]
                temporary_dirs.append({"check": check["id"], "path": path, "currently_exists": Path(path).exists(), "absence_does_not_establish_deletion": True})
        runtime.append({"check": check["id"], "source": candidate, "tree": actual_tree, "returncode": meta["returncode"], "counts": counts, "outcome": check["outcome"], "wall_seconds": meta["wall_seconds"], "complete_output_bytes": len(members[output_member]), "complete_output_sha256": sha(members[output_member]), "argv": meta["argv"], "cwd": meta["cwd"], "environment_overrides": meta["environment_overrides"], "native_calls": native_calls})
    checks.append({"check": check["id"], "metadata_member": metadata_member, "source": candidate, "source_before_after_match": True, "after_status_clean_recorded": True, "returncode": meta["returncode"], "counts": counts, "outcome": check["outcome"]})
assert len(runtime) == 10
assert len(origins) == 91 and len(backend_files) == 21
