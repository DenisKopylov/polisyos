HANDOFF = ROOT + 'search-space-numeric-continuation.json'

def git(*args):
    return subprocess.check_output(['git', *args], cwd=REPO)

def digest(data):
    return hashlib.sha256(data).hexdigest()

def data(sha, path):
    return git('show', f'{sha}:{path}')

def bind(sha, path):
    raw = data(sha, path)
    return {'source_sha': sha, 'path': path, 'bytes': len(raw), 'sha256': digest(raw), 'git_blob': git('rev-parse', f'{sha}:{path}').decode().strip()}

handoff = json.loads(data(TRANSPORT, HANDOFF))
payload = []
for row in handoff['evidence_manifest']:
    actual = bind(TRANSPORT, row['path'])
    assert actual['sha256'] == row['sha256'] and actual['bytes'] == row['bytes']
    payload.append({**actual, 'status': 'MATCH'})
changed = []
for row in handoff['changed_paths']:
    actual = bind(SOURCE, row['path'])
    assert actual['sha256'] == row['sha256'] and actual['git_blob'] == row['git_blob']
    changed.append({**actual, 'status': 'MATCH'})

origins, inputs, runtime = [], [], []
for check in handoff['checks']:
    source = check['target_sha']
    run = json.loads(data(TRANSPORT, check['input_closure']))
    parent = str(Path(check['input_closure']).parent)
    native = json.loads(data(TRANSPORT, parent + '/junit.native.json'))
    xml = ET.fromstring(data(TRANSPORT, parent + '/junit.xml'))
    suites = [xml] if xml.tag == 'testsuite' else xml.findall('testsuite')
    counts = {key: sum(int(s.attrib.get(key, 0)) for s in suites) for key in ['tests', 'failures', 'errors', 'skipped']}
    assert counts == check['junit']
    assert run['source_sha'] == source == run['source_after']
    assert run['source_tree'] == git('rev-parse', source + '^{tree}').decode().strip()
    assert digest(data(TRANSPORT, check['complete_output'])) == run['output_sha256']
    for path, expected in run['selector_sha256'].items():
        target = 'policy-engine/' + path
        overlay = check['id'] == 'baseline' and path.endswith('/test_space_numeric_admission.py')
        actual = bind(SOURCE if overlay else source, target)
        assert actual['sha256'] == expected
        inputs.append({'cell': check['id'], **actual, 'status': 'EXACT_UNTRACKED_CHARACTERIZATION_OVERLAY' if overlay else 'MATCH'})
    for path, expected in run['changed_file_sha256'].items():
        actual = bind(source, path)
        assert actual['sha256'] == expected
        inputs.append({'cell': check['id'], **actual, 'status': 'MATCH_CHANGED_INPUT'})
    for module, row in native['modules'].items():
        if '/policy-engine/' in row['path']:
            path = 'policy-engine/' + row['path'].split('/policy-engine/', 1)[1]
            actual = bind(source, path)
            assert actual['sha256'] == row['sha256']
            origins.append({'cell': check['id'], 'module': module, 'loaded_path': row['path'], **actual, 'status': 'GIT_MATCH'})
        else:
            observed = Path(row['path']).read_bytes()
            assert digest(observed) == row['sha256']
            origins.append({'cell': check['id'], 'module': module, 'loaded_path': row['path'], 'sha256': row['sha256'], 'bytes': len(observed), 'status': 'LOCKED_FILE_BYTES_MATCH'})
    runtime.append({'id': check['id'], 'counts': counts, 'returncode': run['returncode'], 'wall_seconds': run['wall_seconds'], 'source_sha': source, 'source_tree': run['source_tree'], 'status_after': run['status_after'], 'argv': run['argv'], 'environment_overrides': run['environment_overrides'], 'native_calls': native['native_calls'], 'complete_output': check['complete_output'], 'interpretation': check['meaning']})
