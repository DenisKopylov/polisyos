def git(*args):
    return subprocess.check_output(['git', *args], cwd=REPO)

def digest(data):
    return hashlib.sha256(data).hexdigest()

def bind(sha, path):
    data = git('show', f'{sha}:{path}')
    return {'source_sha': sha, 'path': path, 'git_blob': git('rev-parse', f'{sha}:{path}').decode().strip(), 'bytes': len(data), 'sha256': digest(data)}

def junit(name):
    root = ET.fromstring((OUT / name).read_bytes())
    suites = [root] if root.tag == 'testsuite' else root.findall('testsuite')
    counts = {key: sum(int(s.attrib.get(key, 0)) for s in suites) for key in ['tests', 'failures', 'errors', 'skipped']}
    failures = [t.attrib for t in root.iter('testcase') if t.find('failure') is not None]
    return counts, failures, root

    return [ast.dump(n, include_attributes=False) for n in ast.walk(tree) if isinstance(n, ast.FunctionDef) and n.name == 'execute']
