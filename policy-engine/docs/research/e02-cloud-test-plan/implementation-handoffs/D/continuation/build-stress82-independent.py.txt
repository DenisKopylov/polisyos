def git(*args):
    return subprocess.check_output(['git', *args], cwd=REPO)

def digest(data):
    return hashlib.sha256(data).hexdigest()

def source_file(sha, path):
    data = git('show', f'{sha}:{path}')
    return {'path': path, 'source_sha': sha, 'git_blob': git('rev-parse', f'{sha}:{path}').decode().strip(), 'bytes': len(data), 'sha256': digest(data)}

def summary_junit(path):
    root = ET.fromstring(path.read_bytes())
    suites = [root] if root.tag == 'testsuite' else list(root.findall('testsuite'))
    return {k: sum(int(s.attrib.get(k, 0)) for s in suites) for k in ['tests', 'failures', 'errors', 'skipped']}

origin_checks = []
for name, sha in [('challenge-frozen-complete.txt', SOURCE), ('schema-context-frozen-b-complete.txt', 'c2cbd72c0d11dab73b277c97f389840dfd187b17'), ('projection-red-complete.txt', RED)]:
    text = (OUT / name).read_text()
    line = next(line for line in text.splitlines() if line.startswith('RUNTIME_ENVIRONMENT='))
    footer = json.loads(line.split('=', 1)[1])
    for module in footer['modules']:
        if module['path'] is None:
            origin_checks.append({'log': name, 'source_sha': sha, 'module': module['module'], 'status': 'UNIMPORTED'})
            continue
        path = 'policy-engine/' + module['path'].split('/policy-engine/', 1)[1]
        expected = source_file(sha, path)
        actual = module['sha256']
        assert actual == expected['sha256'], (name, module['module'])
        origin_checks.append({'log': name, 'source_sha': sha, 'module': module['module'], 'loaded_path': module['path'], 'source_path': path, 'sha256': actual, 'status': 'MATCH'})
