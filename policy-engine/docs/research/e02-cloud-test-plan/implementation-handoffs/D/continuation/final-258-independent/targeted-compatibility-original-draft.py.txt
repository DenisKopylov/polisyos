PREFIX = 'policy-engine/docs/research/e02-cloud-test-plan/implementation-handoffs/D/continuation/'


def tree(sha: str) -> dict:
    values = {}
    for row in subprocess.check_output(['git', '-C', REPO, 'ls-tree', '-rz', sha]).split(b'\0'):
        if row:
            metadata, path = row.split(b'\t', 1)
            values[path.decode()] = metadata.decode().split()[2]
    return values
