"""Read captured pytest evidence only; never imports product or starts tests."""
from __future__ import annotations

import argparse
from collections import Counter, defaultdict
import hashlib
import json
from pathlib import Path
import re
import xml.etree.ElementTree as ET


def read_json(path: Path) -> dict:
    return json.loads(path.read_text())


def identity(path: Path) -> dict:
    value = path.read_bytes()
    return {"bytes": len(value), "sha256": hashlib.sha256(value).hexdigest()}


def xml_key(nodeid: str) -> tuple[str, str]:
    before, sep, parameter = nodeid.partition("[")
    parts = before.split("::")
    parts[0] = re.sub(r"\.py$", "", parts[0].replace("/", "."))
    parts[-1] += sep + parameter
    return ".".join(parts[:-1]), parts[-1]


def outcome(element: ET.Element) -> str:
    for tag, name in (("failure", "FAIL"), ("error", "ERROR"), ("skipped", "SKIP")):
        if element.find(tag) is not None:
            return name
    return "PASS"


def expected_rows(phases: list[dict]) -> Counter:
    by = {row["when"]: row["outcome"] for row in phases}
    if by.get("call") == "failed" and by.get("teardown") == "failed":
        # Native pytest xunit emits separate FAIL and ERROR rows for this overlap.
        return Counter({"FAIL": 1, "ERROR": 1})
    if by.get("setup") == "failed" or by.get("teardown") == "failed":
        return Counter({"ERROR": 1})
    if by.get("call") == "failed":
        return Counter({"FAIL": 1})
    if by.get("setup") == "skipped" or by.get("call") == "skipped":
        return Counter({"SKIP": 1})
    return Counter({"PASS": 1})


def reconcile(capture: Path) -> dict:
    paths = {name: capture / name for name in (
        "execution-manifest.json", "collected-nodeids.json", "events.jsonl",
        "observer-summary.json", "junit.xml", "result.json")}
    if not paths["execution-manifest.json"].is_file() and (capture / "manifest.json").is_file():
        paths["execution-manifest.json"] = capture / "manifest.json"
    report = {"schema": "policyos.e02.independent_pytest_denominator.v1",
              "capture_root": str(capture), "read_only_metadata": True,
              "new_runtime": False, "formal_closed": 0, "G_accepted": False,
              "input_artifact_identities": {name: identity(path) for name, path in paths.items() if path.is_file()}}
    missing = [name for name, path in paths.items() if not path.is_file()]
    report["missing_final_artifacts"] = missing
    if missing:
        report["verdict"] = "ERROR_OR_ACTIVE_CAPTURE_INCOMPLETE_NO_RUNTIME_INFERENCE"
        return report
    observation = read_json(paths["observer-summary.json"])
    collected = read_json(paths["collected-nodeids.json"])
    actual = collected["nodeids"]
    cases = observation["cases"]
    event_phases: dict[str, list[dict]] = defaultdict(list)
    collection = []
    starts, finishes = Counter(), Counter()
    xfails, failures = [], []
    event_line_count = 0
    with paths["events.jsonl"].open() as stream:
        for index, line in enumerate(stream, 1):
            event = json.loads(line)
            event_line_count = index
            kind = event["kind"]
            if kind == "case_start":
                starts[event["nodeid"]] += 1
            elif kind == "case_finish":
                finishes[event["nodeid"]] += 1
            elif kind == "case_report":
                event_phases[event["nodeid"]].append({key: event[key] for key in ("when", "outcome", "duration_seconds")})
                if event.get("wasxfail"):
                    xfails.append({"nodeid": event["nodeid"], "phase": event["when"], "outcome": event["outcome"], "reason": event["wasxfail"], "events_line": index})
                if event["outcome"] in ("failed", "skipped"):
                    failures.append({"nodeid": event["nodeid"], "phase": event["when"], "outcome": event["outcome"], "events_line": index, "reason_sha256": hashlib.sha256(event.get("longrepr", "").encode()).hexdigest(), "full_reason": "events.jsonl:" + str(index)})
            elif kind == "collection_report":
                collection.append({"nodeid": event["nodeid"], "outcome": event["outcome"], "events_line": index, "reason": event.get("longrepr", "")})
    raw_xml = ET.fromstring(paths["junit.xml"].read_bytes())
    rows = defaultdict(Counter)
    xml_details = []
    for index, element in enumerate(raw_xml.iter("testcase")):
        key = (element.get("classname", ""), element.get("name", ""))
        result = outcome(element)
        rows[key][result] += 1
        xml_details.append({"index": index, "class": key[0], "name": key[1], "outcome": result,
                            "failure_elements": len(element.findall("failure")),
                            "error_elements": len(element.findall("error")),
                            "skip_elements": len(element.findall("skipped"))})
    gaps = []
    if len(actual) != len(set(actual)) or collected["count"] != len(actual) or collected["unique_count"] != len(set(actual)):
        gaps.append({"operation": "unique_collected_count"})
    if actual != observation["collected_nodeids"]:
        gaps.append({"operation": "collected_ids_vs_observer"})
    if set(actual) != set(cases):
        gaps.append({"operation": "collected_vs_case_id_set", "missing": sorted(set(actual) - set(cases)), "extra": sorted(set(cases) - set(actual))})
    expected = defaultdict(Counter)
    identity_to_nodes = defaultdict(list)
    phase_outcomes = Counter()
    overlaps = []
    for nodeid in actual:
        case = cases.get(nodeid, {})
        phases = case.get("phases", [])
        shape = [p["when"] for p in phases]
        allowed = ["setup", "call", "teardown"] if phases and phases[0]["outcome"] == "passed" else ["setup", "teardown"]
        valid = (shape == allowed and starts[nodeid] == finishes[nodeid] == 1
                 and "started_wall_ns" in case and "finished_wall_ns" in case
                 and case["finished_wall_ns"] >= case["started_wall_ns"]
                 and event_phases[nodeid] == phases)
        if not valid:
            gaps.append({"operation": "complete_case_phase_shape", "nodeid": nodeid, "shape": shape,
                         "starts": starts[nodeid], "finishes": finishes[nodeid], "events_equal_summary": event_phases[nodeid] == phases})
        key = xml_key(nodeid)
        identity_to_nodes[key].append(nodeid)
        expected[key].update(expected_rows(phases))
        phase_outcomes.update((p["when"], p["outcome"]) for p in phases)
        if sum(expected_rows(phases).values()) > 1:
            overlaps.append({"nodeid": nodeid, "expected_junit_rows": dict(expected_rows(phases))})
    for row in collection:
        key = xml_key(row["nodeid"])
        expected[key].update({"ERROR" if row["outcome"] == "failed" else "SKIP": 1})
    for key in set(expected) | set(rows):
        if expected[key] != rows[key]:
            gaps.append({"operation": "junit_identity_outcome_multiset", "class": key[0], "name": key[1], "expected": dict(expected[key]), "actual": dict(rows[key]), "collected_nodeids": identity_to_nodes[key]})
    collisions = [{"class": key[0], "name": key[1], "nodeids": values} for key, values in identity_to_nodes.items() if len(values) != 1]
    if collisions:
        gaps.append({"operation": "junit_identity_collision", "collisions": collisions})
    unique_bad = sorted({row["nodeid"] for row in failures if row["outcome"] == "failed"})
    skipped = sorted({row["nodeid"] for row in failures if row["outcome"] == "skipped"})
    report.update(event_line_count=event_line_count, collected_count=len(actual), unique_collected_count=len(set(actual)),
                  case_summary_count=len(cases), case_start_count=sum(starts.values()), case_finish_count=sum(finishes.values()),
                  phase_reports={when + ":" + result: count for (when, result), count in phase_outcomes.items()},
                  collection_reports=collection, runtime_failure_or_skip_refs=failures,
                  unique_runtime_FAILED_case_count=len(unique_bad), unique_runtime_FAILED_cases=unique_bad,
                  runtime_SKIPPED_case_count=len(skipped), runtime_SKIPPED_cases=skipped,
                  xfail_reports=xfails, junit_rows=len(xml_details), junit_outcomes=dict(Counter(row["outcome"] for row in xml_details)),
                  junit_FAIL_ERROR_rows=[row for row in xml_details if row["outcome"] in ("FAIL", "ERROR")],
                  call_failure_teardown_error_overlap=overlaps, gaps=gaps,
                  verdict="PASS_DENOMINATOR_IDENTITY_RECONCILIATION_ONLY" if not gaps else "ERROR_DENOMINATOR_OR_IDENTITY_GAP",
                  numerical_properties="Not inferred from import, skip, xfail or denominator success; actual source/backend execution is reviewed separately.")
    return report


def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("capture", type=Path)
    parser.add_argument("--output", type=Path, required=True)
    args = parser.parse_args()
    assert not args.output.exists(), "append-only fresh review output"
    result = reconcile(args.capture)
    args.output.write_text(json.dumps(result, indent=2) + "\n")
    print(json.dumps({key: result.get(key) for key in ("verdict", "collected_count", "junit_rows", "unique_runtime_FAILED_case_count", "gaps")}))


if __name__ == "__main__":
    main()
