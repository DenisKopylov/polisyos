def git(*args):
    return subprocess.check_output(["git", *args], cwd=root, text=True).strip()

def binding(relative):
    data = (product / relative).read_bytes()
    return {"path": "policy-engine/" + relative, "git_blob": git("rev-parse", "HEAD:policy-engine/" + relative), "bytes": len(data), "sha256": hashlib.sha256(data).hexdigest()}

def exports(relative):
    tree = ast.parse((product / relative).read_text())
    names = []
    for node in tree.body:
        if isinstance(node, (ast.Assign, ast.AnnAssign)):
            targets = node.targets if isinstance(node, ast.Assign) else [node.target]
            if any(isinstance(t, ast.Name) and t.id == "__all__" for t in targets):
                names = ast.literal_eval(node.value)
        if isinstance(node, ast.Expr) and isinstance(node.value, ast.Call):
            call = node.value
            if isinstance(call.func, ast.Attribute) and isinstance(call.func.value, ast.Name) and call.func.value.id == "__all__" and call.func.attr == "extend":
                names.extend(ast.literal_eval(call.args[0]))
    return {**binding(relative), "names": names, "export_count": len(names), "unique_count": len(set(names))}
