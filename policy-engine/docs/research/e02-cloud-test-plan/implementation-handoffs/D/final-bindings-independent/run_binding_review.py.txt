def require(condition: bool, message: str) -> None:
    if not condition:
        raise ValueError(message)


def git(*args: str) -> bytes:
    return subprocess.check_output(["git", "--no-optional-locks", *args], cwd=REPO)


def blob(sha: str, path: str) -> bytes:
    return git("show", f"{sha}:{path}")


def digest(raw: bytes) -> str:
    return hashlib.sha256(raw).hexdigest()


def describe(raw: bytes) -> dict[str, Any]:
    return {"bytes": len(raw), "sha256": digest(raw)}


def normalize(path: str) -> str:
    if path.startswith("implementation-handoffs/"):
        return "policy-engine/docs/research/e02-cloud-test-plan/" + path
    if path.startswith("docs/"):
        return "policy-engine/" + path
    return path


def checked(sha: str, path: str, expected: str, length: int | None) -> dict[str, Any]:
    path = normalize(path)
    raw = blob(sha, path)
    require(digest(raw) == expected, f"hash mismatch {sha}:{path}")
    require(length is None or len(raw) == length, f"length mismatch {sha}:{path}")
    return {"source_sha": sha, "path": path, **describe(raw), "status": "MATCH"}


def path_cells(value: Any, publisher: str, at: str = "") -> list[dict[str, Any]]:
    cells: list[dict[str, Any]] = []
    if isinstance(value, dict):
        path = value.get("path")
        expected = value.get("sha256")
        if isinstance(path, str) and not path.startswith("/") and isinstance(expected, str):
            sha = value.get("source_sha", value.get("sha", publisher))
            cell = checked(sha, path, expected, value.get("bytes"))
            cell["receipt_pointer"] = at
            if "decoded_sha256" in value:
                decoded = gzip.decompress(blob(sha, normalize(path)))
                require(digest(decoded) == value["decoded_sha256"], f"decoded hash {at}")
                require(len(decoded) == value["decoded_bytes"], f"decoded length {at}")
                cell["decoded"] = describe(decoded)
            cells.append(cell)
        for key, child in value.items():
            child_publisher = publisher
            if at == "" and key == "unchanged_tests" and "source_commit" in value:
                child_publisher = value["source_commit"]
            cells.extend(path_cells(child, child_publisher, f"{at}/{key}"))
    elif isinstance(value, list):
        for i, child in enumerate(value):
            cells.extend(path_cells(child, publisher, f"{at}/{i}"))
    return cells


def junit(raw: bytes) -> dict[str, Any]:
    tree = ET.fromstring(raw)
    tests = list(tree.iter("testcase"))
    failed = [t.get("name") for t in tests if t.find("failure") is not None]
    errors = [t.get("name") for t in tests if t.find("error") is not None]
    skipped = [t.get("name") for t in tests if t.find("skipped") is not None]
    counts = {"tests": len(tests), "failures": len(failed), "errors": len(errors), "skipped": len(skipped)}
    suites = list(tree.iter("testsuite"))
    declared = {k: sum(int(s.get(k, "0")) for s in suites) for k in counts}
    require(declared == counts, "JUnit declared/case counts disagree")
    return {**counts, "passed": len(tests) - len(failed) - len(errors) - len(skipped), "failed_names": failed}


def encoded_bundle(sha: str, output: dict[str, Any], path_key: str) -> tuple[dict[str, Any], dict[str, bytes]]:
    path = normalize(output[path_key])
    packed = blob(sha, path)
    require(digest(packed) == output.get("compressed_rawfile_sha256", output.get("rawfile_sha256")), "bundle compressed hash")
    require(len(packed) == output["compressed_bytes"], "bundle compressed length")
    raw = gzip.decompress(packed)
    require(digest(raw) == output["uncompressed_rawfile_sha256"], "bundle decoded hash")
    require(len(raw) == output["uncompressed_bytes"], "bundle decoded length")
    payload = json.loads(raw)
    entries = payload["complete_raw_files"]
    require(len(entries) == output.get("file_count", output.get("full_raw_files")), "bundle denominator")
    files: dict[str, bytes] = {}
    cells = []
    for item in entries:
        name = item["path"]
        require(name not in files, f"duplicate bundle path {name}")
        content = base64.b64decode(item["content_base64"], validate=True)
        require(digest(content) == item["rawfile_sha256"], f"bundle file hash {name}")
        require(len(content) == item["bytes"], f"bundle file bytes {name}")
        files[name] = content
        cells.append({"path": name, **describe(content), "status": "MATCH"})
    return {"source_sha": sha, "path": path, "compressed": describe(packed), "decoded": describe(raw), "file_count": len(files), "unique_paths": len(files), "files": cells}, files
