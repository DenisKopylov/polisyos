def git(*args):
    return subprocess.check_output(['git', *args], cwd=REPO)

def blob(sha, path):
    assert git('cat-file', '-t', f'{sha}:{path}').strip() == b'blob', (sha, path)
    return git('show', f'{sha}:{path}')

def digest(raw):
    return hashlib.sha256(raw).hexdigest()

def ref(sha, path):
    raw = blob(sha, path)
    q = dict(source_sha=sha, path=path, bytes=len(raw), sha256=digest(raw))
    AUDIT.append(q)
    return q

def document(sha, path):
    return json.loads(blob(sha, path))

def supplied_ref(sha, q):
    actual = ref(sha, q['path'])
    assert actual['bytes'] == q['bytes'], (actual, q)
    assert actual['sha256'] == q['sha256'], (actual, q)
    return actual

def counts(raw):
    root = ET.fromstring(raw)
    suites = list(root.iter('testsuite'))
    result = {k: sum(int(s.get(k, 0)) for s in suites) for k in ('tests', 'failures', 'errors', 'skipped')}
    result['passed'] = result['tests'] - result['failures'] - result['errors'] - result['skipped']
    return result

def byte_locator(raw, start, end):
    lines = raw.splitlines(keepends=True)
    offset = sum(map(len, lines[:start - 1]))
    selected = b''.join(lines[start - 1:end])
    return offset, selected
