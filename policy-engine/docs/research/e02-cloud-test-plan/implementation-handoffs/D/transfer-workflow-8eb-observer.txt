class NativeObserver:
    def __init__(self):
        self.results=[]
        self.current=None
        self.depth=0
    def profile(self, frame, event, arg):
        if self.current is None:
            return
        file=frame.f_code.co_filename
        name=frame.f_code.co_name
        if file.endswith('/search/strategies/bayesian.py') and name=='set_state':
            if event=='call': self.depth+=1
            elif event=='return': self.depth-=1
        if event!='call':
            return
        phase='restore' if self.depth else 'ordinary'
        key=None
        if file.endswith('/botorch/fit.py') and name=='fit_gpytorch_mll':
            key='native_fit_gpytorch_mll'
        elif file.endswith('/botorch/models/gp_regression.py') and name=='__init__' and type(frame.f_locals.get('self')).__name__=='SingleTaskGP':
            key='native_SingleTaskGP_constructor'
        elif file.endswith('/gpytorch/mlls/marginal_log_likelihood.py') and name=='__init__' and type(frame.f_locals.get('self')).__name__=='ExactMarginalLogLikelihood':
            key='native_ExactMarginalLogLikelihood_constructor'
        elif file.endswith('/search/strategies/bayesian.py') and name=='_fit_full_gp':
            key='receiver_full_fit'
        if key:
            self.current[phase+'.'+key]+=1
    @pytest.hookimpl(hookwrapper=True)
    def pytest_runtest_call(self, item):
        if 'test_transfer_workflow.py' not in item.nodeid:
            yield
            return
        self.current=Counter()
        self.depth=0
        sys.setprofile(self.profile)
        try:
            yield
        finally:
            sys.setprofile(None)
            self.results.append({'nodeid':item.nodeid,'observed_calls':dict(self.current)})
            self.current=None
    def pytest_sessionfinish(self, session, exitstatus):
        out={'method':'sys.setprofile call observation only; no replaced native model, MLL, fitting, acquisition or checkpoint function','exitstatus':exitstatus,'tests':self.results}
        (ROOT/'native-call-observations.json').write_text(json.dumps(out,indent=2)+'\n')
        print('NATIVE_CALL_OBSERVATIONS='+json.dumps(out,sort_keys=True))
