def context(root,run):
    return ExecutionContext(store=FileSystemCAS(root),run=run,logger=logging.getLogger('independent-legacy-consumer'))

def state(ref,sim):
    return ExperimentState(run_id='independent-legacy-readback',inputs={INPUT_CALIBRATION_REPORT_REF:ref},
        artifacts_index={ARTIFACT_SIMULATION_RESULT_REF:sim},params={
        'propagation_config':{'preferred_method':'delta','delta_covariance_jitter':0},
        'propagation_sensitivity':{'difference':{'A.rate':1.,'B.rate':-1.}}})

def raw(store,payload,kind='foundry.calibration_report',schema='polisyos.foundry.CalibrationReport',version='2.0',inputs=()):
    return store.put_json(payload,PutOptions(kind=kind,media_type='application/json',
        schema=SchemaInfo(name=schema,version=version),inputs=list(inputs)),
        canon_spec=CanonSpec(forbid_floats=False,exclude_none=False))


def test_actual_calibrator_v2_cas_fresh_node_retains_tied_covariance_and_nongating(native):
    root,store,run,ref,_,sim=native
    fresh=context(root,run)
    loaded=load_calibration_report(fresh.store,ref)
    assert loaded.schema_version=='2.0'
    collected=node._collect_input_envelopes(fresh,state(ref,sim))
    assert set(collected)=={'A.rate','B.rate'}
    assert loaded.coordinate_projection.field_order==('A.rate','B.rate')
    assert loaded.coordinate_projection.coordinate_order==('shared_rate',)
    assert loaded.coordinate_projection.matrix==((1.,),(1.,))
    # Each of two scalar tax nodes contributes derivative100 to one shared
    # coordinate; known observational sigma10 gives H=(100+100)^2/10^2=400.
    # Thus coordinate variance1/400=.0025; J=[1,1]^T gives a singular all-.0025
    # field covariance. Var(A-B)=[1,-1]Sigma[1,-1]^T=0, independent proxy .005.
    cov=np.asarray(build_covariance_matrix(['A.rate','B.rate'],collected,use_full_covariance=True,jitter=0))
    np.testing.assert_allclose(cov,np.full((2,2),1/400),rtol=1e-4,atol=1e-8)
    assert np.linalg.matrix_rank(cov)==1
    assert np.array([1.,-1.])@cov@np.array([1.,-1.])==0
    assert all(env.gate_eligible is False and env.confidence_level is None for env in collected.values())
    outcome=node.PropagateUncertaintyNode().execute(fresh,state(ref,sim))
    assert outcome.status=='ok'
    sim_loaded=SimulationResult.model_validate(from_canonical_bytes(fresh.store.get_bytes(outcome.state.artifacts_index[ARTIFACT_SIMULATION_RESULT_REF].artifact_id)))
    output=load_uncertainty_envelope(FileSystemCAS(root),sim_loaded.uncertainty_envelopes['difference'])
    assert output.gate_eligible is False
    from polisyos.ir.analytics.uncertainty import PosteriorSamplesCarrier
    assert isinstance(output.distribution_payload,PosteriorSamplesCarrier)
    assert len(output.distribution_payload.samples)==1000
    assert all(value==0 for value in output.distribution_payload.samples)
    report=from_canonical_bytes(fresh.store.get_bytes(outcome.state.artifacts_index[ARTIFACT_PROPAGATION_REPORT_REF].artifact_id))
    assert report['input_envelope_count']==2 and report['mapped_params']==['A.rate','B.rate']
    assert report['methods']==['monte_carlo']  # Heuristic carriers intentionally take sampling before preferred Delta.
    RESULTS.append(dict(test='real_calibrator_v2_fresh_node',cas=str(root),input_ref=ref.model_dump(mode='json'),
        config_edges=[i.model_dump(mode='json') for i in store.get_manifest(ref).inputs],
        covariance=cov.tolist(),independent_marginals_variance=.005,true_tied_difference_variance=0,
        node_status=outcome.status,output_ref=sim_loaded.uncertainty_envelopes['difference'].model_dump(mode='json'),
        propagation_report=report,output_gate_eligible=output.gate_eligible))


@pytest.mark.parametrize('defect',['wrong_kind','wrong_schema','schema_payload_version','unsupported_schema_version',
    'missing_config','duplicate_config','wrong_config_kind','bad_config_payload',
    'missing_objective_profile','wrong_objective_profile_kind','wrong_objective_profile_payload','invalid_report_payload'])
def test_report_admission_precedes_any_propagation_callback(native,monkeypatch,defect):
    root,store,run,ref,_,sim=native
    payload=from_canonical_bytes(store.get_bytes(ref.artifact_id))
    payload['diagnostics'].append('independent-negative-'+defect)
    manifest=store.get_manifest(ref)
    inputs=list(manifest.inputs);kind='foundry.calibration_report';schema='polisyos.foundry.CalibrationReport';version='2.0'
    if defect=='wrong_kind':kind='foundry.funnel_calibration_report'
    elif defect=='wrong_schema':schema='polisyos.foundry.FunnelCalibrationReport'
    elif defect=='schema_payload_version':version='1.0'
    elif defect=='unsupported_schema_version':version='9.0'
    elif defect=='missing_config':inputs=[i for i in inputs if i.role!='calibration_config']
    elif defect=='duplicate_config':
        old=next(i for i in inputs if i.role=='calibration_config');cfg=from_canonical_bytes(store.get_bytes(old.artifact_id));cfg['seed']=91
        duplicate=raw(store,cfg,kind='foundry.calibration_config',schema='polisyos.ir.CalibrationConfig',version=cfg['schema_version'])
        inputs.append(InputRef(artifact_id=duplicate.artifact_id,role='calibration_config',manifest_profile_sha256=duplicate.manifest_profile_sha256))
    elif defect in ('wrong_config_kind','bad_config_payload'):
        old=next(i for i in inputs if i.role=='calibration_config');cfg=from_canonical_bytes(store.get_bytes(old.artifact_id))
        cfg['targets']='invalid' if defect=='bad_config_payload' else cfg['targets']
        replacement=raw(store,cfg,kind='fake.config' if defect=='wrong_config_kind' else 'foundry.calibration_config',
            schema='polisyos.ir.CalibrationConfig',version=cfg['schema_version'])
        inputs=[i for i in inputs if i.role!='calibration_config']+[InputRef(artifact_id=replacement.artifact_id,role='calibration_config',manifest_profile_sha256=replacement.manifest_profile_sha256)]
    elif defect=='missing_objective_profile':inputs=[i for i in inputs if i.role!='calibration_objective_profile']
    elif defect in ('wrong_objective_profile_kind','wrong_objective_profile_payload'):
        old=next(i for i in inputs if i.role=='calibration_objective_profile');profile=from_canonical_bytes(store.get_bytes(old.artifact_id))
        if defect=='wrong_objective_profile_payload':profile['identity']='changed-present-profile'
        replacement=raw(store,profile,kind='fake.profile' if defect=='wrong_objective_profile_kind' else 'foundry.calibration_objective_profile',
            schema='polisyos.foundry.GaussianObservationProfile',version='1.0')
        inputs=[i for i in inputs if i.role!='calibration_objective_profile']+[InputRef(artifact_id=replacement.artifact_id,role='calibration_objective_profile',manifest_profile_sha256=replacement.manifest_profile_sha256)]
    else:payload['total_loss']='not-numeric'
    bad=raw(store,payload,kind,schema,version,inputs)
    with pytest.raises(ValueError):load_calibration_report(FileSystemCAS(root),bad)
    calls=[]
    def forbidden(*args,**kwargs):calls.append(1);raise AssertionError('invalid report reached propagation callback')
    monkeypatch.setattr(node.PropagationDispatcher,'propagate',forbidden)
    fresh=context(root,run);bad_state=state(bad,sim)
    with pytest.raises(ValueError):node._collect_input_envelopes(fresh,bad_state)
    with pytest.raises(ValueError) as rejected:node.PropagateUncertaintyNode().execute(fresh,bad_state)
    assert calls==[]
    RESULTS.append(dict(test='refusal',defect=defect,ref=bad.model_dump(mode='json'),node_status='refused',error=str(rejected.value),propagation_calls=0))

def test_reader_property_removal_reopens_populated_wrong_kind_report(native,monkeypatch):
    root,store,run,ref,_,sim=native
    payload=from_canonical_bytes(store.get_bytes(ref.artifact_id));payload['diagnostics'].append('independent-removal-fake')

        node.PropagateUncertaintyNode().execute(context(root,run),state(bad,sim))
    assert calls==[1]

def test_exact_902d_falsifier_input_replayed_on_frozen_consumer(native,monkeypatch):
    from polisyos.core.artifacts.manifest import ArtifactRef
    _,_,run,_,_,_=native
    prefix=Path('/workspace/e02-E-pr38-r2-receipts/independent-cal-reviewer/legacy-calibration')

    store=FileSystemCAS(prefix/'baseline-native-cas-902d')
    ref=ArtifactRef.model_validate(decided['input_ref'])
    before=hashlib.sha256(store.get_bytes(ref.artifact_id)).hexdigest()
