def git(*args):

def write(path, data):
    path.write_text(json.dumps(data, indent=2) + '\n')

def preserve(source, target):
    target.mkdir(parents=True, exist_ok=True)
    records = []
    for p in sorted(source.rglob('*')):
        if not p.is_file():
            continue
        q = target / p.relative_to(source)
        if q.suffix == '.py':
            q = q.with_suffix('.py.txt')
        q.parent.mkdir(parents=True, exist_ok=True)
        shutil.copyfile(p, q)
        content = p.read_bytes()
        assert content == q.read_bytes()
        records.append(dict(source_absolute_path=str(p), git_path=str(q.relative_to(ROOT)), bytes=len(content), sha256=hashlib.sha256(content).hexdigest()))
    write(target / 'copy-index.json', dict(schema='policyos.e02.evidence_custody.v1', candidate_sha=FROZEN, files=records))

def collect_abs(value):
    if isinstance(value, str) and value.startswith('/') and '(' not in value:
        candidate_strings.add(value)

            collect_abs(item)

            if 'preserve' not in key and 'retain' not in key:
