def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--repository", type=Path, required=True)
    parser.add_argument("--candidate", required=True)
    parser.add_argument("--handoff-index", type=Path, required=True)
    parser.add_argument("--output", type=Path, required=True)
    args = parser.parse_args()
    def git(*arguments: str) -> str:
        return subprocess.check_output(["git", *arguments], cwd=args.repository, text=True).strip()
    candidate = git("rev-parse", args.candidate)
    tree = git("rev-parse", candidate + "^{tree}")
    identities = {}
    def read(path: str) -> str:
        raw = subprocess.check_output(["git", "show", candidate + ":" + path], cwd=args.repository)
        identities[path] = {"git_blob": git("rev-parse", candidate + ":" + path), "sha256": hashlib.sha256(raw).hexdigest()}
        return raw.decode()
    manifest = json.loads(read(ORIGINAL + "/bundle_manifest.json"))
    coverage = json.loads(read(E02 + "/closure-decisions/coverage.json"))
    bundle_owners = list(csv.DictReader(io.StringIO(read(E02 + "/execution-organization/bundle-owners.tsv")), delimiter="\t"))
    finding_owners = list(csv.DictReader(io.StringIO(read(E02 + "/execution-organization/finding-owners.tsv")), delimiter="\t"))
    ledger = json.loads(read(LEDGER))
    source_documents = {name: read(ORIGINAL + "/source/" + name + "_original.md") for name in ("B_r19", "LA_r09")}
    bundles_by_id = {item["id"]: item for item in manifest["bundles"]}
    covered = {item["id"]: item for item in coverage["findings"] if item["unit"] == "E"}
    owned = {item["finding_id"]: item for item in finding_owners if item["unit"] == "E"}
    bundle_ids = {item["bundle_id"] for item in bundle_owners if item["unit"] == "E"}
    ledger_rows = {item["id"]: item for item in ledger["rows"]}
    assert len(bundle_owners) == 127 and len(finding_owners) == len(ledger_rows) == 282
    assert set(ROWS) == set(owned) == set(covered), (set(ROWS) ^ set(owned), set(ROWS) ^ set(covered))
    attachments = []
    for bundle_id in sorted(bundle_ids):
        item = bundles_by_id[bundle_id]
        ids = [*item["findings"], *item["legacy_cards"]]
        attachments.extend((bundle_id, finding_id) for finding_id in ids)
        raw_bundle = read(ORIGINAL + "/bundles/" + bundle_id + ".md")
        for finding_id in ids:
            kind = "LA" if finding_id.startswith("LA-") else "B"
            assert "<!-- SOURCE_BEGIN " + kind + ":" + finding_id + " -->" in raw_bundle
    assert len(bundle_ids) == 22 and len(owned) == 54 and len(attachments) == 55
    assert collections.Counter(fid for _, fid in attachments)["LA-051"] == 2
    assert {fid for _, fid in attachments} == set(owned)
    handoffs = json.loads(args.handoff_index.read_text())
    family_receipts = {item["family"]: item for item in handoffs}
    assert len(family_receipts) == 8
    rows = []
    for finding_id in sorted(owned, key=lambda name: (name.startswith("LA-"), int(name.removeprefix("LA-").removeprefix("B")))):
        family, criterion, evidence, remaining, owner = ROWS[finding_id]
        ledger_status = ledger_rows[finding_id]["status"]
        assert ledger_status == owned[finding_id]["source_status"] == covered[finding_id]["ledger_status_historical"]
        criterion_refs = []
        for ref in covered[finding_id]["criterion_refs"]:
            start, end = ref["lines"]
            source = source_documents[ref["document"]]
            block = "".join(source.splitlines(keepends=True)[start - 1:end])
            assert hashlib.sha256(block.encode()).hexdigest() == ref["sha256"], finding_id
            criterion_refs.append({"path": ORIGINAL + "/source/" + ref["document"] + "_original.md", "lines": [start, end], "sha256": ref["sha256"]})
        receipt = family_receipts[family]
        verdict = "closed" if finding_id == "B198" else "held" if finding_id in ("B201", "B202") else "limited"
        rows.append({
            "finding_id": finding_id,
            "bundle_ids": sorted(bundle for bundle, fid in attachments if fid == finding_id),
            "ledger_status_preserved": ledger_status,
            "proposed_verdict": verdict,
            "criterion_property_assessment": (
                "receipt-supported bounded criterion-complete candidate; final independent source replay pending"
                if finding_id in PROPERTY_COMPLETE_CANDIDATES else
                "criterion includes an active producer/consumer/method/semantic owner boundary"
            ),
            "post_freeze_owner_adjudication_recommendation": (
                "Retain historical closed regression unless a new defining-property falsifier appears"
                if finding_id == "B198" else
                "Owner may adjudicate bounded finding closure after exact final replay and review; do not require unrelated production or deployment authority"
                if finding_id in PROPERTY_COMPLETE_CANDIDATES else
                "Retain limited/held at the stated active boundary; PASS counts or code acceptance do not resolve it"
            ),
            "formal_ledger_change": False,
            "criterion": criterion,
            "criterion_refs": criterion_refs,
            "deciding_scoped_evidence": evidence,
            "receipt_ref": receipt["handoff_path"] + "@" + receipt["branch_head"],
            "implementation_commits": receipt["data"]["implementation_commits"],
            "implementation_tree": receipt["data"]["candidate_tree_sha"],
            "independent_extra_evidence": "ddm01-independent.json@cb9fb322a45efcac1cbdfa04d5f59a03d7dbe90a" if finding_id == "LA-056" else None,
            "remaining_condition": remaining,
            "next_owner": owner,
            "final_frozen_wave": "UNRUN/pending at reconciliation",
            "bounded_import_criterion_closure_recommendation": "closed after final DDM input identity replay and owner adjudication" if finding_id == "LA-056" else None,
        })
    args.output.mkdir(parents=True, exist_ok=True)
    output = {
        "schema": "policyos.e02.independent_criterion_reconciliation.v1",
        "unit": "E", "reviewer": "closure_reconcile_r3 independent leaf",
        "review_source_sha": candidate, "review_source_tree": tree,
        "entry_base_sha": BASE,
        "denominator": {"manifest_bundles": len(manifest["bundles"]), "bundle_owner_rows": len(bundle_owners), "finding_owner_rows": len(finding_owners), "ledger_rows": len(ledger_rows), "E_bundles": len(bundle_ids), "E_unique_findings": len(owned), "E_bundle_links": len(attachments)},
        "ledger_status_counts_preserved": dict(collections.Counter(item["ledger_status_preserved"] for item in rows)),
        "proposal_counts": dict(collections.Counter(item["proposed_verdict"] for item in rows)),
        "bounded_property_complete_candidate_ids": sorted(PROPERTY_COMPLETE_CANDIDATES),
        "predicate_basis": "recomputed criterion/input-set reconciliation; native outcomes supplied by exact referenced author/independent receipts, not recounted as finding closure",
        "authority": "No ledger mutation and no final-wave claim. Code acceptance, method scope, formal closure and serving are separate decisions.",
        "input_identities": identities,
        "corrections": [
