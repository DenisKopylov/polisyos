def digest(path):
    with path.open('rb') as handle:
        return hashlib.file_digest(handle, 'sha256').hexdigest()

xml_counts = []
failed_cases = []
seen = Counter()
for file in sorted(WAVE.glob('*.xml')):
    root = ET.parse(file).getroot()
    cases = root.findall('.//testcase')
    counts = Counter()
    for case in cases:
        seen[(case.get('classname'), case.get('name'))] += 1
        kind = next((kind for kind in ('failure','error','skipped') if case.find(kind) is not None), 'pass')
        counts[kind] += 1
        if kind != 'pass':
            node = case.find(kind)
            failed_cases.append({'xml':file.name, 'classname':case.get('classname'), 'name':case.get('name'),
                                 'outcome':kind.upper(), 'message':node.get('message')})
    for suite in root.findall('testsuite'):
        assert int(suite.get('tests')) == len(cases)
        assert int(suite.get('failures')) == counts['failure']
        assert int(suite.get('errors')) == counts['error']
        assert int(suite.get('skipped')) == counts['skipped']
    xml_counts.append({'xml':file.name,'test_cases':len(cases), 'PASS':counts['pass'],
                       'FAIL':counts['failure'],'ERROR':counts['error'],'SKIP':counts['skipped'],
                       'sha256':digest(file)})
assert len(xml_counts) == 6
assert sum(x['test_cases'] for x in xml_counts) == 1125
assert sum(x['PASS'] for x in xml_counts) == 1123
assert len(failed_cases) == 2 and len(seen) == 1125 and all(count == 1 for count in seen.values())

arch_text = (WAVE/'architecture.stdout.txt').read_text()
reported_edges = []
new_E_edges = []
cache = {}
def imported_modules(sha, path):
    key = (sha,path)
    if key in cache: return cache[key]
    process = subprocess.run(['git','show',sha+':'+path], cwd=ROOT, text=True, capture_output=True)
    if process.returncode:
        cache[key]=set()
        return cache[key]
    tree=ast.parse(process.stdout)
    modules=set()
    for node in ast.walk(tree):
        if isinstance(node,ast.ImportFrom) and node.module: modules.add(node.module)
        elif isinstance(node,ast.Import): modules.update(alias.name for alias in node.names)
    cache[key]=modules
    return modules

for line in arch_text.splitlines():
    match=re.match(r'- New deep-import creep detected: (\S+) -> (\S+) \(([^)]+)\)\.',line)
    if not match: continue
    source,target,path=match.groups()
    path='policy-engine/'+path
    before=imported_modules(BASE,path)
    after=imported_modules(TARGET,path)
    row={'source_module':source,'target_module':target,'path':path,'E_changed_path':path in changed,
         'exact_target_import_present_at_base':target in before,
         'exact_target_import_present_at_candidate':target in after}
    reported_edges.append(row)
    if path in changed and target not in before and target in after: new_E_edges.append(row)
assert len(reported_edges) == 163
assert sum(row['E_changed_path'] for row in reported_edges) == 35
assert len(new_E_edges) == 12
