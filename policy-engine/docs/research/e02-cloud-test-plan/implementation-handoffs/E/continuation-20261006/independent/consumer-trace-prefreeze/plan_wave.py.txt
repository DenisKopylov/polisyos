#!/usr/bin/env python3
"""Partition the full immutable E manifest without selecting or running tests."""
from __future__ import annotations
import argparse
import json
from pathlib import Path
from collections import defaultdict


def main():
    parser=argparse.ArgumentParser()
    parser.add_argument('--audit',type=Path,required=True)
    parser.add_argument('--output',type=Path,required=True)
    args=parser.parse_args()
    audit=json.loads(args.audit.read_text())
    groups=defaultdict(list)
    for path in audit['test_wave']['existing_unique_paths']:
        if '/foundry/calibration/' in path or '/remediation/test_cal_' in path or path.endswith('test_propagate_welfare.py'):
            group='CAL_and_welfare_consumer'
        elif '/foundry/uncertainty/' in path or '/remediation/test_uqp_' in path or '/remediation/test_uqs_' in path or 'propagate_uncertainty' in path or path.endswith('tests/unit/ir/test_uncertainty.py'):
            group='MC_joint_law_support_and_Scientist_consumer'
        elif '/calibration/' in path or 'calibration_diagnostics_report' in path:
            group='PCL_continuous_persistence'
        elif '/ddm/' in path or '/remediation/test_ddm_' in path:
            group='DDM_runtime_and_facade'
        elif '/doe/' in path or '/remediation/test_doe_' in path or 'test_sensitivity_bridge.py' in path:
            group='DOE_SALib_and_consumer_factory'
        else:
            group='BKT_FRC_S10_and_adjacent_report_consumers'
        groups[group].append(path)
    output={
        'schema':'policyos.e02.frozen_wave_groups.v1',
        'target_sha':audit['target_sha'],
        'target_tree_sha':audit['target_tree_sha'],
        'total_paths':len(audit['test_wave']['existing_unique_paths']),
        'denominator':{'bundles':22,'findings':54,'id_to_bundle_links':55,'E_result_cells':175,'E_baseline_unique_test_paths':56},
        'source_manifest':str(args.audit),
        'groups':dict(groups),
        'group_counts':{group:len(paths) for group,paths in groups.items()},
        'resource_rules':'Groups may run concurrently on identical immutable source; isolate pytest temporary roots/CAS/scratch/cache and serialize shared mutable fixtures only. No thread/process/worker caps. Record actual backend SKIP/UNRUN separately. This partitions execution planning, not criterion acceptance by test count.',
    }
    assert sum(output['group_counts'].values())==output['total_paths']
    args.output.write_text(json.dumps(output,indent=2)+'\n')
    print(json.dumps({'target_sha':output['target_sha'],'total_paths':output['total_paths'],'group_counts':output['group_counts']},indent=2))

if __name__=='__main__':main()
