
import importlib
import json
import pickle
from datetime import UTC, datetime
from enum import Enum
from pathlib import Path
from pydantic import BaseModel, ValidationError
from jsonschema import Draft202012Validator, FormatChecker

canonical = importlib.import_module("polisyos.ddm.contracts.events")
legacy = importlib.import_module("polisyos.ddm.integration.events")
root = importlib.import_module("polisyos.ddm")
integration = importlib.import_module("polisyos.ddm.integration")
budget = importlib.import_module("polisyos.ddm.contracts.metric_budget")
mapper = importlib.import_module("polisyos.ddm.readiness.readiness_mapper")
assert mapper.MetricBudgetPolicy is budget.MetricBudgetPolicy
classes = {name: value for name, value in vars(canonical).items() if isinstance(value, type) and value.__module__ == canonical.__name__}
assert set(classes) == set(legacy.__all__), (set(classes), legacy.__all__)
for name, cls in classes.items():
    assert getattr(legacy, name) is cls
    assert getattr(integration, name) is cls
    if name in root.__all__:
        assert getattr(root, name) is cls
    assert pickle.loads(pickle.dumps(cls)) is cls
    historical = ("c" + legacy.__name__ + "\n" + name + "\n.").encode()
    assert pickle.loads(historical) is cls
    if issubclass(cls, Enum):
        for member in cls:
            assert pickle.loads(pickle.dumps(member)) is member
            assert cls(member.value) is member
for module in ("polisyos.ddm.readiness.readiness_mapper", "polisyos.ddm.contracts.metric_budget"):
    historical = ("c" + module + "\nMetricBudgetPolicy\n.").encode()
    assert pickle.loads(historical) is budget.MetricBudgetPolicy

}
assert set(samples) == {name for name, cls in classes.items() if issubclass(cls, BaseModel)}
for name, payload in samples.items():
    cls = classes[name]
    instance = cls.model_validate(payload)
    wire = instance.model_dump_json(by_alias=True)
    assert cls.model_validate_json(wire) == instance
    assert pickle.loads(pickle.dumps(instance)) == instance
    old_pickle = pickle.dumps(instance, protocol=0).replace(canonical.__name__.encode(), legacy.__name__.encode())
    assert pickle.loads(old_pickle) == instance
    Draft202012Validator(cls.model_json_schema(by_alias=True), format_checker=FormatChecker()).validate(json.loads(wire))
audit = canonical.CalibrationAudit.model_validate(samples["CalibrationAudit"])
assert audit.pass_ is True
assert audit.model_dump(mode="json", by_alias=True)["pass"] is True
assert canonical.CalibrationAudit.model_validate(dict(samples["CalibrationAudit"], pass_=True)).pass_ is True

manual_schema = json.loads(Path("src/polisyos/ddm/integration/shift_event.schema.json").read_text())
manual = Draft202012Validator(manual_schema, format_checker=FormatChecker())
generated = Draft202012Validator(canonical.ShiftDetectedEvent.model_json_schema(), format_checker=FormatChecker())
full_shift = canonical.ShiftDetectedEvent.model_validate(shift).model_dump(mode="json")
manual.validate(full_shift)
for channel, value in (("p_value",.01),("e_value",2.0),("ert",1000.0)):
    single = {key:value_ for key,value_ in full_shift.items() if key not in ("p_value","e_value","ert")}
    single[channel] = value
    canonical.ShiftDetectedEvent.model_validate(single)
    manual.validate(single)
missing = {key:value for key,value in full_shift.items() if key not in ("p_value","e_value","ert")}
try:
    canonical.ShiftDetectedEvent.model_validate(missing)
except ValidationError:
    pass
else:
    raise AssertionError("runtime admitted missing numerical evidence")
assert not manual.is_valid(missing)
assert generated.is_valid(missing), "documented differential unexpectedly disappeared"
reverse = json.loads(json.dumps(full_shift))
reverse["current_window"]["start"], reverse["current_window"]["end"] = reverse["current_window"]["end"], reverse["current_window"]["start"]
try:
    canonical.ShiftDetectedEvent.model_validate(reverse)
except ValidationError:
    pass
else:
    raise AssertionError("runtime admitted reversed window")
assert manual.is_valid(reverse)
assert generated.is_valid(reverse)
