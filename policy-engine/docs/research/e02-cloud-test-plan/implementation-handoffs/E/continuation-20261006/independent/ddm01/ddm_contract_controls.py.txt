def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--repository", type=Path, required=True)
    parser.add_argument("--output", type=Path, required=True)
    args = parser.parse_args()
    project = args.repository / "policy-engine"
    args.output.mkdir(parents=True, exist_ok=True)
    ddm_paths = sorted(
        subprocess.check_output(
            ["git", "ls-files", "policy-engine/src/polisyos/ddm", "policy-engine/tests/unit/ddm"],
            cwd=args.repository, text=True,
        ).splitlines()
    )
    def identities() -> dict[str, str]:
        return {path: hashlib.sha256((args.repository / path).read_bytes()).hexdigest() for path in ddm_paths}
    before = identities()
    head = subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=args.repository, text=True).strip()
    tree = subprocess.check_output(["git", "rev-parse", "HEAD^{tree}"], cwd=args.repository, text=True).strip()
    records = []
    for name, code, parameters, expected in (
        ("contracts-only", BOUNDARY, ["positive"], 0),
        ("root-lazy-removal", BOUNDARY, ["root-eager"], 1),
        ("integration-lazy-removal", BOUNDARY, ["integration-eager"], 1),
        ("identity-json-pickle-schema-consumer", COMPAT, [], 0),
    ):
        script = args.output / (name + ".py")
        script.write_text(code)
        command = [sys.executable, str(script), *parameters]
        started = time.monotonic()
        run = subprocess.run(command, cwd=project, capture_output=True, text=True)
        wall = time.monotonic() - started
        (args.output / (name + ".stdout")).write_text(run.stdout)
        (args.output / (name + ".stderr")).write_text(run.stderr)
        records.append({"name": name, "command": command, "cwd": str(project), "exit_code": run.returncode, "expected_exit_code": expected, "wall_s": wall, "outcome": "PASS" if run.returncode == expected else "FAIL", "stdout": run.stdout, "stderr": run.stderr})
    command = [sys.executable, "-m", "pytest", "-q", "tests/unit/ddm/test_facade.py", "tests/unit/ddm/mirror_contracts/test_events.py", "tests/unit/ddm/test_stationary_replay.py", "tests/unit/ddm/test_full_acceptance.py", "tests/unit/ddm/test_readiness_mapping.py"]
    started = time.monotonic()
    run = subprocess.run(command, cwd=project, capture_output=True, text=True)
    (args.output / "native-ddm.stdout").write_text(run.stdout)
    (args.output / "native-ddm.stderr").write_text(run.stderr)
    records.append({"name": "native-ddm-public-consumers", "command": command, "cwd": str(project), "exit_code": run.returncode, "wall_s": time.monotonic()-started, "outcome": "PASS" if run.returncode == 0 else "FAIL", "stdout": run.stdout, "stderr": run.stderr})
    assert before == identities(), "DDM input bytes changed during independent controls"
    receipt = {"finding_ids": ["LA-056"], "bundle_ids": ["DDM-01"], "target_sha": head, "target_tree": tree, "input_path_denominator": len(ddm_paths), "input_sha256": before, "environment": {"python": sys.version, "executable": sys.executable, "platform": platform.platform(), "thread_controls": {key: os.environ.get(key) for key in ("OMP_NUM_THREADS","OPENBLAS_NUM_THREADS","MKL_NUM_THREADS","NUMEXPR_NUM_THREADS","XLA_FLAGS","JAX_PLATFORMS")}}, "checks": records, "predicate_basis": "recomputed", "fixture_boundary": "Native import/class/wire/monitor component property. No production observation or deployment authority asserted.", "pending": "Final combined frozen-SHA wave and accountable owner closure."}
    (args.output / "ddm01-independent.json").write_text(json.dumps(receipt, indent=2) + "\n")
    print(json.dumps({"target_sha": head, "checks": [{"name": item["name"], "outcome": item["outcome"], "exit_code": item["exit_code"]} for item in records]}, indent=2))
    if any(item["outcome"] != "PASS" for item in records):
        raise SystemExit(1)


if __name__ == "__main__":
    main()
