def record(name, action):
    start = time.perf_counter()
    try:
        result = action()
        row = {'case': name, 'outcome': 'PASS', 'result': result}
    except Exception as exc:
        row = {'case': name, 'outcome': 'FAIL', 'exception': type(exc).__name__, 'detail': str(exc)}
    row['wall_seconds'] = time.perf_counter() - start
    CASES.append(row)
    print(json.dumps(row, sort_keys=True), flush=True)

def uniform(low=0., high=1.):
    return UncertaintyEnvelope(point_estimate=(low+high)/2, confidence_interval=(low, high),
        confidence_level=None, distribution_family=DistributionFamily.UNIFORM,
        source=UncertaintySource.ENSEMBLE, propagation_method=PropagationMethod.NONE,
        interval_semantics=IntervalSemantics.DETERMINISTIC_BOUNDS, gate_eligible=True)

def gaussian(std, row=None, order=None):
    metadata = {'std':std}
    if row is not None: metadata.update(covariance_row=row, covariance_params=order)
    return UncertaintyEnvelope(point_estimate=0., confidence_interval=(-1.96*std,1.96*std),
        confidence_level=.95, distribution_family=DistributionFamily.NORMAL,
        source=UncertaintySource.CALIBRATION, propagation_method=PropagationMethod.NONE,
        interval_semantics=IntervalSemantics.CONFIDENCE_INTERVAL, gate_eligible=True,
        metadata=metadata)

def rejected(call):
    try: call()
    except (ValueError, TypeError): return True
    raise AssertionError('present-but-fake claim accepted')

def range_control(method):
    calls=[]
    env=gaussian(1e20,[1e40],['x'])
    for producer_type in (MonteCarloPropagator, PropagationDispatcher, DeltaMethodPropagator):
        with (
            patch.object(MonteCarloPropagator,'_random_input_samples',side_effect=AssertionError('random callback ran')),
            patch.object(MonteCarloPropagator,'_create_qmc_sampler_state_with_seed',side_effect=AssertionError('QMC callback ran')),
        ):
            result=producer_type(PropagationConfig(mc_sampling_method=method,compute_sensitivity=False)).propagate(
                lambda **p:calls.append(p) or {'y':p['x']},{'x':0.},{'x':env},['y'])[0]
        assert not result.envelope.gate_eligible
        assert result.envelope.distribution_family is DistributionFamily.UNKNOWN
    assert not calls
    return {'method':method,'producer_count':3,'sampler_and_evaluator_callbacks':0}

def actual_backend_underflow_boundary():
    from polisyos.foundry.uncertainty.sampling_admission import admit_float32_range
    rejected(lambda:admit_float32_range([[1e-40]]))
    rejected(lambda:admit_float32_range([[1e-50]]))
    tiny=float(np.finfo(np.float32).tiny)
    admitted=admit_float32_range([[tiny]])
    import jax.numpy as jnp
    one=jnp.asarray([1.],dtype=jnp.float32)
    actual=float(one@jnp.asarray(admitted,dtype=jnp.float32)@one)
    assert actual==tiny
    np_cast=float(np.asarray([[1e-40]],dtype=np.float32)[0,0])
    flushed=float(one@jnp.asarray([[1e-40]],dtype=jnp.float32)@one)
    assert np_cast>0 and flushed==0
    zero=MonteCarloPropagator(PropagationConfig(mc_n_samples=128,compute_sensitivity=False)).propagate(
        lambda **p:{'y':p['x']},{'x':0.},{'x':gaussian(0.,[0.],['x'])},['y'])[0]
    assert set(zero.envelope.distribution_payload.samples)=={0.}
    return {'finite_positive_input_boundary':tiny,'subnormal_cast_nonzero':np_cast,
        'subnormal_JAX_quadratic_form':flushed,'normal_boundary_JAX_quadratic_form':actual,
        'true_zero_covariance_samples':[0.]}

def axis_and_backend_law():
    order=['c','a','b']; names=['a','b','c']; expected=np.array([[1,.2,.3],[.2,4,.4],[.3,.4,9.]])
    envs={n:gaussian(math.sqrt(expected[i,i]),expected[i,[2,0,1]].tolist(),order) for i,n in enumerate(names)}
    matrix=np.asarray(build_covariance_matrix(names,envs,use_full_covariance=True,jitter=1e-9))
    assert np.allclose(matrix,expected,rtol=1e-6,atol=1e-7)
    tied={n:gaussian(.05,[.0025,.0025],['a','b']) for n in ['a','b']}
    observed=[]
    for full in (True,False):
        for method in ('delta','analytical','monte_carlo'):
            r=PropagationDispatcher(PropagationConfig(delta_use_full_covariance=full,preferred_method=method,
                mc_n_samples=128,compute_sensitivity=False)).propagate(
                lambda **p:{'y':p['a']-p['b']},{'a':0.,'b':0.},tied,['y'],weights={'a':1.,'b':-1.})[0]
            assert r.envelope.confidence_interval==(0.,0.), (full,method,r.envelope.confidence_interval)
            observed.append({'full_flag':full,'backend':method,'interval':r.envelope.confidence_interval})
    return {'three_axis_matrix':matrix.tolist(),'tied_law':observed}

def singular_and_independent(method):
    tied={n:gaussian(.05,[.0025,.0025],['a','b']) for n in ['a','b']}
    independent={'a':gaussian(.05,[.0025,0.],['a','b']), 'b':gaussian(.05,[0.,.0025],['a','b'])}
    producer=MonteCarloPropagator(PropagationConfig(mc_n_samples=8192,mc_sampling_method=method,
        compute_sensitivity=False))
    variances=[]
    for envs in (tied,independent):
        r=producer.propagate(lambda **p:{'y':p['a']-p['b']},{'a':0.,'b':0.},envs,['y'])[0]
        variances.append(float(np.var(r.envelope.distribution_payload.samples)))
        assert not r.envelope.gate_eligible
    assert variances[0]<1e-20
    assert abs(variances[1]-.005)<.0003
    return {'method':method,'tied_variance':variances[0],'independent_variance':variances[1],
        'analytic_variance_difference':.005}

def failed_support(method):
    def response(x):
        return {'complete':x,'partial':x} if x>=0 else {'complete':x}
    r=MonteCarloPropagator(PropagationConfig(mc_n_samples=1024,mc_sampling_method=method,
        mc_qmc_replicates=4,mc_min_valid_samples=10,compute_sensitivity=False)).propagate(
            response,{'x':0.},{'x':uniform(-1,1)},['complete','partial'])
    partial=r[1]; full=r[0]; receipt=partial.diagnostics['draw_outcome_provenance']
    assert len(receipt['draw_records'])==1024
    assert len(receipt['failure_records'])>400
    assert reconcile_draw_outcomes(receipt,['complete','partial'])=={'partial'}

    assert abs(partial.envelope.point_estimate-.5)<.03
    assert partial.envelope.metadata['distribution_sample_semantics']=='successful_draws_only_conditional_on_execution'
    assert 'mean_estimator_standard_error' not in partial.envelope.metadata
    if method!='random': assert 'mean_estimator_standard_error' in full.envelope.metadata
    return {'method':method,'conditional_mean':partial.envelope.point_estimate,
        'failed':len(receipt['failure_records']),'complete_mean':full.envelope.point_estimate,
        'partial_certificate_kind':partial.envelope.composition_provenance.certificate_kind.value}

    r=MonteCarloPropagator(PropagationConfig(mc_n_samples=1000,mc_sampling_method='sobol',
        mc_qmc_replicates=4,compute_sensitivity=False)).propagate(lambda **p:{'y':p['x']**2},

    return {'requested_rows':1000,'actual_rows':1024,'replica_means':expected_means,
        'estimator_standard_error':expected_se,'predictive_std':metadata['mc_std']}

    controls=[]

    fake=loaded.model_copy(update={'metadata':{**loaded.metadata,
        'mean_error_interval':[loaded.point_estimate,loaded.point_estimate]}})

    fake={**receipt,'draw_records':receipt['draw_records'][:-1]}

        'verified_main_mean':cert.main_mean,'negative_controls':controls}

def bounded_coverage():
    plan=BoundedIIDMeanPlan(metric_id='y'); rows=[]; reps=256
    for probability in (.001,.01,.5,.99):
        hits=0; counts=[]; pilots=[]; max_error=0.
        for seed in range(reps):
            r=MonteCarloPropagator(PropagationConfig(mc_seed=seed,bounded_iid_mean=plan,
                compute_sensitivity=False)).propagate(BoundedIndicatorResponse('x','y',probability),
                    {'x':.5},{'x':uniform()},['y'])[0]
            c=verify_mean_certificate(r.envelope)
            error=abs(c.main_mean-probability); hits+=error<=plan.absolute_error
            max_error=max(max_error,error); counts.append(c.main_samples); pilots.append(c.pilot_mean)
        lower=float(beta.ppf(.025,hits,reps-hits+1)) if hits else 0.
        upper=float(beta.ppf(.975,hits+1,reps-hits)) if hits<reps else 1.
        rows.append({'input':'independent Uniform[0,1] child streams',
            'response':f'indicator(X < {probability})','population_mean':probability,
            'seeds':[0,reps-1],'replications':reps,'covered':hits,'empirical_coverage':hits/reps,
            'coverage_Clopper_Pearson_95':[lower,upper],'N_range':[min(counts),max(counts)],
            'maximum_error':max_error,'zero_pilot_count':pilots.count(0.)})
        assert lower>.95, rows[-1]
    return {'scope':'admitted canonical Bernoulli responses over mathematical Uniform input law only',
        'not_established':['production law custody','general callback boundedness','Uniform-output mean certificate'],
        'experiments':rows}

def full_recipe_controls():
    plan=BoundedIIDMeanPlan(metric_id='y')
    r=MonteCarloPropagator(PropagationConfig(bounded_iid_mean=plan,compute_sensitivity=False)).propagate(
        BoundedIndicatorResponse('x','y',.001),{'x':.5},{'x':uniform()},['y'])[0]
    assert verify_mean_certificate(r.envelope).main_samples==408
    raw=r.envelope.metadata['mean_estimator_certificate']; recipe=raw['evaluator_recipe']
    variants=[('response_discriminator',{**recipe,'response':'fake_other_response'}),
        ('extra_key',{**recipe,'unverified_extra':'same-markers'}),
        ('boolean_threshold',{**recipe,'threshold':True})]
    variants.extend((f'missing_{key}',{k:v for k,v in recipe.items() if k!=key}) for key in recipe)
    for name,fake_recipe in variants:
        fake=r.envelope.model_copy(update={'metadata':{**r.envelope.metadata,
            'mean_estimator_certificate':{**raw,'evaluator_recipe':fake_recipe}}})
        rejected(lambda:verify_mean_certificate(fake))
    invalid_responses=[BoundedIndicatorResponse('x','y',True),
        BoundedIndicatorResponse('','y',.5),BoundedIndicatorResponse('x','',.5)]
    with patch.object(BoundedIndicatorResponse,'__call__',side_effect=AssertionError('invalid recipe evaluated')):
        for response in invalid_responses:
            unavailable=MonteCarloPropagator(PropagationConfig(bounded_iid_mean=plan,
                compute_sensitivity=False)).propagate(response,{'x':.5},{'x':uniform()},['y'])[0]
            assert unavailable.envelope.distribution_family is DistributionFamily.UNKNOWN
            assert 'mean_estimator_certificate' not in unavailable.envelope.metadata
    return {'main_samples':408,'recipe_keys':sorted(recipe),'refused_variants':[n for n,_ in variants]}

def direct_covariance_full_quantity():
    from polisyos.foundry.uncertainty.analytical import AnalyticalPropagator
    import jax.numpy as jnp
    direct=AnalyticalPropagator.propagate_linear_combination
    e={'x':gaussian(1.)}
    control_names=[]
    for name,cov in [('negative',[[-1.]]),('wrong_marginal',[[100.]]),
                     ('null_override',[[0.]]),('wrong_dimension',np.eye(2))]:
        rejected(lambda cov=cov:direct(weights={'x':1.},input_envelopes=e,
            output_metric_id='y',covariance=jnp.asarray(cov)))
        control_names.append(name)
    rejected(lambda:direct(weights={'x':1.},input_envelopes={'x':uniform()},
        output_metric_id='y',covariance=jnp.asarray([[1.]])))
    rejected(lambda:direct(weights={},input_envelopes={},output_metric_id='y',
        covariance=jnp.asarray([[]])))
    for scale in (1.,1e-6,1e-12):
        envelopes={n:gaussian(math.sqrt(scale)) for n in ['a','b']}
        tied=direct(weights={'a':1.,'b':-1.},input_envelopes=envelopes,
            output_metric_id='y',covariance=jnp.asarray([[scale,scale],[scale,scale]]))
        assert tied.diagnostics['output_variance']==0.
        for name,values in [('indefinite',[[scale,2*scale],[2*scale,scale]]),
                            ('asymmetric',[[scale,0.],[.1*scale,scale]])]:
            rejected(lambda values=values:direct(weights={'a':1.,'b':-1.},
                input_envelopes=envelopes,output_metric_id='y',covariance=jnp.asarray(values)))
            control_names.append(f'{name}_scale_{scale}')
    return {'refused_controls':control_names+['non_Gaussian','empty_inputs'],
        'true_singular_zero_variance_scales':[1.,1e-6,1e-12],
        'axis_contract':'direct supplied arrays follow sorted parameter-name row/column order',
        'authority':'candidate/non-gating declared Gaussian law only'}
