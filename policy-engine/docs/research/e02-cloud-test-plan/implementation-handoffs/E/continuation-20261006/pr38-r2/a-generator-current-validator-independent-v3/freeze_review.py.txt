def validate(index):
 if index['file_count']!=len(index['files']) or index['total_bytes']!=sum(r['bytes'] for r in index['files']): raise ValueError('denominator mismatch')
 if len({r['path'] for r in index['files']})!=len(index['files']): raise ValueError('duplicate path')
 for row in index['files']:
