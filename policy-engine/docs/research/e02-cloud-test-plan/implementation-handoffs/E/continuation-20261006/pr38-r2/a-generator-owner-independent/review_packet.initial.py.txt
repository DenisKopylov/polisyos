def git(*args):return subprocess.check_output(['git','-C',str(REPO),*args])
def dig(data):return {'bytes':len(data),'sha256':hashlib.sha256(data).hexdigest()}
def readref(ref):
    data=git('show',ref['commit']+':'+ref['path'])
    assert git('rev-parse',ref['commit']+':'+ref['path']).decode().strip()==ref['git_blob']
    assert dig(data)=={'bytes':ref['bytes'],'sha256':ref['sha256']}
    return data
def counts(cases):
    result=dict(cases=0,passed=0,failed=0,errors=0,skipped=0)
    for c in cases:
        result['cases']+=1
        k='failed' if c.find('failure') is not None else 'errors' if c.find('error') is not None else 'skipped' if c.find('skipped') is not None else 'passed'
        result[k]+=1
    return result
