"""Author property-removal delta, adapted with attribution from independent DoE freeze-role probe; zero native/gate commands."""
from pathlib import Path
from argparse import Namespace
import hashlib
import json
import runpy
import shutil

ROOT = Path('/workspace/e02-E-continuation-20261006')
OUT = Path(__file__).parent
SOURCE = Path('/workspace/e02-E-pr38-r2-receipts/collector-v4-author-controls/full_source_role_guard_removed.py.txt')
OLD = Path('/workspace/e02-E-pr38-r2-receipts/common-wave-publication-prep/synthetic-controls-v2/positive/wave')
module = runpy.run_path(str(SOURCE), run_name='independent_synthetic_freeze_role_probe')
assert hashlib.sha256(SOURCE.read_bytes()).hexdigest() == 'a6399cc46cbfe25b103fa210259eff2be9483f582ae7cba773632f079f7194ec'

def rewrite(value, wave):
    if isinstance(value, str):
        return value.replace(str(OLD), str(wave))
    if isinstance(value, list):
        return [rewrite(v, wave) for v in value]
    if isinstance(value, dict):
        return {k: rewrite(v, wave) for k, v in value.items()}
    return value

rows = []
for role in ['raw', 'private']:
    wave = OUT/'synthetic-role-removal'/('freeze-'+role)/'wave'
    shutil.copytree(OLD, wave)
    for path in wave.rglob('*.json'):
        path.write_text(json.dumps(rewrite(json.loads(path.read_text()), wave), indent=2)+'\n')
    plan = json.loads((wave/'plan.json').read_text())
    freeze = wave/role/'source-freeze.json' if role != 'standalone' else wave.parent/'source-freeze.json'
    freeze.parent.mkdir(parents=True,exist_ok=True)
    data = dict(frozen_sha=plan['candidate_sha'], tree=plan['candidate_tree_sha'], one_wave_output_root=str(wave), remote_ls_readback_sha=plan['candidate_sha'], source_tree_dirty=False, reviews=[], synthetic_only_marker='private/raw role marker is synthetic and carries no actual config')
    freeze.write_text(json.dumps(data, indent=2)+'\n')
    publication = wave.parent/'publication'
    try:
        observed = module['collect'](Namespace(repo=ROOT, candidate=plan['candidate_sha'], wave_root=wave, publication_root=publication, freeze_receipt=freeze, max_moderate_bytes=8*1024*1024, capture_incomplete=False))
        state=observed['collection_state'];reason=None
    except module['EvidenceAdmissionError'] as error:
        state='TYPED_REFUSAL';reason=str(error)
    copied = publication/'source-freeze-receipt.json'
    payload_copied=copied.exists() and copied.read_bytes()==freeze.read_bytes()
    assert state=='CONTROL_COMPLETE_BOUND' and payload_copied
    rows.append({'role': role, 'source_path': str(freeze), 'state':state,'reason':reason,'synthetic_payload_copied':payload_copied, 'product_native_or_gate_commands_executed': 0})
result = {'source_sha256': hashlib.sha256(SOURCE.read_bytes()).hexdigest(), 'scope': 'Synthetic role-only probe; no actual private input touched', 'controls': rows, 'property': 'Removing only full-source-component predicate restores both synthetic raw/private escape and public copy; all function/field names retained. No actual private input copied.'}
(OUT/'role-removal-probe.json').write_text(json.dumps(result, indent=2)+'\n')
print(json.dumps(result, indent=2))
