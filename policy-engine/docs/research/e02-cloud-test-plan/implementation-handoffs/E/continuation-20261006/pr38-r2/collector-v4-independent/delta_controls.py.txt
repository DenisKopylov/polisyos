def validate_index(path):
    data = json.loads(path.read_text())
    assert len(data['files']) == data['file_count']
    assert sum(row['bytes'] for row in data['files']) == data['total_bytes']
    for row in data['files']:
        assert identity(Path(row['path'])) == dict(bytes=row['bytes'], sha256=row['sha256'])
    return dict(path=str(path), **identity(path), assets=data['file_count'], asset_bytes=data['total_bytes'])

def rewrite(value, wave):
    if isinstance(value, str):
        return value.replace(str(OLD_FIXTURE), str(wave))
    if isinstance(value, list):
        return [rewrite(v, wave) for v in value]
    if isinstance(value, dict):
        return {k:rewrite(v,wave) for k,v in value.items()}
    return value

def freeze_fixture(name, role, source_module=module):
    wave = OUT/'synthetic-controls'/name/'wave'
    shutil.copytree(OLD_FIXTURE, wave)
    for path in wave.rglob('*.json'):
        path.write_text(json.dumps(rewrite(json.loads(path.read_text()), wave), indent=2)+'\n')
    plan = json.loads((wave/'plan.json').read_text())
    assert plan['collector_control_only'] is True
    freeze = wave/role/'source-freeze.json' if role in {'raw','private'} else wave.parent/'standalone-source-freeze.json'
    freeze.parent.mkdir(exist_ok=True)
    data = dict(frozen_sha=plan['candidate_sha'], tree=plan['candidate_tree_sha'], one_wave_output_root=str(wave), remote_ls_readback_sha=plan['candidate_sha'], source_tree_dirty=False, reviews=[], synthetic_only_marker='Harmless metadata marker, zero actual private input.')
    freeze.write_text(json.dumps(data, indent=2)+'\n')
    pub = wave.parent/'publication'
    try:
        observed = source_module['collect'](Namespace(repo=ROOT, candidate=plan['candidate_sha'], wave_root=wave, publication_root=pub, freeze_receipt=freeze, max_moderate_bytes=8*1024*1024, capture_incomplete=False))
        state = observed['collection_state']
        reason = None
    except source_module['EvidenceAdmissionError'] as error:
        state = 'TYPED_REFUSAL'
        reason = str(error)
    copied = pub/'source-freeze-receipt.json'
    copied_equal = copied.exists() and copied.read_bytes() == freeze.read_bytes()
    return {'name':name, 'role':role, 'state':state, 'reason':reason, 'publication_created':pub.exists(), 'synthetic_payload_copied':copied_equal, 'product_native_or_gate_commands_executed':0}
