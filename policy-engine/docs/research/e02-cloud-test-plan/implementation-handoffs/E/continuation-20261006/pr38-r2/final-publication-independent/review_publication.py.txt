def git(*args):
    return subprocess.check_output(['git',*args], cwd=ROOT)

def parsed(path, revision=HEAD):
    return json.loads(git('show', f'{revision}:{path}'))

def content(data):
    return {'bytes':len(data),'sha256':hashlib.sha256(data).hexdigest()}

def blob_record(path, revision):
    data = git('show',f'{revision}:{path}')
    return {'path':path,'revision':revision,'git_blob':git('rev-parse',f'{revision}:{path}').decode().strip(),**content(data)}

def delta(start,end,expected):
    records = []
    for line in git('diff','--name-status',start,end).decode().splitlines():
        status,path = line.split('\t')
        assert status == 'A' and path.startswith(PREFIX)
        assert not any(segment.lower() in ['raw','private','git-config-private'] for segment in PurePosixPath(path).parts)
        assert not path.endswith('.raw.json')
        records.append({'status':status,**blob_record(path,end)})
    assert len(records) == expected
    return records

def validate_copy(row, revision):
    path = row['copied_path']
    assert type(row['bytes']) is int and row['bytes'] >= 0
    assert isinstance(path,str) and path.startswith(PREFIX) and '..' not in PurePosixPath(path).parts
    assert not Path(path).is_absolute() and not (ROOT/path).is_symlink()
    assert not any(segment.lower() in ['raw','private','git-config-private'] for segment in PurePosixPath(path).parts)
    assert not path.endswith('.raw.json')
    actual = git('show',f'{revision}:{path}')
    original = Path(row['source_path']).read_bytes()
    working = (ROOT/path).read_bytes()
    assert original == working == actual
    assert content(actual) == {'bytes':row['bytes'],'sha256':row['sha256']}
    return {'source_path':row['source_path'],'copied_path':path,'revision':revision,'git_blob':git('rev-parse',f'{revision}:{path}').decode().strip(),**content(actual)}

def validate_index(path, revision):
    index = parsed(path,revision)
    rows = index['files']
    assert len(rows) == index['file_count']
    assert sum(r['bytes'] for r in rows) == index['total_bytes']
    assert len({r['copied_path'] for r in rows}) == len(rows)
    validated = [validate_copy(row,revision) for row in rows]
    all_copies.extend(validated)
    for row in validated:
        source_targets.setdefault(row['source_path'],[]).append(row)
    result = blob_record(path,revision) | {'records':len(rows),'record_bytes':index['total_bytes']}
    index_results.append(result)
    return result
