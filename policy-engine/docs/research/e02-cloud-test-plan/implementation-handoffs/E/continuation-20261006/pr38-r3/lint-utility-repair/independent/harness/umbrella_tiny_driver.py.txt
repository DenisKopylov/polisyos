import argparse, hashlib, json, os, subprocess, sys, types
from pathlib import Path
S = Path('/workspace/e02-E-pr38-r3-receipts/harness-independent-r3')
R = Path('/workspace/e02-E-cal05-20261006')
PY = Path('/workspace/e02-E-continuation-20261006/policy-engine/.venv/bin/python')
SHA = 'f3ec0335e992fa7a5107000224f8ba6deefd02e5'
p = argparse.ArgumentParser(); p.add_argument('variant'); args = p.parse_args()
root = S / ('umbrella-' + args.variant)
root.mkdir()
sys.path.insert(0, str(R / 'policy-engine'))
from tools.devx.workspace import ci_parity, verify
from tools.devx.workspace._common import CommandSpec
for module in (ci_parity, verify, sys.modules['tools.devx.workspace._common']):
    path = Path(module.__file__)
    expected = subprocess.check_output(['git', 'show', SHA + ':' + path.relative_to(R).as_posix()], cwd=R)
    assert path.read_bytes() == expected

# This bounded synthetic factory preserves the actual CommandSpec/run_command/recorder seam.
# It does not claim the full repository verify or CI stage denominator.
def factory(**kwargs):
    first = CommandSpec(
        label='tiny actual uncapped pytest child',
        argv=(str(PY), '-m', 'pytest', '-q', '--confcutdir', str(S), str(S / 'tiny_uncapped_environment.py')),
        cwd=R / 'policy-engine',
        env={**verify.PYTEST_NUMERICAL_ENV, 'HARNESS_ENV_OBSERVER': str(root / 'observed-child-environment.json')},
    )
    code = 0 if args.variant == 'success' else 7
    second = CommandSpec(label='tiny deliberate exit ' + str(code), argv=(str(PY), '-c', 'raise SystemExit(' + str(code) + ')'), cwd=R / 'policy-engine')
    third = CommandSpec(label='tiny successor actual marker', argv=(str(PY), '-c', 'from pathlib import Path;Path(' + repr(str(root / 'successor-ran.txt')) + ').write_text("successor actually executed\\n")'), cwd=R / 'policy-engine')
    return [first, second, third]
verify._backend_commands = factory

def ci_factory(**kwargs):
    return [CommandSpec(label='tiny nested canonical verify seam', argv=(str(PY), '-m', 'tools.devx.workspace.verify', '--skip-doctor', '--backend-only'), cwd=R / 'policy-engine'), CommandSpec(label='tiny CI successor marker', argv=(str(PY), '-c', 'from pathlib import Path;Path(' + repr(str(root / 'ci-successor-ran.txt')) + ').write_text("CI successor actually executed\\n")'), cwd=R / 'policy-engine')]
ci_parity._backend_commands = ci_factory
source = (S / 'sources/uncapped_umbrella.py').read_text()

    old = '    verify.PYTEST_NUMERICAL_ENV = {}\n'
    assert source.count(old) == 1
    source = source.replace(old, '    # Independent control removes only the operational uncapping assignment.\n')
elif args.variant == 'remove-failfast-property':
    old = '            for command in commands:\n                recorded_run(command)\n'
    new = '            for command in commands:\n                try:\n                    recorded_run(command)\n                except subprocess.CalledProcessError:\n                    pass\n'
    assert source.count(old) == 1
    source = source.replace(old, new)
(root / 'executed-umbrella-source.py.txt').write_text(source)
original = R / 'policy-engine/docs/research/e02-cloud-test-plan/implementation-handoffs/E/continuation-20261006/pr38-r2/wave-controls/uncapped_umbrella.py'

gate = 'ci-parity' if args.variant == 'nested-failfast' else 'verify'

code = m.main()
(root / 'driver-receipt.json').write_text(json.dumps({'candidate_sha': SHA, 'variant': args.variant, 'native_exit_code': code, 'synthetic_factory_control': True, 'full_canonical_stage_wave_executed': False, 'factory_driver_source_sha256': hashlib.sha256(Path(__file__).read_bytes()).hexdigest(), 'executed_umbrella_sha256': hashlib.sha256(source.encode()).hexdigest()}, indent=2) + '\n')
