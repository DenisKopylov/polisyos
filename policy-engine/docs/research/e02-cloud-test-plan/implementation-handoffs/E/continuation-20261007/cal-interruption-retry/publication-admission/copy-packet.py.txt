def copy(source, destination, role, expected=None):
    data = source.read_bytes()
    actual = {'sha256':hashlib.sha256(data).hexdigest(),'bytes':len(data)}
    if expected is not None:
        assert all(actual[key] == expected[key] for key in ('sha256','bytes'))
    path = Path(destination)
    assert not path.is_absolute() and '..' not in path.parts and path.as_posix() == destination
    target = packet/path
    target.parent.mkdir(parents=True,exist_ok=True)
    with target.open('xb') as handle:
        handle.write(data)
    assert target.read_bytes() == data
    rows.append({'source_path':str(source),'portable_path':destination,'role':role,**actual})
