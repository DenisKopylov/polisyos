import runpy,sys,tempfile
original=tempfile.TemporaryDirectory
def only_retained(*a,**k):
    if k.get("delete",True) is not False:
        raise RuntimeError("cleanup-enabled temporary request refused before allocation")
    return original(*a,**k)
tempfile.TemporaryDirectory=only_retained
script=sys.argv.pop(1)
runpy.run_path(script,run_name="__main__")
