def source_guard():
    sha = subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=ROOT, text=True).strip()
    tree = subprocess.check_output(['git', 'rev-parse', 'HEAD^{tree}'], cwd=ROOT, text=True).strip()
    assert (sha, tree) == (expected_source, expected_tree)
    rows = []
    for relative in paths:
        body = (ROOT / relative).read_bytes()
        original = subprocess.check_output(['git', 'show', f'{expected_source}:{relative}'], cwd=ROOT)
        assert body == original
        rows.append({'path': relative, 'bytes': len(body), 'sha256': hashlib.sha256(body).hexdigest()})
    return {'sha': sha, 'tree': tree, 'assets': rows,
            'status': subprocess.check_output(['git', 'status', '--porcelain=v1', '-uall'], cwd=ROOT, text=True)}

def raises_value_error(fn):
    try:
        fn()
    except ValueError as error:
        return str(error)
    raise AssertionError('required ValueError was absent')

# Independent algebra: finite binary64 weights are exact rational ratios.
tiny = Fraction.from_float(1e-20)
half = Fraction(1, 2)
next_half = float(np.nextafter(0.5, np.inf))
pre_half = float(np.nextafter(0.5, -np.inf))
three_ratios = (half/(1+tiny), (half+tiny)/(1+tiny), Fraction(1))
assert Fraction.from_float(pre_half) < three_ratios[0] < half
assert half < three_ratios[1] < Fraction.from_float(next_half)
four_ratios = (half/(1+2*tiny), (half+tiny)/(1+2*tiny), (half+2*tiny)/(1+2*tiny), Fraction(1))
assert Fraction.from_float(pre_half) < four_ratios[0] < half
assert four_ratios[1] == half
assert half < four_ratios[2] < Fraction.from_float(next_half)
gold3 = np.asarray([0.5, next_half, 1.0])
gold4 = np.asarray([0.5, 0.5, next_half, 1.0])
add('independent_fraction_gold', check_status='PASS', tiny_binary64_ratio=str(tiny),
    three_exact_cdf=[str(x) for x in three_ratios], three_expected_cuts=[x.hex() for x in gold3],
    four_exact_cdf=[str(x) for x in four_ratios], four_unadmitted_cuts=[x.hex() for x in gold4],
    basis='independently_reconciled', semantics='finite-U mesh approximation; no continuum/arbitrary-real exactness promise')

config = PropagationConfig(mc_n_samples=256, mc_sampling_method='sobol', mc_qmc_scramble=False,
                           compute_sensitivity=False)

def native(weights):
    samples = tuple(float(i) for i in range(len(weights)))
    envelope = UncertaintyEnvelope(
        point_estimate=1.0, confidence_interval=(0.0, float(len(weights)-1)),
        confidence_level=None, distribution_family=DistributionFamily.BOOTSTRAP,
        source=UncertaintySource.ENSEMBLE, propagation_method=PropagationMethod.NONE,
        interval_semantics=IntervalSemantics.DETERMINISTIC_BOUNDS, gate_eligible=False,
        numeric_policy=NumericPolicySpec(mode=NumericToleranceMode.DECIMAL_EXACT),
        distribution_payload=PosteriorSamplesCarrier(samples=samples, weights=tuple(weights)),
    )
    calls = []
    def evaluator(**inputs):
        calls.append(float(inputs['x']))
        return {'y': inputs['x']}
    result = MonteCarloPropagator(config).propagate(evaluator, {'x': 1.0}, {'x': envelope}, ['y'])[0]
    return result, calls

positive, calls = native([0.5, 1e-20, 0.5])
assert len(calls) == 257
assert Counter(calls[1:]) == {0.0: 128, 1.0: 1, 2.0: 127}
