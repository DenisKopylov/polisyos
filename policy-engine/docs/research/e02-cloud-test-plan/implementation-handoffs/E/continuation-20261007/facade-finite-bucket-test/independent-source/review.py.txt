def typed_equal(left, right):
    if type(left) is not type(right):
        return False
    if isinstance(left, dict):
        return left.keys() == right.keys() and all(typed_equal(left[k], right[k]) for k in left)
    if isinstance(left, list):
        return len(left)==len(right) and all(typed_equal(a,b) for a,b in zip(left,right,strict=True))
    return left == right

def admit_source_refs(claim):
    if not typed_equal(claim, expected):
        raise ValueError('source/ref claim differs from immutable recomputation')
    return True
