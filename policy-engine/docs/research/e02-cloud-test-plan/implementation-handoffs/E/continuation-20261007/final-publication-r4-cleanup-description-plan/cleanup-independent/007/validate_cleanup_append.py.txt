def require(value, message):
    if not value:
        raise ValueError(message)


def digest(path):
    h = hashlib.sha256()
    with Path(path).open('rb') as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b''):
            h.update(chunk)
    return h.hexdigest()


def ref(path):
    return {'path': str(path), 'bytes': Path(path).stat().st_size, 'sha256': digest(path)}


def load(path):
    return json.loads(Path(path).read_bytes())


def bind(row):
    path = Path(row.get('path', row.get('source_path')))
    require(type(row['bytes']) is int and path.stat().st_size == row['bytes'], 'bound byte size:' + str(path))
    require(digest(path) == row['sha256'], 'bound digest:' + str(path))
    return path


def git(*args):

def typed_equal(left, right):
    if type(left) is not type(right):
        return False
    if isinstance(left, dict):
        return left.keys() == right.keys() and all(typed_equal(left[k], right[k]) for k in left)
    if isinstance(left, list):
        return len(left) == len(right) and all(typed_equal(a, b) for a, b in zip(left, right))
    return left == right

        bind(r)
    return {'members': len(actual), 'bytes': sum(r['bytes'] for r in actual)}
