def require(value, message):
    if not value:
        raise ValueError(message)


def record(path):
    raw = path.read_bytes()
    return {'path': str(path), 'bytes': len(raw), 'sha256': hashlib.sha256(raw).hexdigest()}


def load_module(name, raw):
    module = types.ModuleType(name)
    module.__file__ = str(V2)
    sys.modules[name] = module
    exec(compile(raw, str(V2), 'exec'), module.__dict__)
    return module


def probe():
    original = V2.read_bytes()
    require(hashlib.sha256(original).hexdigest() == EXPECTED_V2, 'source_hash_changed')
    predicate = b'    summary_receipt = admit_summary_claims(base, packet)\n'
    require(original.count(predicate) == 1, 'unique_removal_predicate_missing')
    mutant = original.replace(predicate, b'    summary_receipt = {}\n')
    # Only the unique RHS inside validate is removed. All other AST nodes survive.
    before = ast.parse(original)
    after = ast.parse(mutant)
    transformed = copy.deepcopy(before)
    funcs = [n for n in transformed.body if isinstance(n, ast.FunctionDef) and n.name == 'validate']
    require(len(funcs) == 1, 'validate_denominator_changed')
    assigns = [n for n in funcs[0].body if isinstance(n, ast.Assign) and any(isinstance(t, ast.Name) and t.id == 'summary_receipt' for t in n.targets)]
    require(len(assigns) == 1, 'summary_assignment_denominator_changed')
    assigns[0].value = ast.Dict(keys=[], values=[])
    require(ast.dump(transformed, include_attributes=False) == ast.dump(after, include_attributes=False), 'mutant_changed_more_than_summary_inlet')
    current = load_module('independent_v2_current', original)
    removed = load_module('independent_v2_summary_removed', mutant)
    packet = json.loads(PACKET.read_bytes())
    healthy = current.validate(BASE, packet)
    require(healthy['claim_count'] == 65 and len(healthy['structured_claims']) == 65, 'healthy_claim_denominator_changed')
    require(removed.validate(BASE, packet) == {}, 'removal_baseline_not_healthy')
    mutations = {
        'full_original_HTTP_counts_1585_to_one_PASS': lambda p: p['native_full_HTTP']['full_original_counts'].update(cases=1, passed=1, failed=0, errors=0, skipped=0),
        'main_catalog_observations_zero_to_hundred': lambda p: p['minimum_fixture_inputs']['actual_catalog_counts'].update(ds_observations=100),
        'extra_product_patch_path': lambda p: p['source_footprint']['unapplied_proposal_paths'].append('policy-engine/src/polisyos/runtime/http/services/control/run_lifecycle.py'),
    }
    controls = []
    for name, mutate in mutations.items():
        corrupt = copy.deepcopy(packet)
        mutate(corrupt)
        markers = (corrupt['finding_closure'] is False and corrupt['browser_cases_executed'] is False and corrupt['overlay_is_committed_implementation'] is False)
        refs = corrupt['receipts'] == packet['receipts'] and all(corrupt['native_full_HTTP'][key] == packet['native_full_HTTP'][key] for key in current.HTTP_PINS)
        require(markers and refs, 'negative_control_changed_markers_or_closed_refs')
        try:
            current.validate(BASE, corrupt)
        except ValueError as error:
            reason = str(error)
        else:
            raise ValueError('current_validator_admits_' + name)
        require(reason.startswith('summary_mismatch:'), 'not_defined_summary_property_refusal')
        removed_receipt = removed.validate(BASE, corrupt)
        require(removed_receipt == {}, 'removal_did_not_readmit_same_summary_escape')
        controls.append({'name': name, 'current_expected_refusal_outcome': 'PASS', 'current_reason': reason, 'removed_expected_refusal_outcome': 'FAIL', 'removed_validator_admitted': True, 'effective_removal_discriminator': 'PASS', 'closed_native_HTTP_refs_unchanged': refs, 'authority_denial_markers_preserved': markers})
    mutant_path = OUT / 'v2-summary-inlet-removed.py.txt'
    if not mutant_path.exists():
        mutant_path.write_bytes(mutant)
    else:
        require(mutant_path.read_bytes() == mutant, 'prior_mutant_bytes_changed')
    return {'schema': 'policyos.e02.independent_catalog_summary_class_probe.v2', 'outcome': 'PASS', 'source': SOURCE, 'tree': TREE, 'optimized': sys.flags.optimize, 'healthy_claim_count': 65, 'controls': controls, 'effective_removal_unique_predicate': predicate.decode().strip(), 'removal_exact_AST_delta': True, 'original_source': record(V2), 'mutant_source': record(mutant_path), 'no_native_selectors_or_ROOT_interpreter': True}


def source_guard(packet):
    rows = []
    for ref in packet['source_guard_inputs']:
        raw = Path(ref['path']).read_bytes()
        expected = subprocess.check_output(['git', '-C', str(REPO), 'show', SOURCE + ':' + ref['repo_relative']]) if 'repo_relative' in ref else None
        require(len(raw) == ref['bytes'] and hashlib.sha256(raw).hexdigest() == ref['sha256'], 'physical_source_input_changed')
        if expected is not None:
            require(raw == expected, 'physical_input_vs_Git_changed')
        rows.append(record(Path(ref['path'])))
    return rows


def main():
    if '--probe' in sys.argv:
        print(json.dumps(probe(), indent=2))
        return
    packet = json.loads(PACKET.read_bytes())
    fixed = [BASE / 'publication-selection.json', PACKET, BASE / 'validate_closed_catalog_packet.py', V2, BASE / 'evidence-summary-correction-v2.json', BASE / 'publication-selection-v2-pending-review.json']
    before = [record(path) for path in fixed]
    old = json.loads((BASE / 'publication-selection.json').read_bytes())
    pending = json.loads((BASE / 'publication-selection-v2-pending-review.json').read_bytes())
    # Preserve all originally selected bytes and exact aliases. No scans of raw CAS/DB.
    verified_assets = []
    for selection in (old, pending):
        for asset in selection['assets']:
            actual = record(Path(asset['source_path']))
            require(actual['bytes'] == asset['bytes'] and actual['sha256'] == asset['sha256'], 'selection_asset_changed')
            verified_assets.append(actual)
    commands = []
    for label, flags in [('normal', []), ('optimized', ['-O'])]:
        cmd = [sys.executable, *flags, '-B', str(V2), '--base', str(BASE), '--packet', str(PACKET)]
        result = subprocess.run(cmd, capture_output=True, check=False)
        stdout = OUT / ('v2-independent-' + label + '.stdout.json')
        stderr = OUT / ('v2-independent-' + label + '.stderr.txt')
        stdout.write_bytes(result.stdout)
        stderr.write_bytes(result.stderr)
        require(result.returncode == 0 and not result.stderr, 'v2_canonical_utility_failed')
        parsed = json.loads(result.stdout)
        require(parsed['outcome'] == 'PASS' and parsed['checks'] == 21 and parsed['summary_admission_basis']['claim_count'] == 65, 'v2_outcome_denominator_changed')
        author = BASE / ('packet-validation-v2-' + label + '.json')
        require(result.stdout == author.read_bytes(), 'v2_author_output_not_reproduced')
        commands.append({'argv': cmd, 'exit_code': result.returncode, 'stdout': record(stdout), 'stderr': record(stderr), 'same_author_output_bytes': True, 'checks': 21, 'structured_claims': 65})
        cmd = [sys.executable, *flags, '-B', str(Path(__file__)), '--probe']
        result = subprocess.run(cmd, capture_output=True, check=False)
        stdout = OUT / ('v2-removal-' + label + '.stdout.json')
        stderr = OUT / ('v2-removal-' + label + '.stderr.txt')
        stdout.write_bytes(result.stdout)
        stderr.write_bytes(result.stderr)
        require(result.returncode == 0 and not result.stderr, 'v2_independent_removal_probe_failed')
        parsed = json.loads(result.stdout)
        require(parsed['outcome'] == 'PASS' and len(parsed['controls']) == 3, 'removal_probe_denominator_changed')
        commands.append({'argv': cmd, 'exit_code': result.returncode, 'stdout': record(stdout), 'stderr': record(stderr), 'actual_three_current_refusals': 3, 'removed_required_refusals': 0, 'effective_removal': True})
    after = [record(path) for path in fixed]
    require(before == after, 'immutable_author_packet_changed')
    receipt = {'schema': 'policyos.e02.independent_catalog_summary_class_review.v2', 'outcome': 'GO', 'source': SOURCE, 'tree': TREE, 'review_scope': 'Evidence-only current summary/ref/source/footprint admission and effective removal, reusing unchanged prior native mechanism/source/CAS independent GO.', 'specification': 'PASS bounded admission class', 'engineering_quality': 'PASS bounded pure observer utility', 'environment': {'executable': sys.executable, 'python': sys.version, 'platform': platform.platform(), 'rootenv_or_packages_mutated': False}, 'fixed_author_inputs_before': before, 'fixed_author_inputs_after': after, 'author_inputs_unchanged': True, 'verified_selected_logical_asset_rows': len(verified_assets), 'commands': commands, 'property_path': 'validate -> admit_summary_claims -> typed_equal / read_record / fixed read-only Git diff and numstat', 'predicate_basis': {'native_numeric_and_binding_claims': 'independently_reconciled', 'raw_ref_integrity': 'recomputed', 'actual_patch_footprint': 'recomputed', 'narrative_owner_and_concurrency_limits': 'consumer_asserted', 'production_or_verifier_authority': 'not_established'}, 'P37_basis_frozen': True, 'P40_class_repair': 'Full structured summary class: types/denominators/numeric flags/refs/source/environment/patch footprint, not a single observation-count plug.', 'native_checks_repeated': False, 'finding_closure': False, 'owner_helper_patch_accepted': False, 'remaining_owner': 'Canonical A runtime fixture owner accepts/applies helper patch, honors same-process factory-construction serialization or provides typed injection; A/G then measure full native HTTP/browser/CI successors. DataForge semantic owners retain source/schema authority.', 'limitations': ['15 original source references and exact original native/overlay receipts were independently reviewed in prior native-owner-packet-review.json; no source/application execution repeated.', 'Process-global constructor patch remains a fixture-only owner consideration; prose matching confers no concurrency guarantee.', 'Full native HTTP historical FAIL and metadata ds_observations=0 remain unchanged; CSV3 rows are separate persisted native CAS inputs, without invented catalog join.', 'This source-specific observer reads absolute receipt paths and fixed Git source objects. Published archive reconstruction must retain those refs or provide an explicitly rebound receipt.'], 'prior_native_GO': record(OUT / 'native-owner-packet-review.json')}
    target = OUT / 'v2-summary-class-independent-review.json'
    target.write_text(json.dumps(receipt, indent=2) + '\n')
    print(json.dumps({'outcome': 'GO', 'receipt': record(target), 'commands': len(commands), 'canonical_utility_checks_per_mode': 21, 'class_claims': 65, 'current_three_refusals_per_mode': 3, 'effective_removed_three_readmissions_per_mode': 3, 'author_bytes_unchanged': True, 'no_native_reexecution': True}, indent=2))
