def require(value: object, message: str) -> None:
    if not value:
        raise ValueError(message)


def typed_equal(actual: object, expected: object) -> bool:
    """Preserve JSON count/flag types as well as the full row/field denominator."""
    if type(actual) is not type(expected):
        return False
    if isinstance(actual, dict):
        return actual.keys() == expected.keys() and all(
            typed_equal(actual[key], expected[key]) for key in actual
        )
    if isinstance(actual, list):
        return len(actual) == len(expected) and all(
            typed_equal(left, right) for left, right in zip(actual, expected, strict=True)
        )
    return actual == expected


def json_bytes(raw: bytes) -> object:
    def invalid_constant(value: str) -> None:
        raise ValueError(f"nonfinite_JSON:{value}")
    return json.loads(raw, parse_constant=invalid_constant)


def read_record(base: Path, record: dict[str, object], *, absolute: bool = False) -> bytes:
    require(type(record) is dict and set(record) == {"path", "bytes", "sha256"}, "record_shape_invalid")
    path, size, digest = record["path"], record["bytes"], record["sha256"]
    require(type(path) is str and bool(path) and Path(path).parts and ".." not in Path(path).parts, "record_path_invalid")
    require(Path(path).is_absolute() is absolute, "record_path_role_invalid")
    require(type(size) is int and size >= 0, "record_size_type_invalid")
    require(type(digest) is str and re.fullmatch(r"[0-9a-f]{64}", digest) is not None, "record_digest_invalid")
    raw = (Path(path) if absolute else base / path).read_bytes()
    require(len(raw) == size and hashlib.sha256(raw).hexdigest() == digest, "record_byte_binding_invalid")
    return raw


def old_validate(base: Path, packet: dict[str, object]) -> None:
    path = base / "validate_closed_catalog_packet.py"
    require(hashlib.sha256(path.read_bytes()).hexdigest() == OLD_VALIDATOR_HASH, "historical_validator_changed")
    spec = importlib.util.spec_from_file_location("historical_catalog_packet_validator", path)
    require(spec is not None and spec.loader is not None, "historical_validator_missing")
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    module.validate(base, packet)
