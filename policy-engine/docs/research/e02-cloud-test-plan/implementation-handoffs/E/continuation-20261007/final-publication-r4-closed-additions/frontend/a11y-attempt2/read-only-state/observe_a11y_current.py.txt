def binding(path: Path):
    body = path.read_bytes()
    return {'source_path': str(path), 'bytes': len(body), 'sha256': hashlib.sha256(body).hexdigest()}

def git_source(relative: str):
    body = subprocess.check_output(['/usr/bin/git', 'show', f'{SOURCE}:{relative}'], cwd=REPO)
    if (REPO / relative).read_bytes() != body:
        raise ValueError(f'Current tracked source differs: {relative}')
    return {'source_sha': SOURCE, 'path': relative, 'bytes': len(body), 'sha256': hashlib.sha256(body).hexdigest()}

def proc_record(pid):
    root = Path('/proc') / str(pid)
    raw = (root / 'stat').read_text()
    fields = raw[raw.rfind(')') + 2:].split()
    args = (root / 'cmdline').read_bytes().replace(b'\x00', b' ').decode(errors='replace').strip()
    sockets = []
    for path in (root / 'fd').iterdir():
        try:
            target = path.readlink().as_posix()
            if target.startswith('socket:['):
                sockets.append({'fd': path.name, 'inode': target[8:-1]})
        except (FileNotFoundError, PermissionError):
            continue
    return {'pid': pid, 'ppid': int(fields[1]), 'state': fields[0], 'utime_ticks': int(fields[11]), 'stime_ticks': int(fields[12]), 'start_ticks': int(fields[19]), 'rss_pages': int(fields[21]), 'args': args, 'wchan': (root / 'wchan').read_text().strip(), 'sockets': sockets}

def process_tree():
    all_parents = {}
    for path in Path('/proc').iterdir():
        if path.name.isdecimal():
            try:
                raw = (path / 'stat').read_text()
                all_parents[int(path.name)] = int(raw[raw.rfind(')') + 2:].split()[1])
            except (FileNotFoundError, ProcessLookupError):
                continue
    children = {PID}
    while True:
        updated = children | {pid for pid, parent in all_parents.items() if parent in children}
        if children == updated:
            break
        children = updated
    records = []
    for pid in sorted(children):
        try:
            records.append(proc_record(pid))
        except (FileNotFoundError, ProcessLookupError):
            continue
    return records

def address(raw, ipv6):
    host, port = raw.split(':')
    if ipv6:
        body = bytes.fromhex(host)
        host = socket.inet_ntop(socket.AF_INET6, b''.join(body[i:i + 4][::-1] for i in range(0, 16, 4)))
    else:
        host = socket.inet_ntop(socket.AF_INET, bytes.fromhex(host)[::-1])
    return {'host': host, 'port': int(port, 16)}

def socket_snapshot(records):
    owners = {}
    for row in records:
        for descriptor in row['sockets']:
            owners.setdefault(descriptor['inode'], []).append({'pid': row['pid'], 'fd': descriptor['fd']})
    rows = []
    for name in ['tcp', 'tcp6']:
        for line in (Path('/proc/net') / name).read_text().splitlines()[1:]:
            fields = line.split()
            local = address(fields[1], name == 'tcp6')
            remote = address(fields[2], name == 'tcp6')
            inode = fields[9]
            if inode in owners or local['port'] == PORT or remote['port'] == PORT:
                rows.append({'family': name, 'local': local, 'remote': remote, 'state_hex': fields[3], 'tx_queue_hex': fields[4].split(':')[0], 'rx_queue_hex': fields[4].split(':')[1], 'inode': inode, 'owners': owners.get(inode, [])})
    return rows

def stdout_snapshot(label):

def static_http(path, label):
    # Only GET static content. Never call websocket, test session, runner or RPC endpoint.
    start = time.monotonic()
    conn = http.client.HTTPConnection('::1', PORT, timeout=10)
    result = {'method': 'GET', 'host': '[::1]', 'port': PORT, 'path': path, 'timeout_seconds': 10, 'purpose': 'Native static HTTP responsiveness only, not browser navigation or test execution'}
    try:
        conn.request('GET', path, headers={'Connection': 'close'})
        response = conn.getresponse()
        body = response.read(262145)
        if len(body) > 262144:
            raise ValueError('Static probe body exceeded bounded capture')
        captured = OUT / f'a11y-http-{label}.body.txt'
        captured.write_bytes(body)
        result.update(status=response.status, reason=response.reason, headers=response.getheaders(), complete_body=binding(captured))
    except (OSError, ValueError, http.client.HTTPException) as exc:
        result.update(error_type=type(exc).__name__, error=str(exc), limitation='This single static request failure cannot identify the browser awaited phase')
    finally:
        conn.close()
    result['wall_seconds'] = time.monotonic() - start
    return result

def native_excerpt(pattern, ranges):
    matches = list((ENGINE / 'node_modules/.pnpm').glob(pattern))
    if len(matches) != 1:
        raise ValueError(f'Ambiguous installed source: {pattern}: {matches}')
    path = matches[0]
    lines = path.read_text().splitlines()
    return {'input': binding(path), 'excerpts': [{'first_line': first, 'last_line': last, 'literal_text': '\n'.join(lines[first - 1:last])} for first, last in ranges]}
