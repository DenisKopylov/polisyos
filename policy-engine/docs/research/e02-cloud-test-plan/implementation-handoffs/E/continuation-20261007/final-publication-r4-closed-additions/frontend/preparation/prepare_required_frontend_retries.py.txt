import json

REPO = Path('/workspace/e02-E-continuation-20261006')
SOURCE = '64d7444a18c55df7b88b71b7699a2f1b25ca24bd'

checks = []

def check(name, value):
    checks.append({'name': name, 'passed': bool(value)})
    if not value:
        raise ValueError(name)

def bind(path):
    body = path.read_bytes()
    return {'path': str(path), 'bytes': len(body), 'sha256': hashlib.sha256(body).hexdigest()}

def source(path):
    relative = path.relative_to(REPO).as_posix()
    exact = subprocess.check_output(['/usr/bin/git', 'show', SOURCE + ':' + relative], cwd=REPO)
    if exact != path.read_bytes():
        raise ValueError('Current source mismatch: ' + relative)
    return {**bind(path), 'source_sha': SOURCE, 'repository_path': relative}

original = json.loads(original_path.read_bytes())

check('root_exact_source_and_tree', subprocess.check_output(['/usr/bin/git', 'rev-parse', 'HEAD', 'HEAD^{tree}'], cwd=REPO).decode().splitlines() == [SOURCE, TREE])

a11y_argv = ['npm', 'run', 'test:a11y']

support_source = """import{pathToFileURL}from'node:url';import{readFileSync}from'node:fs';import path from'node:path';
const{p:parseCLI}=await import(pathToFileURL(process.argv[1]).href);const original=JSON.parse(process.argv[2]);const parsed=parseCLI([...original]);
const source=readFileSync(process.argv[3],'utf8');const start=source.indexOf('function resolveScreenshotPath(');const end=source.indexOf('\\nasync function getBrowserProvider(',start);if(start<0||end<0)throw Error('actual native resolver missing');
const literal=source.slice(start,end);const native=new Function('dirname','basename','resolve','relative',literal+';return resolveScreenshotPath;')(path.dirname,path.basename,path.resolve,path.relative);
