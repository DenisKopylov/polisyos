def identity(path: Path) -> dict:
    digest = hashlib.sha256()
    count = 0
    with path.open('rb') as handle:
        while block := handle.read(1024 * 1024):
            digest.update(block)
            count += len(block)
    return dict(path=str(path), bytes=count, sha256=digest.hexdigest())


def require_ref(row: dict, path_key: str = 'path') -> dict:
    actual = identity(Path(row[path_key]))
    if actual['sha256'] != row['sha256'] or actual['bytes'] != row['bytes']:
        raise ValueError('Content identity mismatch: ' + row[path_key])
    return actual


def dump(name: str, value: object) -> dict:
    path = OUT / name
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n')

def fresh_process_scan(prefixes: list[str]) -> dict:
    """Independent agent-UID scope, not a privileged host-wide process assertion."""
    observations, exclusions, unreadable, active = [], [], [], []
    for proc in sorted(Path('/proc').iterdir(), key=lambda p: p.name):
        if not proc.name.isdecimal():
            continue
        try:
            lines = (proc / 'status').read_text().splitlines()
            uid = int(next(s for s in lines if s.startswith('Uid:')).split()[1])
            state = next(s for s in lines if s.startswith('State:')).split()[1]
            if state == 'Z':
                continue
            if uid != os.getuid():
                exclusions.append(dict(pid=int(proc.name), uid=uid, reason='Outside complete live agent UID read scope; no host-wide visibility asserted.'))
                continue
            argv = (proc / 'cmdline').read_bytes().split(b'\0')
            cwd = os.readlink(proc / 'cwd')
            matches = []
            for prefix in prefixes:
                if cwd == prefix or cwd.startswith(prefix + '/'):
                    matches.append(dict(kind='cwd', candidate=prefix))
                if any(prefix.encode() in arg for arg in argv):
                    matches.append(dict(kind='argv', candidate=prefix))
                for item in (proc / 'environ').read_bytes().split(b'\0'):
                    if item.startswith((b'VIRTUAL_ENV=', b'UV_PROJECT_ENVIRONMENT=', b'PYTHONPATH=')) and prefix.encode() in item:
                        matches.append(dict(kind='selected_environment', candidate=prefix))
                if any(prefix + '/' in line for line in (proc / 'maps').read_text().splitlines()):
                    matches.append(dict(kind='mapped_file', candidate=prefix))
                for descriptor in (proc / 'fd').iterdir():
                    try:
                        target = os.readlink(descriptor)
                    except FileNotFoundError:
                        continue
                    if target == prefix or target.startswith(prefix + '/'):
                        matches.append(dict(kind='open_file', candidate=prefix))
            row = dict(pid=int(proc.name), uid=uid, target_matches=matches)
            observations.append(row)
            if matches:
                active.append(row)
        except (FileNotFoundError, ProcessLookupError):
            continue
        except PermissionError as error:
            unreadable.append(dict(pid=int(proc.name), error=str(error)))
    return dict(observed_unix=time.time(), agent_uid=os.getuid(), exact_candidate_prefixes=prefixes, live_agent_process_count=len(observations), observations=observations, active_users=active, unreadable_agent_processes=unreadable, outside_agent_scope=exclusions)

        raise ValueError('Immediate process proof is not pre-action/fresh')
