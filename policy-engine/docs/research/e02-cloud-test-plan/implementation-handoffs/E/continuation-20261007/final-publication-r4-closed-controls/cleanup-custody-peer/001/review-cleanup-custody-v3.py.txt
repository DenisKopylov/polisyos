def require(condition: bool, reason: str) -> None:
    if not condition:
        raise ValueError(reason)


def integer(value: object, reason: str) -> int:
    if type(value) is not int or value < 0:
        raise ValueError(reason)
    return value


def stream_hash(path: Path) -> dict[str, Any]:
    hashed = hashlib.sha256()
    size = 0
    with path.open("rb") as stream:
        for data in iter(lambda: stream.read(BUFFER_BYTES), b""):
            size += len(data)
            hashed.update(data)
    return {"path": str(path), "bytes": size, "sha256": hashed.hexdigest()}


def read_ref(ref: dict[str, Any]) -> bytes:
    integer(ref["bytes"], "exact integer ref size")
    path = Path(ref["path"])
    require(path.is_absolute(), "absolute closed receipt reference required")
    actual = stream_hash(path)
    require(
        actual["bytes"] == ref["bytes"] and actual["sha256"] == ref["sha256"],
        "closed receipt byte custody",
    )
    return path.read_bytes()


def read_pinned(path: Path, digest: str) -> dict[str, Any]:
    raw = path.read_bytes()
    require(hashlib.sha256(raw).hexdigest() == digest, "exact fixed input digest")
    value = json.loads(raw)
    require(isinstance(value, dict), "object evidence required")
    return value


def canonical_relative(value: object) -> str:
    if not isinstance(value, str) or not value or value.startswith("/"):
        raise ValueError("nonempty inventory relative path required")
    require(
        not any(part in ("", ".", "..") for part in value.split("/")), "inventory traversal refused"
    )
    return value


def inventory_counts(rows: list[dict[str, Any]]) -> dict[str, int]:
    require(len({row["path"] for row in rows}) == len(rows), "inventory paths unique")
    values = {"records": len(rows), "files": 0, "directories": 0, "symlinks": 0, "bytes": 0}
    for row in rows:
        canonical_relative(row["path"])
        integer(row["mode"], "typed inventory mode")
        kind = row["type"]
        require(kind in ("file", "directory", "symlink"), "complete admitted object types")
        if kind == "file":
            values["files"] += 1
            values["bytes"] += integer(row["bytes"], "typed inventory file size")
            require(
                isinstance(row["sha256"], str)
                and re.fullmatch(r"[0-9a-f]{64}", row["sha256"]) is not None,
                "inventory content digest",
            )
        elif kind == "directory":
            values["directories"] += 1
        else:
            values["symlinks"] += 1
            require(
                isinstance(row["target"], str) and row["target"].startswith("/"),
                "original absolute link literal",
            )
    return values


def admit_inventory_triple(arrays: list[list[dict[str, Any]]]) -> list[dict[str, Any]]:
    require(len(arrays) == 3, "complete before/copy/old-link inventory triple")
    counters = []
    for rows in arrays:
        inventory_counts(rows)
        for row in rows:
            if "allocated_bytes" in row:
                integer(row["allocated_bytes"], "typed allocation metadata")
        counters.append(
            Counter(
                json.dumps(
                    {key: value for key, value in row.items() if key != "allocated_bytes"},
                    sort_keys=True,
                )
                for row in rows
            )
        )
    require(
        counters[0] == counters[1] == counters[2],
        "all defining path/type/mode/bytes/hash/links inventory fields conserved",
    )
    original = {row["path"]: row for row in arrays[0]}
    changes = []
    for number in (1, 2):
        for row in arrays[number]:
            before = original[row["path"]].get("allocated_bytes")
            after = row.get("allocated_bytes")
            if before != after:
                changes.append(
                    {
                        "path": row["path"],
                        "inventory_index": number,
                        "allocated_bytes_before": before,
                        "allocated_bytes_after": after,
                    }
                )
    return changes


def admit_expected_actions(results: list[dict[str, Any]], expected: list[str]) -> None:
    actual = Counter(action["source"] for result in results for action in result["actions"])
    require(actual == Counter(expected), "complete expected21 relocation source multiset")
    require(len(expected) == 21 and len(actual) == 21, "exact21 source denominator")


def admit_action_counts(action: dict[str, Any], values: dict[str, int]) -> None:
    for field, key in [

        and action["all_bytes_modes_links_verified_before_removal"] is True
        and action["old_path_full_readback_equal"] is True,
        "historical verified-copy sequence markers retained",
    )


def current_members(destination: Path) -> set[str]:
    values = set()
    pending = [destination]
    while pending:
        parent = pending.pop()
        for child in parent.iterdir():
            relative = child.relative_to(destination).as_posix()
            values.add(relative)
            if child.is_dir() and not child.is_symlink():
                pending.append(child)
    return values


def admit_current_members(actual: set[str], rows: list[dict[str, Any]], retired: set[str]) -> None:
    expected = {row["path"] for row in rows} - retired
    require(
        actual == expected,
        "complete current membership equals inventory minus exact3 archived plaintext files",
    )

    infra = value["unreadable_infrastructure"]
    require(
        {integer(item["pid"], "typed unreadable infrastructure PID") for item in infra}

    return {
        "reference": ref,
        "readable_processes": len(value["readable_processes"]),
        "active_candidate_users": 0,
        "unreadable_infrastructure": [199, 250],
        "scope": (

def archive_decode(action: dict[str, Any]) -> dict[str, Any]:
    integer(action["bytes"], "typed original archive size")
    ref = action["archive"]
    compressed = stream_hash(Path(ref["path"]))
    require(compressed == ref, "actual compressed archive digest and size")
    hashed = hashlib.sha256()
    size = 0
    with gzip.open(ref["path"], "rb") as reader:
        for block in iter(lambda: reader.read(BUFFER_BYTES), b""):
            size += len(block)
            hashed.update(block)
    actual = {"bytes": size, "sha256": hashed.hexdigest()}
    expected = {"bytes": action["bytes"], "sha256": action["sha256"]}
    require(
        actual == expected == action["input_stream"] == action["complete_decoded_stream"],
        "all decoded bytes match original plaintext refs",
    )
    require(
        action["plaintext_unavailable_until_decode"] is True
        and action["old_path_is_not_compressed_symlink"] is True
        and action["complete_bytes_verified_before_plaintext_retirement"] is True,
        "archive retirement sequence and type labels retained",
    )
    require(
        not Path(action["old"]).exists() and not Path(action["real"]).exists(),
        "exact plaintext paths retired",
    )
    require(
        not Path(action["old"]).is_symlink() and not Path(action["real"]).is_symlink(),
        "compressed bytes never disguised as JSON symlink",
    )
    require(
        action["reopen_recipe"]["shell_argv"] == ["gzip", "-dc", "--", ref["path"]],
        "exact explicit reopen recipe",
    )
    return {
        "old_path": action["old"],
        "real_path": action["real"],
        "archive": compressed,
        "actual_complete_decoded_stream": actual,
        "materialized_plaintext_bytes": 0,
        "buffer_bytes": BUFFER_BYTES,
    }


def git_identity() -> dict[str, Any]:

            cwd=ROOT,
            check=True,
            capture_output=True,
        )
        .stdout.decode()
        .strip()
        .split()
    )
    require(
        len(identity) == 2 and all(re.fullmatch(r"[0-9a-f]{40}", item) for item in identity),
        "actual current ROOT HEAD/tree",
    )
    # Actual HEAD is admitted above as exact40-hex; only fixed read-only ancestry argv.
    subprocess.run(  # noqa: S603
        ["/usr/bin/git", "--no-optional-locks", "merge-base", "--is-ancestor", SOURCE, identity[0]],
        cwd=ROOT,
        check=True,

    status = subprocess.run(
        ["/usr/bin/git", "--no-optional-locks", "status", "--porcelain=v1"],
        cwd=ROOT,
        check=True,

    paths = [line[3:] for line in status.decode().splitlines()]
    # Current parent integration and untracked native outputs are observations, not cleanup gates.

def actual_counter_controls(
    results: list[dict[str, Any]], expected: list[str]
) -> list[dict[str, Any]]:
    admitted = copy.deepcopy(results)
    omitted = admitted[-1]["actions"].pop()
    try:
        admit_expected_actions(admitted, expected)
    except ValueError:
        refusal = True
    else:
        raise ValueError("actual closed source omitted without refusal")
    module = ast.parse(Path(__file__).read_text())
    target = next(
        node
        for node in module.body
        if isinstance(node, ast.FunctionDef) and node.name == "admit_expected_actions"
    )
    target.body = [ast.Pass()]
    namespace = {"__name__": "actual_cleanup_source_counter_removal", "__file__": __file__}
    exec(compile(ast.fix_missing_locations(module), __file__, "exec"), namespace)  # noqa: S102
    namespace["admit_expected_actions"](admitted, expected)
    require(
        len(expected) == 21 and sum(len(value["actions"]) for value in admitted) == 20,
        "actual21 source marker and divergent20 retained",
    )
    return [
        {
            "name": "actual_closed21_source_omission_refused",
            "outcome": "PASS",
            "actual_refusal": refusal,
            "omitted_source": omitted["source"],
        },
        {
            "name": "actual_source_counter_removal_readmits20_without_transfer",
            "outcome": "PASS",
            "removed_property": "admit_expected_actions body only",
            "all_original_inventory_and_archive_refs_unchanged": True,
        },
    ]

def controls() -> dict[str, Any]:
    rows = [
        {"path": "folder", "mode": 448, "type": "directory"},
        {"path": "folder/x", "mode": 384, "type": "file", "bytes": 5, "sha256": "a" * 64},
        {"path": "alias", "mode": 511, "type": "symlink", "target": "/absolute/target"},
    ]
    values = inventory_counts(rows)
    allocation_variant = copy.deepcopy(rows)
    allocation_variant[1]["allocated_bytes"] = 4096
    original_variant = copy.deepcopy(rows)
    original_variant[1]["allocated_bytes"] = 8192
    require(
        len(admit_inventory_triple([original_variant, allocation_variant, allocation_variant]))
        == 2,
        "allocation changes do not alter defining byte/type comparison",
    )
    corrupted_inventory = copy.deepcopy(allocation_variant)
    corrupted_inventory[1]["sha256"] = "0" * 64
    try:
        admit_inventory_triple([original_variant, corrupted_inventory, allocation_variant])
    except ValueError:
        pass
    else:
