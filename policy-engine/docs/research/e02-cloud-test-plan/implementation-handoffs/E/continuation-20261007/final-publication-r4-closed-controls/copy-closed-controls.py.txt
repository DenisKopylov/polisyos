def sha(body):
    return hashlib.sha256(body).hexdigest()

def canonical(value):
    return json.dumps(value, ensure_ascii=False, sort_keys=True, separators=(',', ':'), allow_nan=False)

def strict_json(body):
    def pairs(items):
        out = {}
        for key, value in items:
            if key in out:
                raise ValueError('duplicate JSON key ' + key)
            out[key] = value
        return out
    def nonfinite(value):
        raise ValueError('nonfinite JSON ' + value)
    return json.loads(body, object_pairs_hook=pairs, parse_constant=nonfinite)

def identity(raw):
    if not isinstance(raw, dict):
        raise ValueError('identity record type')
    candidates = [raw.get(key) for key in ('stored_at', 'path', 'source_path', 'source')]
    path = next((value for value in candidates if isinstance(value, str) and Path(value).is_absolute()), None)
    size = raw.get('bytes')
    digest = raw.get('sha256')
    if not isinstance(path, str) or not Path(path).is_absolute() or type(size) is not int or size < 0 or not isinstance(digest, str) or len(digest) != 64 or any(c not in '0123456789abcdef' for c in digest):
        raise ValueError('complete typed source identity')
    return {'path': path, 'bytes': size, 'sha256': digest}

def admit_body(ref, lookup):
    body = Path(ref['path']).read_bytes() if lookup is None else lookup[(ref['sha256'], ref['bytes'])]
    if len(body) != ref['bytes'] or sha(body) != ref['sha256']:
        raise ValueError('original or stored bytes/hash ' + ref['path'])
    return body

def passive(family, name, body):
    if not isinstance(name, str) or not name or Path(name).is_absolute() or any(part in ('', '.', '..') for part in name.split('/')) or '\\' in name or any(ord(c) < 32 for c in name):
        raise ValueError('canonical relative passive route')
    if name.endswith('.py'):
        name += '.txt'
    if name.endswith('.json'):
        try:
            strict_json(body)
        except (ValueError, UnicodeError):
            name = name[:-5] + '.txt'
    return family + '/' + name

def expand(spec, doc, lookup=None):
    family, selector_path = spec['family'], spec['path']
    kind = spec['member_class']
    rows = [] if kind == 'none' else doc[kind]
    if not isinstance(rows, list) or len(rows) != spec['members']:
        raise ValueError('closed original complete member denominator')
    out = []
    physical = {}
    if kind == 'logical_aliases':
        for raw in doc['assets']:
            ref = identity(raw)
            body = admit_body(ref, lookup)
            route = raw['portable_path']
            if route in physical:
                raise ValueError('duplicate physical storage route')
            physical[route] = (ref, body, raw['storage'])
        used = set()
    for index, raw in enumerate(rows):
        if kind == 'logical_aliases':
            original = identity(raw['source'])
            stored, body, storage = physical[raw['portable_path']]
            used.add(raw['portable_path'])
            if canonical(storage) != canonical(raw['storage']):
                raise ValueError('logical/physical exact storage identity')
            mode = storage['mode']
            if mode == 'literal':
                if (identity(storage['original'])['bytes'], identity(storage['original'])['sha256']) != (original['bytes'], original['sha256']) or (stored['bytes'], stored['sha256']) != (original['bytes'], original['sha256']):
                    raise ValueError('literal source/storage identity')
            elif mode == 'gzip_literal_archive':
                if (identity(storage['decoded_original'])['bytes'], identity(storage['decoded_original'])['sha256']) != (original['bytes'], original['sha256']) or canonical(identity(storage['archive'])) != canonical(stored):
                    raise ValueError('compressed source/storage identity')
                decoded = gzip.decompress(body)
                if len(decoded) != original['bytes'] or sha(decoded) != original['sha256']:
                    raise ValueError('complete decompressed original bytes/hash')
            else:
                raise ValueError('unsupported closed storage mode')
            if lookup is None:
                admit_body(original, None)
            name = raw['portable_path']
        else:
            original = stored = identity(raw)
            body = admit_body(stored, lookup)
            mode = 'literal'
            name = raw.get('portable_path', raw.get('portable_name', raw.get('at', f'{index:03d}/' + Path(raw.get('logical_at', original['path'])).name)))

def selector_row(spec):
    ref = {key: spec[key] for key in ('path', 'bytes', 'sha256')}
    return {'family': spec['family'], 'selection_source_path': spec['path'], 'member_class': spec['input_kind'], 'original': ref, 'stored': ref, 'storage_mode': 'literal', 'logical_destination': 'selectors/' + spec['family'] + '.json', 'selected_record': None}

def carriers(rows):
    first = {}
    out = []
    logical = []
    for row in rows:
        key = (row['stored']['sha256'], row['stored']['bytes'])
        destination = first.setdefault(key, row['logical_destination'])
        logical.append(row | {'canonical_destination': destination, 'deduplicated': key in {(x['stored']['sha256'], x['stored']['bytes']) for x in out}})
        if not any(x['destination'] == destination for x in out):
            out.append({'stored': row['stored'], 'destination': destination})
    return logical, out

def carrier_root(p, manifest, layout):
    if layout not in ('stage', 'published'):
        raise ValueError('explicit admitted carrier layout required')
    primary = p.parent / 'payload' if layout == 'stage' else p.parent
    other = p.parent if layout == 'stage' else p.parent / 'payload'
    relatives = []
    for row in manifest['canonical_copies']:
        rel = row['destination']
        if not isinstance(rel, str) or Path(rel).is_absolute() or any(part in ('', '.', '..') for part in rel.split('/')) or '\\' in rel or rel.endswith('.py'):
            raise ValueError('pure full carrier route preflight')
        relatives.append(rel)
    if any((other / rel).exists() for rel in relatives):
        raise ValueError('ambiguous stage/published carrier layout')
    if any(not (primary / rel).is_file() for rel in relatives):
        raise ValueError('incomplete or wrong explicit carrier layout')
    return primary


def recompute_expected(p, manifest, payload_root):
    lookup = {}
    for row in manifest['canonical_copies']:
        ref = identity(row['stored'])
        rel = row['destination']
        if not isinstance(rel, str) or Path(rel).is_absolute() or any(part in ('', '.', '..') for part in rel.split('/')) or '\\' in rel or rel.endswith('.py'):
            raise ValueError('unsafe staged carrier path')
        body = (payload_root / rel).read_bytes()
        if passive('x', rel, body) != 'x/' + rel:
            raise ValueError('passive carrier format')
        if len(body) != ref['bytes'] or sha(body) != ref['sha256']:
            raise ValueError('complete staged carrier bytes/hash')
        lookup[(ref['sha256'], ref['bytes'])] = body
    expected = []
    for spec in EXPECTED:
        selector = admit_body(spec, lookup)
        expected.extend(expand(spec, strict_json(selector), lookup))
        expected.append(selector_row(spec))
    return expected

    got = collections.Counter(canonical({key: row[key] for key in fields}) for row in manifest['logical_sources'])
    want = collections.Counter(canonical(row) for row in expected)
    if got != want:
        raise ValueError('exact complete original member Counter')
    observed = [{key: row[key] for key in fields} for row in manifest['logical_sources']]
    logical, copies = carriers(observed)
    if canonical(manifest['logical_sources']) != canonical(logical) or canonical(manifest['canonical_copies']) != canonical(copies):
        raise ValueError('exact all logical/carrier route and occurrence identity')
    counts = {'logical_count': len(observed), 'original_logical_bytes': sum(row['original']['bytes'] for row in observed), 'stored_logical_bytes': sum(row['stored']['bytes'] for row in observed), 'unique_payload_files': len(copies), 'unique_payload_bytes': sum(row['stored']['bytes'] for row in copies), 'deduplicated_occurrences': len(observed) - len(copies)}

        raise ValueError('recomputed typed complete denominators')

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--manifest', required=True)
    parser.add_argument('--destination-root', required=True)
    parser.add_argument('--layout', choices=('stage', 'published'), required=True)
    args = parser.parse_args()
    p = Path(args.manifest)
    raw = p.read_bytes()
    manifest = strict_json(raw)
    counts = validate(p, manifest, args.layout)
    payload_root = carrier_root(p, manifest, args.layout)
    metadata = {name: (p.parent / name).read_bytes() for name in METADATA}
    if metadata['manifest.json'] != raw:
        raise ValueError('manifest changed during observation')
    if metadata['copy-closed-controls.py.txt'] != TRUSTED_COPIER_BYTES:
        raise ValueError('trusted executing program companion identity')
    for name, ref in FIXED_METADATA.items():
        if len(metadata[name]) != ref['bytes'] or sha(metadata[name]) != ref['sha256']:
            raise ValueError('complete mandatory companion bytes/hash before writes')
    for name, body in metadata.items():
        if name.endswith('.json'):
            strict_json(body)
    if canonical(strict_json(metadata['qualification.json'])) != canonical(QUALIFICATION):
        raise ValueError('current typed qualification companion')
    summary = counts | {'metadata_files': len(METADATA), 'expected_git_paths': counts['unique_payload_files'] + len(METADATA)}
    if canonical(strict_json(metadata['build.stdout.json'])) != canonical(summary):
        raise ValueError('complete actual build summary')
    payload = {}
    for row in manifest['canonical_copies']:
        body = (payload_root / row['destination']).read_bytes()
        if len(body) != row['stored']['bytes'] or sha(body) != row['stored']['sha256']:
            raise ValueError('cached complete carrier identity before writes')
        payload[row['destination']] = body
    target = Path(args.destination_root) / PREFIX
    if target.exists():
        raise ValueError('existing destination')
    for name, body in payload.items():
        destination = target / name
        destination.parent.mkdir(parents=True, exist_ok=True)
        destination.write_bytes(body)
    for name, body in metadata.items():

        destination.parent.mkdir(parents=True, exist_ok=True)
        destination.write_bytes(body)
