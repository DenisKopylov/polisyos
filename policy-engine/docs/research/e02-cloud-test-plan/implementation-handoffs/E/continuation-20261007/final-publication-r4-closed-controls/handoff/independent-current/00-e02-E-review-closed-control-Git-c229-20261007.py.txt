def ref(raw: dict) -> dict:
    candidates = [raw.get(k) for k in ['stored_at', 'path', 'source_path', 'source']]

    fail_if(path is None or type(raw['bytes']) is not int or raw['bytes'] < 0, 'typed source ref')
    return {'path': path, 'bytes': raw['bytes'], 'sha256': raw['sha256']}
