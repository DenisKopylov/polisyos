def require(value: object, message: str) -> None:
    if not value:
        raise ValueError(message)


def token(value: object) -> str:
    return json.dumps(value, sort_keys=True, separators=(",", ":"), allow_nan=False)


def same(actual: object, expected: object, message: str) -> None:
    require(token(actual) == token(expected), message)


def checked(path: pathlib.Path, size: int, digest: str) -> bytes:
    require(type(size) is int and size >= 0, "typed_bytes")
    raw = path.read_bytes()
    same(len(raw), size, "content_size")
    same(hashlib.sha256(raw).hexdigest(), digest, "content_digest")
    return raw


def pinned(name: str) -> dict[str, object]:
    return json.loads(checked(*PINNED[name]))


def parse_cases(raw: bytes, job: str) -> list[dict[str, object]]:
    require(
        b"<!DOCTYPE" not in raw.upper() and b"<!ENTITY" not in raw.upper(), "unsafe_XML_entities"
    )
    # DTD/entity declarations explicitly refused before stdlib parser entry.
    root = ET.fromstring(raw)  # noqa: S314
    cases = []
    for case in root.iter("testcase"):
        outcome = (
            "errors"
            if case.find("error") is not None
            else "failed"
            if case.find("failure") is not None
            else "skipped"
            if case.find("skipped") is not None
            else "passed"
        )
        cases.append(
            {
                "name": case.get("name"),
                "classname": case.get("classname", ""),
                "file": case.get("file"),
                "seconds": case.get("time", "0"),
                "outcome": outcome,
                "job_name": job,
            }
        )
    return cases


def counts(cases: list[dict[str, object]]) -> dict[str, int]:
    counter = Counter(case["outcome"] for case in cases)
    require(set(counter) <= {"passed", "failed", "errors", "skipped"}, "case_outcome")
    return {
        "cases": len(cases),
        **{k: counter[k] for k in ("passed", "failed", "errors", "skipped")},
    }


def admit_cases(actual: list[dict[str, object]], expected: list[dict[str, object]]) -> None:
    same(actual, expected, "complete_ordered_JUnit_case_sequence")


def admit_claims(
    receipt: dict[str, object],
    case_index: dict[str, object],
    all_cases: list[dict[str, object]],
    a_cases: list[dict[str, object]],
) -> None:
    for key, value in {
        "candidate_sha": SOURCE,
        "candidate_tree_sha": TREE,
        "collection_state": "COMPLETE_BOUND",
        "finding_closure": False,
        "numeric_counts": counts(all_cases),
        "native_numeric_counts_excluding_A_packets": counts(
            [c for c in all_cases if c not in a_cases]
        ),
        "unattributed_owner_packet_counts": counts([]),
        "native_numeric_attribution_complete": True,
    }.items():
        same(receipt[key], value, "receipt_claim:" + key)
    same(case_index["source_sha"], SOURCE, "case_source")
    same(case_index["source_tree"], TREE, "case_tree")
    same(case_index["case_count"], len(all_cases), "case_count_typed")
    same(case_index["numeric_counts"], counts(all_cases), "flat_case_counts")
    same(case_index["finding_closure"], False, "flat_finding_closure")
    admit_cases(case_index["cases"], all_cases)
    same(receipt["foreign_owner_A_packet_cases"][0]["cases"], a_cases, "actual_A_cases")
    same(receipt["foreign_owner_A_packet_cases"][0]["counts"], counts(a_cases), "actual_A_counts")


def source_blob(path: str, admitted: set[str]) -> bytes:
    require(path in admitted and not path.startswith("-"), "admitted_source_path")
    require(
        "\x00" not in path and ".." not in pathlib.PurePosixPath(path).parts,
        "canonical_source_path",
    )
    # Constant executable/revision and paths admitted by the closed immutable plan; no shell.
    return subprocess.check_output(  # noqa: S603
        ["/usr/bin/git", "show", SOURCE + ":" + path], cwd=REPO, shell=False
    )


def remove_body(name: str) -> object:
    tree = ast.parse(pathlib.Path(__file__).read_text())
    tree.body = [node for node in tree.body if not isinstance(node, ast.If)]
    functions = [
        node for node in tree.body if isinstance(node, ast.FunctionDef) and node.name == name
    ]
    require(len(functions) == 1, "one_actual_guard")
    functions[0].body = [ast.Pass()]
    scope = {"__file__": __file__}
    # Own observer source only; this transformation removes one measured admission body.
    exec(compile(ast.fix_missing_locations(tree), __file__, "exec"), scope)  # noqa: S102
    return scope[name]


def refusal(callback: object, name: str) -> dict[str, object]:
    try:
        callback()
    except ValueError as error:
        return {"case": name, "outcome": "PASS", "actual_refusal": str(error)}
    raise ValueError("actual_corruption_readmitted:" + name)
