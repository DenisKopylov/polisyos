def digest(raw):return hashlib.sha256(raw).hexdigest()
def git(*args):return subprocess.run(['git',*args],cwd=ROOT,check=True,capture_output=True).stdout

def admit(value):
    raw=(OUT/'current-source-input-qualification.json').read_bytes()
    if digest(raw)!=TRUSTED:raise ValueError('closed_report_changed')
    expected=json.loads(raw)
    if value!=expected:raise ValueError('claims_changed_from_closed_observer')
    if value['source']!=SOURCE or value['parent_source']!=PARENT or value['source_immutable'] is not True:raise ValueError('source_identity')
    if git('diff','--name-status','--find-renames',PARENT,SOURCE).decode()!=value['full_rename_aware_footprint']:raise ValueError('full_footprint')
    for which,ref in [(PARENT,value['changed_input']['old']),(SOURCE,value['changed_input']['new'])]:
        b=git('show',which+':'+PATH)
        if digest(b)!=ref['sha256'] or len(b)!=ref['bytes']:raise ValueError('source_blob')
    rows=[]
    for name in ['complete-current-static-input-index.json','complete-current-architecture-input-edge-index.json']:
        a=(OUT/name).read_bytes();z=(OUT/(name+'.gz')).read_bytes()
        if gzip.decompress(z)!=a:raise ValueError('gzip_custody')
        rows.append(json.loads(a))
    static,architecture=rows
    if type(value['static_denominator']['current_files'])is not int or value['static_denominator']['current_files']!=len(static['rows']):raise ValueError('typed_static_denominator')
    if value['static_denominator']['current_sha256']!=static['mapping_sha256']:raise ValueError('static_mapping')
    if any(type(value['architecture_collector'][key])is not int for key in ['files','complete_edges','raw_violations','reported_edges']):raise ValueError('typed_arch_denominator')
    if value['architecture_collector']['complete_edges']!=len(architecture['edges']) or value['architecture_collector']['reported_edges']!=len(architecture['reported_tuples']):raise ValueError('complete_vs_reported_census')
    if value['static_denominator']['native_graph_or_gate_rerun']is not False or value['finding_closure']is not False or value['P41']!='not_established':raise ValueError('scope')
    return True
