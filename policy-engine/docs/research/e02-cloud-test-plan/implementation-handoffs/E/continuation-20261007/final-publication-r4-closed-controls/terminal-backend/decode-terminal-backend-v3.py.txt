def require(condition: bool, reason: str) -> None:
    if not condition:
        raise ValueError(reason)


def read(path: Path) -> dict[str, Any]:
    value = json.loads(path.read_bytes())
    require(isinstance(value, dict), "object input required")
    return value


def record(path: Path) -> dict[str, Any]:
    raw = path.read_bytes()
    return {"path": str(path), "bytes": len(raw), "sha256": hashlib.sha256(raw).hexdigest()}


def xml_decode(raw: bytes) -> tuple[dict[str, int], list[dict[str, Any]]]:
    text = raw.decode("utf-8")
    require(
        "<!DOCTYPE" not in text.upper() and "<!ENTITY" not in text.upper(), "DTD/entity refused"
    )
    # Strict UTF-8 admission and DTD/entity refusal precede parsing this existing evidence.
    root = ElementTree.fromstring(text)  # noqa: S314
    counts = {"cases": 0, "passed": 0, "failed": 0, "errors": 0, "skipped": 0}
    cases = []
    for ordinal, case in enumerate(root.iter("testcase")):
        counts["cases"] += 1
        child = None
        if case.find("failure") is not None:
            outcome, field = "FAIL", "failed"
            child = case.find("failure")
        elif case.find("error") is not None:
            outcome, field = "ERROR", "errors"
            child = case.find("error")
        elif case.find("skipped") is not None:
            outcome, field = "SKIP", "skipped"
            child = case.find("skipped")
        else:
            outcome, field = "PASS", "passed"
        counts[field] += 1
        detail = "" if child is None else child.text or ""
        cases.append(
            {
                "ordinal": ordinal,
                "classname": case.get("classname", ""),
                "name": case.get("name", ""),
                "outcome": outcome,
                "native_message": None if child is None else child.get("message"),
                "native_detail_bytes_utf8": len(detail.encode()),
                "native_detail_sha256_utf8": hashlib.sha256(detail.encode()).hexdigest(),
            }
        )
    return counts, cases


def validate_fields(
    receipt: dict[str, Any], launch: dict[str, Any], counts: dict[str, int] | None
) -> str:
    require(
        receipt["candidate_sha"] == SOURCE
        and receipt["candidate_tree_sha"] == TREE
        and receipt["head_at_end"] == SOURCE,
        "exact terminal source/head/tree required",
    )
    require(receipt["source_identity_before"] == FRAME, "admitted physical source frame drift")
    after = receipt["source_identity_after"]
    require(set(after) == set(FRAME), "terminal frame fields")
    require(
        all(type(after[k]) is int and after[k] >= 0 for k in ["tracked_paths", "bytes"]),
        "terminal frame typed denominator",
    )
    require(
        re.fullmatch(r"[0-9a-f]{64}", after["framed_sha256"]) is not None, "terminal frame digest"
    )
    require(type(receipt["source_immutable"]) is bool, "source frame flag type")
    require(
        receipt["source_immutable"] is (after == FRAME),
        "source frame flag does not match admitted/end source identities",
    )
    config = receipt["git_input_config"]
    require(config["stable"] is True, "Git config drift")
    require(
        config["sha256"] == config["after_sha256"]
        and re.fullmatch(r"[0-9a-f]{64}", config["sha256"]) is not None,
        "Git configuration identity mismatch",
    )
    require(launch["source"] == SOURCE and launch["tree"] == TREE, "launch source drift")
    require(launch["added_test_exclusions"] == [], "added native exclusions")
    require(launch["artificial_process_thread_worker_cpu_caps"] is False, "native cap declared")
    require(
        launch["native_scope"]
        == ["pytest", "-m", "not integration", "--ignore=tests/unit/runtime/http"],
        "native scope drift",
    )
    argv = launch["argv"]
    require(argv.count("--") == 1, "unique runner/child boundary")
    expected_command = argv[argv.index("--") + 1 :]
    require(receipt["command"] == expected_command, "effective full native command changed")
    require(
        receipt["cwd"] == "/workspace/e02-E-continuation-20261006/policy-engine",
        "native cwd drift",
    )
    env = receipt["environment"]
    require(
        env["executable"]
        == "/workspace/e02-E-continuation-20261006/policy-engine/.venv/bin/python",
        "ROOT interpreter drift",
    )
    require(env["python"].startswith("3.14.7 "), "ROOT Python drift")
    require(
        all(env["packages"].get(k) == v for k, v in NUMERICAL.items()), "numerical package drift"
    )
    variables = env["selected_variables"]
    require(all(variables.get(k) is None for k in CAPS), "effective inherited cap")
    require(
        all(variables[k] == v for k, v in launch["environment_set"].items() if k in variables),
        "actual launch environment drift",
    )
    recorded_counts = receipt["counts"]
    if recorded_counts is not None:
        require(isinstance(recorded_counts, dict), "recorded JUnit object required")
        require(
            all(type(v) is int and v >= 0 for v in recorded_counts.values()),
            "recorded noninteger JUnit count",
        )
    require(recorded_counts == counts, "actual JUnit case-count drift")
    if counts is not None:
        require(all(type(v) is int and v >= 0 for v in counts.values()), "noninteger JUnit count")
        require(
            counts["cases"] == sum(counts[k] for k in ["passed", "failed", "errors", "skipped"]),
            "JUnit complete outcome denominator",
        )
    exit_code = receipt["exit_code"]
    require(type(exit_code) is int, "integer native exit required")
    if receipt["backend_error"] is not None:
        expected = "UNRUN"
    elif exit_code != 0 or receipt["source_immutable"] is False:
        expected = "FAIL"
    elif counts is not None and counts["cases"] > 0 and counts["skipped"] == counts["cases"]:
        expected = "SKIP"
    else:
        expected = "PASS"
    require(receipt["outcome"] == expected, "actual process/check status disagreement")
    if expected == "PASS" and counts is not None:
        require(counts["failed"] == counts["errors"] == 0, "PASS with native failing cases")
    return expected


def admit_drift_observation(value: dict[str, Any]) -> None:
    require(value["head"] == SOURCE and value["tree"] == TREE, "drift observation source")
    require(value["tracked_input_count"] == len(value["rows"]) == 29, "drift input denominator")
    require(value["observed_changed_product_or_test_Python"] is False, "drift code classification")
    require(value["ROOT_HEAD_changed"] is False, "drift HEAD classification")
    require(value["native_source_input_immutable_claim"] is False, "observed input drift erased")
    require(value["snapshot_is_atomic_whole_run"] is False, "snapshot cannot establish fullrun")
    require(value["restoration_or_cleanup_performed"] is False, "observer custody")
    for row in value["rows"]:
        require(
            row["path"].startswith("policy-engine/architecture/policy_design_case/layer3_g"),
            "drift path domain",
        )
        require(row["path"].endswith(".json"), "drift generated input type")
        require(row["current"]["sha256"] != row["head"]["sha256"], "actual input changed")
        for kind in ["current", "head"]:
            ref = row[kind]
            name = ref["portable_name"]
            require(
                re.fullmatch(r"[0-9a-f]{64}\.(current|head)\.json", name) is not None,
                "portable drift path admission",
            )
            body = (DRIFT.parent / name).read_bytes()
            require(type(ref["bytes"]) is int and len(body) == ref["bytes"], "drift input size")
            require(hashlib.sha256(body).hexdigest() == ref["sha256"], "drift input digest")

def installed_metadata() -> dict[str, Any]:
    require(record(POSTINSTALL)["sha256"] == POSTINSTALL_HASH, "postinstall207 snapshot drift")
    installed = read(POSTINSTALL)["metadata"]["distributions"]
    require(len(installed) == 207, "installed207 denominator")
    changed = []
    for ref in installed:
        if record(Path(ref["path"])) != {k: ref[k] for k in ["path", "bytes", "sha256"]}:
            changed.append(ref["name"])
    require(changed == [], "installed distribution metadata changed")
    return {
        "postinstall207_snapshot": record(POSTINSTALL),
        "all207_metadata_record_bodies_match_terminal_readback": True,
        "existing203_and_protected_metadata": {
