def require(condition: bool, reason: str) -> None:
    if not condition:
        raise ValueError(reason)


def record(path: Path) -> dict[str, Any]:
    raw = path.read_bytes()
    return {"path": str(path), "bytes": len(raw), "sha256": hashlib.sha256(raw).hexdigest()}


def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--output", type=Path, required=True)
    args = parser.parse_args()
    require(not args.output.exists(), "append-only current layer required")
    objects = {}
    refs = {}
    for key, (path, digest) in PINS.items():
        ref = record(path)
        require(ref["sha256"] == digest, "exact current input drift")
        value = json.loads(path.read_bytes())
        require(isinstance(value, dict), "object input required")
        objects[key], refs[key] = value, ref
    evidence = objects["current_actual_counterexamples"]
    require(evidence["source"] == SOURCE and evidence["tree"] == TREE, "probe identity drift")
    require(evidence["source_unchanged"] is True, "probe source changed")

    conditions = objects["independent_owner_response"]["condition_responses"]

        require(old[finding]["historical_ledger_status"] == "held", "historical held identity")

        rows.append(
            {

                "frozen_previous_E_proposal": "closed",
                "current_accountable_E_decision": "open",
                "reason": (
                    "Current exact700/64d source has confirmed technical counterexamples. "
                    "The implementation requires correction and independent defining-property "
                    "acceptance."
                ),
                "semantic_owner": "E scoped B201/B202 appointment a0ac10fc",
                "four_tracked_semantic_decisions_preserved": True,
                "missing_semantic_owner_is_not_the_blocker": True,
                "next_owner": "E canonical producer/profile and shared MC mechanism writer",
                "minimal_inputs": (
                    "Already available native equal-length/no-alignment marginal fixture and "
                    "100 positive atoms (99 zeros, last1e15). No production history needed "
                    "for these fixes."
                ),
                "next_verifiable_result": (
                    "Append-only implementation SHA/tree; healthy declared-aligned input "
                    "through CAS "
                    "and fresh native consumers; absent alignment refuses before callbacks; "
                    "one admitted finite probability convention for named mean, draw CDF "
                    "and sorted "
                    "quantile CDF, or explicit versioned approximation/error admission; "
                    "independent adversarial and property-removal evidence."
                ),
                "source_fit_noise_institutional_authority_boundary": (
                    "Actual posterior scientific quality/source fit/noise remains B197's separate "
                    "held criterion; it does not excuse these available mechanical defects."
                ),
                "formal_ledger_change": False,
            }
        )
    result = {
        "schema": "policyos.e02.E.current_IR_technical_counterexample_dispositions.v1",
        "source": SOURCE,
        "tree": TREE,
        "input_refs": refs,

        "historical_ledger_unchanged": {"partial": 49, "held": 4, "closed": 1},
        "frozen_previous_E_proposals_unchanged": {"closed": 43, "limited": 9, "held": 2},
        "current_E_proposals_after_exact_two_row_correction": {
            "closed": 41,
            "limited": 9,
            "held": 2,
            "open": 2,
        },
        "scope": (
            "Two current technical corrections only; no duplicate54 plan or "
            "new semantic ratification"
        ),
        "rows": rows,
        "independent_exact_condition_responses": conditions,
        "frozen54_V2_rewritten": False,

    }
    args.output.write_text(json.dumps(result, ensure_ascii=False, indent=2) + "\n")
