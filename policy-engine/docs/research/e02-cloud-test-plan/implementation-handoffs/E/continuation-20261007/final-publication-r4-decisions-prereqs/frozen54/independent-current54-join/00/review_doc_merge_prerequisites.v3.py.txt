def git(*args):
    return subprocess.run(['git', '-C', str(ROOT), *args], capture_output=True, check=True).stdout


def text(*args):
    return git(*args).decode().strip()


def write(path, value):
    with path.open('x') as stream:
        stream.write(json.dumps(value, indent=2, sort_keys=True) + '\n')


def require(condition, message):
    if not condition:
        raise ValueError(message)

all_paths = []
for owner, revision in HEADS.items():
    base = text('merge-base', FROZEN, revision)
    incoming = git('diff', '--name-status', '--no-renames', '-z', base, revision).split(b'\0')
    pairs = []
    index = 0
    while incoming[index:index + 1] != [b''] and index < len(incoming):
        status = incoming[index].decode()
        require(not status.startswith(('R', 'C')), 'rename/copy needs separate merge review')
        pairs.append((status, incoming[index + 1].decode()))
        index += 2
    own = set(git('diff', '--name-only', base, FROZEN).decode().splitlines())
    records = []

    for status, path in pairs:

        source_revision = base if status == 'D' else revision

        records.append({'status': status, 'path': path, 'bytes': len(raw), 'sha256': hashlib.sha256(raw).hexdigest(), 'git_blob': text('rev-parse', source_revision + ':' + path), 'blob_revision': source_revision, 'active_python': path.endswith('.py') and status != 'D'})

            if path.endswith('.json'):
                value = json.loads(raw)
                item['top_keys'] = list(value) if isinstance(value, dict) else None
                item['identity_and_scope'] = {key: value[key] for key in ('candidate_sha', 'candidate_tree_sha', 'base_sha', 'source_sha', 'source', 'tree', 'schema', 'finding_closure', 'formal_ledger_change', 'path_count', 'python_path_count', 'deciding_output_count') if isinstance(value, dict) and key in value}
            companions.append(item)
    renames = []
    for item in records:
        if item['status'] == 'D':
            matches = [r for r in records if r['status'] == 'A' and r['path'] == item['path'][:-5] + '.txt']
            require(item['path'].endswith('.stdout.json') and len(matches) == 1 and matches[0]['sha256'] == item['sha256'] and matches[0]['bytes'] == item['bytes'], 'doc deletion not exact byte-preserved stdout role rename')

    topics[owner] = {'byte_preserved_stdout_role_renames': renames, 'revision': revision, 'tree': text('rev-parse', revision + '^{tree}'), 'merge_base': base, 'incoming_path_count': len(records), 'incoming_bytes': sum(r['bytes'] for r in records), 'complete_footprint': records, 'root_changed_path_overlap': [], 'mandatory_receipt_publication_companions': companions, 'last_commit_parent': text('rev-parse', revision + '^'), 'last_commit_path_count': len(git('diff', '--name-only', revision + '^', revision).decode().splitlines()), 'active_python_paths': [r['path'] for r in records if r['active_python']]}
    all_paths.extend(record['path'] for record in records)
