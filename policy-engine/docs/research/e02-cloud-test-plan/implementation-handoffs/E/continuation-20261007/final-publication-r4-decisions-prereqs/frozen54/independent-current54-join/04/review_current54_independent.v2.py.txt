def digest(raw): return hashlib.sha256(raw).hexdigest()
def record(path):
    raw=Path(path).read_bytes()

def canonical(value):
    return digest(json.dumps(value,sort_keys=True,ensure_ascii=False,separators=(',',':')).encode())
def write(name,value):
