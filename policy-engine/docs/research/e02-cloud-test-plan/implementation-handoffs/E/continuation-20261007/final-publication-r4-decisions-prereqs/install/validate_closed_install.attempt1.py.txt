def require(condition, reason):
    if not condition:
        raise ValueError(reason)


def digest(raw):
    return hashlib.sha256(raw).hexdigest()


def typed_equal(a, b):
    if type(a) is not type(b):
        return False
    if isinstance(a, dict):
        return a.keys() == b.keys() and all(typed_equal(a[k], b[k]) for k in a)
    if isinstance(a, list):
        return len(a) == len(b) and all(typed_equal(x, y) for x, y in zip(a, b))
    return a == b


def normalize(name):
    return name.lower().replace('_', '-').replace('.', '-')


def admit_sha(value, size=40):
    require(isinstance(value, str) and re.fullmatch('[0-9a-f]{' + str(size) + '}', value) is not None, 'full hex identity')


def admit_relative(value):
    require(isinstance(value, str) and bool(value), 'relative path type')
    p = PurePosixPath(value)
    require(not p.is_absolute() and bool(p.parts) and all(x not in ('.', '..') for x in p.parts) and '\\' not in value and '\x00' not in value and not value.startswith('-'), 'relative path admission')
    require(str(p) == value, 'canonical relative path')


def load_inputs(base):

    rows = index['files']
    require(type(rows) is list and bool(rows), 'full input rows')
    names = set(); originals = set(); result = {}; raw = {}
    for row in rows:
        require(type(row) is dict and set(row) == {'key', 'path', 'bytes', 'sha256', 'stored_at'}, 'input shape')
        require(type(row['key']) is str and row['key'] not in names, 'input key uniqueness')
        require(type(row['path']) is str and row['path'].startswith('/') and row['path'] not in originals, 'original path uniqueness')
        require(type(row['bytes']) is int and row['bytes'] >= 0, 'input byte integer')
        admit_sha(row['sha256'], 64); admit_relative(row['stored_at'])
        names.add(row['key']); originals.add(row['path'])
    for row in rows:
        data = (base / row['stored_at']).read_bytes()
        require(len(data) == row['bytes'] and digest(data) == row['sha256'], 'input custody:' + row['key'])
        raw[row['key']] = data
        result[row['key']] = json.loads(data) if row['stored_at'].endswith('.json') else data.decode()
    require(type(index['total_bytes']) is int and index['total_bytes'] == sum(x['bytes'] for x in rows), 'input denominator')
    require(set(result) == {'historical-selection','recipe','artifacts','requirements','observer','historical-observation','historical-observer-stdout','historical-observer-stderr','a11y-old-admission','readiness','wheels-receipt','prefetch-source','admission-refusal','fresh-before','after','install','delta','install-stdout','install-stderr','preflight','preflight-source','preflight-stdout','preflight-stderr','pyvenv-config','original-workspace14','original-workspace14-stdout','publication','publication-before-correction','publication-stdout','physical-metadata','wheel-readback','source-readback'}, 'complete input class')
    return index, result, raw


def ref_for(index, key):
    row = next(r for r in index['files'] if r['key'] == key)
    return {k: row[k] for k in ('path', 'bytes', 'sha256')}


def bind_ref(index, value, key):
    require(typed_equal(value, ref_for(index, key)), 'bound reference:' + key)


def metadata(observation):

    rows = value['distributions']

    names = set(); paths = set()

        require(type(row['name']) is str and bool(row['name']) and type(row['version']) is str and bool(row['version']), 'metadata names/versions')
        name = normalize(row['name']); require(name not in names, 'metadata name uniqueness'); names.add(name)
        require(type(row['path']) is str and row['path'].startswith(ENV + '/lib/python3.14/site-packages/') and row['path'].endswith('.dist-info/METADATA') and row['path'] not in paths, 'metadata path class'); paths.add(row['path'])
        require(type(row['bytes']) is int and row['bytes'] > 0, 'metadata byte integer'); admit_sha(row['sha256'], 64)
    framed = digest(json.dumps(rows, sort_keys=True, separators=(',', ':')).encode())
    require(type(value['count']) is int and value['count'] == len(rows), 'metadata full count')
    require([r['path'] for r in rows] == sorted(paths), 'metadata row order')
    require(value['framed_sha256'] == framed and observation['metadata_after_framed_sha256'] == framed, 'metadata framed identity')
    require(observation['metadata_stable_during_observation'] is True and observation['environment_mutated'] is False, 'read-only stable observation')

    require(typed_equal(value['protected_numerical_packages'], protected), 'protected complete axes')

    return rows, framed, protected


def source_inputs(index, inputs, repo):

    require([r['path'] for r in record['rows']] == paths, 'complete source footprint')
    for row in record['rows']:
        admit_sha(row['source_sha']); admit_sha(row['git_blob']); admit_sha(row['sha256'],64); admit_relative(row['path'])
        require(row['source_sha'] == SOURCE and row['physical_matches'] is True and type(row['bytes']) is int and row['bytes'] > 0, 'source preflight')
        require(row['command'] == ['git','show',SOURCE + ':' + row['path']], 'source read command')
    # Whole source-object records are admitted before any child callback.
    require(subprocess.check_output(['git','rev-parse',SOURCE + '^{tree}'], cwd=repo, text=True).strip() == TREE, 'immutable source tree')
    data = {}
    for row in record['rows']:
        raw = subprocess.check_output(['git','show',SOURCE + ':' + row['path']], cwd=repo)
        require(len(raw) == row['bytes'] and digest(raw) == row['sha256'], 'immutable Git source bytes')
        require(subprocess.check_output(['git','hash-object','--stdin'], input=raw, cwd=repo, text=True).strip() == row['git_blob'], 'Git blob identity')
        data[row['path']] = raw
    return data

def admit_summary_claims(packet, expected):
    require(type(packet) is dict and packet.get('schema')=='policyos.e02.closed_declared_prerequisite_custody.v1','packet schema')
    require(typed_equal(packet.get('claims'),expected),'summary_mismatch')
    return {'structured_claims':sum(leaf_count(v) for v in expected.values()),'claim_sha256':digest(json.dumps(expected,sort_keys=True,separators=(',',':')).encode())}


def leaf_count(v):
    if isinstance(v,dict):return sum(leaf_count(x) for x in v.values())
    if isinstance(v,list):return sum(leaf_count(x) for x in v)
    return 1


def validate(base, repo, packet=None):
    expected=derive_claims(base,repo)
    if packet is None:packet=json.loads((base/'closed-install-custody.json').read_text())
    summary_receipt=admit_summary_claims(packet,expected)
    return {'check_outcome':'PASS','source':SOURCE,'tree':TREE,'summary_receipt':summary_receipt,'input_denominator':expected['input_denominator'],'native_executed':False,'environment_mutated':False,'global_backend_result_established':False,'finding_closure':False}
