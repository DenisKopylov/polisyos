def require(value, reason):
    if not value:
        raise ValueError(reason)

def ref(path):
    data = path.read_bytes()
    return {'path': str(path), 'bytes': len(data), 'sha256': hashlib.sha256(data).hexdigest()}

def run(name, manifest_path, source_path, expected_exit):
    target = CATALOG / (name + '-written-root')
    require(not target.exists(), 'fresh actual output target')
    argv = [sys.executable, str(source_path), '--manifest', str(manifest_path), '--destination-root', str(target)]
    process = subprocess.run(argv, capture_output=True, text=True)
    stdout = CATALOG / (name + '.stdout.txt')
    stderr = CATALOG / (name + '.stderr.txt')
    stdout.write_text(process.stdout)
    stderr.write_text(process.stderr)
    require(process.returncode == expected_exit, 'unexpected literal native CLI result: ' + name)
    if expected_exit != 0:
        require(not target.exists(), 'refusal before any destination creation')
    return {'name': name, 'source': ref(source_path), 'input': ref(manifest_path), 'argv': argv, 'cwd': str(Path.cwd()), 'exit_code': process.returncode, 'stdout': ref(stdout), 'stderr': ref(stderr), 'destination': str(target), 'destination_created': target.exists()}
