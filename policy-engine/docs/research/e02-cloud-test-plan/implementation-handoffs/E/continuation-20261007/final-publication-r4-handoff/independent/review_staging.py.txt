def require(value, reason):
    if not value:
        raise ValueError(reason)

def ref(path):
    data = path.read_bytes()
    return {'path': str(path), 'bytes': len(data), 'sha256': hashlib.sha256(data).hexdigest()}

def check_bytes(path, size, digest, *, skip_stage_binding=False):
    data = path.read_bytes()
    if not skip_stage_binding:
        require(type(size) is int and len(data) == size and hashlib.sha256(data).hexdigest() == digest, 'actual file bytebinding: ' + str(path))
    return data

def selected_rows(family, value):
    if family == 'postfreeze':
        return [value['deciding_final']] + value['current_complete_deciding'] + [r['portable_source'] for r in value['literal_archived_observers']]
    if family == 'runtime-retry':
        return value['files']
    if family == 'criteria-review':
        return value['rows']
    return value['assets']

def validate(value, *, skip_stage_binding=False, staged_override=None):
    require(value['verification_source'] == '64d7444a18c55df7b88b71b7699a2f1b25ca24bd' and value['verification_tree'] == 'aef23574d3162abb19bf5a1c67d02624ff9d4d1d', 'source/tree')
    require(value['proposed_git_prefix'] == PREFIX, 'fixed prefix')
    require({r['family'] for r in value['selection_inputs']} == SELECTORS.keys() and len(value['selection_inputs']) == 5, 'all five mandatory selectors')
    expected = Counter()
    parsed = {}
    aliases = []
    for selection in value['selection_inputs']:
        family = selection['family']
        source, size, digest = SELECTORS[family]
        require((selection['selection_source_path'], selection['bytes'], selection['sha256']) == (source, size, digest), 'frozen original selector identity')
        literal = check_bytes(Path(source), size, digest)
        selected = json.loads(literal)
        parsed[family] = selected
        rows = selected_rows(family, selected)
        require(selection['selected_count'] == len(rows) and selection['selected_bytes'] == sum(row['bytes'] for row in rows), 'original selector full denominator')
        for row in rows:
            path = row.get('source_path', row.get('path'))
            check_bytes(Path(path), row['bytes'], row['sha256'])
            expected[(family, path, row['bytes'], row['sha256'])] += 1
        expected[('selectors', source, size, digest)] += 1
        if family == 'postfreeze':
            for row in selected['literal_archived_observers']:
                require(Path(row['executed_source_path']).read_bytes() == Path(row['portable_source']['path']).read_bytes(), 'executed observer archive byte identity')
        if family == 'closed-lint':
            for field in ('aliased_optimized_full_review', 'prior_preview_same_bytes'):
                alternate = selected[field]['original_input']
                data = check_bytes(Path(alternate['source_path']), alternate['bytes'], alternate['sha256'])
                primary = next(row for row in rows if row['portable_path'] == selected[field]['primary'])
                require(data == Path(primary.get('source_path', primary.get('path'))).read_bytes(), 'declared owner alias exact primary binding')
                aliases.append({'kind': field, **alternate, 'primary': selected[field]['primary']})
    actual = Counter((row['family'], row['source_path'], row['source_bytes'], row['source_sha256']) for row in value['logical_sources'])
    require(actual == expected, 'complete immutable selector member multiset')
    require(value['logical_count_including_selectors'] == sum(actual.values()) == 123, 'full logical cardinality')
    require(value['logical_bytes_including_selectors'] == sum(row['source_bytes'] for row in value['logical_sources']), 'full logical bytes')

    require(value['unique_payload_files'] == len(canonical) == 107 and value['unique_payload_bytes'] == sum(row['bytes'] for row in canonical.values()) == 3112171, 'complete unique payload denominator')
    require(len({(row['bytes'], row['sha256']) for row in canonical.values()}) == len(canonical), 'content deduplication exact')
    logical_destinations = {}

def refusal(name, mutated):
    try:
        validate(mutated)
    except (ValueError, KeyError) as error:
        return {'name': name, 'status': 'REFUSED', 'reason': str(error)}
    raise ValueError('Independent admission accepted adversarial packet: ' + name)
