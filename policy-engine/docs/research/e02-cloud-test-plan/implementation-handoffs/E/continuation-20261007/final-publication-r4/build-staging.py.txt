def require(ok, text):
    if not ok:
        raise ValueError(text)

def git(repo, *argv):
    return subprocess.check_output(['git', '-C', str(repo), *argv], text=True).strip()

def digest(data):
    return hashlib.sha256(data).hexdigest()

def stream_ref(row):
    path = Path(row.get('source_path', row.get('path')))
    value = dict(row)
    value['source_path'] = str(path)
    if not path.is_file():
        value['actual_custody'] = 'MISSING (excluded original reference, not selected payload)'
        return value
    hasher = hashlib.sha256()
    with path.open('rb') as source:
        for block in iter(lambda: source.read(1024 * 1024), b''):
            hasher.update(block)
    value['actual_bytes'] = path.stat().st_size
    value['actual_sha256'] = hasher.hexdigest()
    value['actual_custody'] = 'PASS' if value['actual_bytes'] == row['bytes'] and value['actual_sha256'] == row['sha256'] else 'FAIL'
    return value

require(git(ROOT, 'rev-parse', 'HEAD^{tree}') == TREE, 'root tree changed')

records = []
selection_records = []
exclusions = []
source_destination = {}
canonical_by_content = {}
canonical = []


def admit(family, source, relative, row, role):
    require(source.is_file(), 'missing selected original: ' + str(source))
    data = source.read_bytes()
    actual = (len(data), digest(data))
    require(actual == (row['bytes'], row['sha256']), 'selected original mismatch: ' + str(source))
    relative = Path(relative)
    require(not relative.is_absolute() and '..' not in relative.parts, 'invalid portable path')
    if relative.name.endswith('.py'):
        relative = relative.with_name(relative.name + '.txt')
    proposed = Path(family) / relative
    require(not str(proposed).endswith('.py'), 'active Python proposal')
    old = source_destination.get(proposed.as_posix())
    require(old is None or old == actual, 'destination collision: ' + str(proposed))
    source_destination[proposed.as_posix()] = actual
    key = (actual[0], actual[1])
    if key in canonical_by_content:
        carrier = canonical_by_content[key]
        require((STAGE / 'payload' / carrier['portable_path']).read_bytes() == data, 'digest collision')
        alias = True
    else:
        target = STAGE / 'payload' / proposed
        target.parent.mkdir(parents=True, exist_ok=True)
        target.write_bytes(data)
        carrier = {'portable_path': proposed.as_posix(), 'bytes': actual[0], 'sha256': actual[1], 'staged_path': str(target)}
        canonical_by_content[key] = carrier
        canonical.append(carrier)
        alias = False
    record = {'family': family, 'source_path': str(source), 'source_bytes': actual[0], 'source_sha256': actual[1], 'logical_destination': proposed.as_posix(), 'canonical_destination': carrier['portable_path'], 'proposed_git_path': PREFIX + '/' + carrier['portable_path'], 'literal_alias': alias, 'role': role}
    records.append(record)

            require(Path(value['executed_source_path']).read_bytes() == Path(value['portable_source']['path']).read_bytes(), 'observer archive differs from executed source')
        exclusions.append({'family': family, 'historical_optional_not_copied': selected['historical_optional_do_not_rewrite'], 'large_ignored_refs': [stream_ref(value) for value in selected['large_ignored_refs']], 'root_owned_existing_carrier_do_not_duplicate': selected['root_owned_do_not_duplicate']})
    elif family == 'runtime-retry':

    elif family == 'criteria-review':
        rows = selected['rows']
    else:
