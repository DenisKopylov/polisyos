def identity(path):
    digest = hashlib.sha256()
    size = 0
    with path.open('rb') as handle:
        while data := handle.read(1024 * 1024):
            digest.update(data)
            size += len(data)
    return {'path': str(path), 'bytes': size, 'sha256': digest.hexdigest()}
