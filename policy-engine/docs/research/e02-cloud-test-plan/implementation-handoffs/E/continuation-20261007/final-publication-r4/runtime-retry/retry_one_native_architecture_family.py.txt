"""UNEXECUTED owner-ready bounded retry. Root generation lease required first.

Uses the canonical measurement body and complete declared-owner manifest.
One selected runtime family is not a full architecture verdict.
"""

import dataclasses
import hashlib
import json
import os
import subprocess
from pathlib import Path

from tools.devx.architecture import guardrails

ROOT = Path('/workspace/e02-E-continuation-20261006/policy-engine')
WORKSPACE = Path('/workspace/e02-E-pr38-r3-receipts/runtime-export-retry-64d7444a-20261007-r1/architecture-retained')
CACHE = Path('/workspace/.polisyos-environment/cache/uv')
FAMILY = 'runtime-openapi-snapshot'
SOURCE = '64d7444a18c55df7b88b71b7699a2f1b25ca24bd'
PREPARATION = Path('/workspace/e02-E-pr38-r3-receipts/postfreeze-import-attribution/runtime-retry-preparation/runtime-retry-preparation.json')

observed_head = subprocess.run(['git', '-C', str(ROOT), 'rev-parse', 'HEAD'], capture_output=True, text=True, check=True).stdout.strip()
if observed_head != SOURCE:
    raise RuntimeError('retry requires exact frozen source; no fallback ref')
for variable in ('OMP_NUM_THREADS', 'OPENBLAS_NUM_THREADS', 'MKL_NUM_THREADS', 'NUMEXPR_NUM_THREADS', 'VECLIB_MAXIMUM_THREADS', 'BLIS_NUM_THREADS', 'POLISYOS_PYTEST_WORKERS', 'POLISYOS_PYTEST_DIST'):
    if os.environ.get(variable) is not None:
        raise RuntimeError('artificial process/thread cap supplied: ' + variable)
prepared = json.loads(PREPARATION.read_text())
for item in prepared['source_inputs']:
    raw = Path(item['path']).read_bytes()
    if len(raw) != item['bytes'] or hashlib.sha256(raw).hexdigest() != item['sha256']:
        raise RuntimeError('canonical prepared source/input changed: ' + item['path'])

families = guardrails._parse_generated_artifacts(ROOT / 'architecture/generated_artifacts.toml')
selected = tuple(f for f in families if f.family_id == FAMILY)
if len(selected) != 1 or not guardrails._requires_default_generated_freshness(selected[0]):
    raise RuntimeError('canonical required runtime family is not uniquely admitted')
cursor = guardrails._GeneratedArtifactMeasurementCursor(selected, [], [])
try:
    violations = guardrails._measure_required_generated_artifacts(
        families,
        expected_root=ROOT,
        cursor=cursor,
        retained_workspace_root=WORKSPACE,
        uv_cache_dir=CACHE,
    )
except guardrails.GeneratedArtifactCheckUnrunError as exc:
    print(json.dumps({'scope': 'one native isolated runtime family only', 'outcome': 'UNRUN', 'reason': str(exc), 'unrun_checks': [dataclasses.asdict(v) for v in exc.unrun_checks], 'completed_violations': [dataclasses.asdict(v) for v in cursor.violations]}, indent=2))
    raise SystemExit(2) from exc
print(json.dumps({'scope': 'one native isolated runtime family only', 'outcome': 'FAIL' if violations else 'PASS', 'violations': [dataclasses.asdict(v) for v in violations], 'workspace': str(WORKSPACE)}, indent=2))
raise SystemExit(1 if violations else 0)
