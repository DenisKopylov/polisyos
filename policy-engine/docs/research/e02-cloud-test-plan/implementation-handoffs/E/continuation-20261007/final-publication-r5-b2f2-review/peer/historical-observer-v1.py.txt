BASE = 'b2f2f65ea8884a6e2ffa3fc444cdf4bbb9157641'

def require(value, label):
    if not value:
        raise ValueError(label)


def digest(body):
    return hashlib.sha256(body).hexdigest()


def ref(path):
    body = path.read_bytes()
    return {'path': str(path), 'bytes': len(body), 'sha256': digest(body)}


def canon(value):
    return json.dumps(value, ensure_ascii=False, sort_keys=True, separators=(',', ':'), allow_nan=False)


def git(*args):
    return subprocess.run(['git', '-C', str(ROOT), *args], capture_output=True, check=True).stdout


def guard():
    return {'head': git('rev-parse', 'HEAD').decode().strip(), 'tree': git('rev-parse', 'HEAD^{tree}').decode().strip(),
            'tracked_status': git('status', '--porcelain', '--untracked-files=no').decode(), 'untracked_not_whole_clean': git('status', '--porcelain', '--untracked-files=normal').decode()}


def absolute_ref(raw):
    candidates = [raw.get(key) for key in ('stored_at', 'path', 'source_path', 'source')]
    path = next((p for p in candidates if isinstance(p, str) and Path(p).is_absolute()), None)
    require(path is not None and type(raw['bytes']) is int and raw['bytes'] >= 0, 'actual_provider_complete_identity')
    return {'path': path, 'bytes': raw['bytes'], 'sha256': raw['sha256']}


def verify_body(body, identity):
    require(len(body) == identity['bytes'] and digest(body) == identity['sha256'], 'whole_selected_body_identity')


def main():
    before = guard()
    require(before['head'] == CANDIDATE and before['tree'] == TREE and before['tracked_status'] == '', 'immutable_DOC_current_Git_guard')
    require(git('rev-parse', CANDIDATE + '^').decode().strip() == BASE, 'source_parent_b2')
    require(git('rev-parse', BASE + '^{tree}').decode().strip() == BASE_TREE, 'source_b2_tree')
    raw_manifest = (STAGE / 'manifest.json').read_bytes()
    require(len(raw_manifest) == 1947707 and digest(raw_manifest) == '7aa3af272de3ad29bc2c1d62f69f8c24db206c7d351fbb46715b0e06e33de257', 'manifest_fixed_pin')

    qualification = json.loads((STAGE / 'qualification.json').read_bytes())

    changes = git('diff', '--name-status', '-z', BASE, CANDIDATE).decode().split('\0')
    pairs = [(changes[i], changes[i + 1]) for i in range(0, len(changes) - 1, 2)]
    require(len(pairs) == 698 and all(s == 'A' and p.startswith(PREFIX + '/') and not p.endswith('.py') for s, p in pairs), 'full698_DOC_only_Git_delta')
    changed_paths = {p.removeprefix(PREFIX + '/') for _, p in pairs}
    require(set(handoff['changed_footprint']) == changed_paths and len(handoff['changed_footprint']) == 698, 'handoff_complete698_footprint')
    require(handoff['active_Python_paths'] == 0, 'no_active_Python')
    entries = []
    for raw in git('ls-tree', '-r', '-z', CANDIDATE, '--', PREFIX).split(b'\0'):
        if not raw:
            continue
        header, path = raw.split(b'\t', 1)
        mode, kind, oid = header.decode().split(' ')
        relative = path.decode().removeprefix(PREFIX + '/')
        require(kind == 'blob' and mode == '100644', 'ordinary_passive_Git_blob_mode')
        entries.append({'relative': relative, 'git_object': oid, 'mode': mode})
    require(len(entries) == 698 and {r['relative'] for r in entries} == changed_paths, 'complete_Git_subtree_set')
    carriers = {r['destination']: r for r in manifest['canonical_copies']}
    require(len(carriers) == 688, 'all688_distinct_carriers')
    metadata = {f.name for f in STAGE.iterdir() if f.is_file()}
    require(len(metadata) == 10 and changed_paths == set(carriers) | metadata, '698_equals688plus10')
    process = subprocess.Popen(['git', '-C', str(ROOT), 'cat-file', '--batch'], stdin=subprocess.PIPE, stdout=subprocess.PIPE, stderr=subprocess.PIPE)
    process.stdin.write(''.join(r['git_object'] + '\n' for r in entries).encode())
    process.stdin.close()

    for entry in entries:
        header = process.stdout.readline().decode().strip().split(' ')
        require(len(header) == 3 and header[0] == entry['git_object'] and header[1] == 'blob', 'Git_batch_header')
        body = process.stdout.read(int(header[2]))
        require(process.stdout.read(1) == b'\n', 'Git_blob_terminator')
        relative = entry['relative']
        stage_path = STAGE / 'payload' / relative if relative in carriers else STAGE / relative
        require(body == stage_path.read_bytes(), 'Git_blob_equals_closed_stage')
        require(body == (ROOT / PREFIX / relative).read_bytes(), 'Git_blob_equals_current_checkout')
        if relative in carriers:
            verify_body(body, carriers[relative]['stored'])
        if relative.endswith('.json'):
            json.loads(body)
        rows.append({**entry, 'bytes': len(body), 'sha256': digest(body)})
    require(process.stdout.read() == b'' and process.wait() == 0 and process.stderr.read() == b'', 'complete_closed_Git_batch')
    total_bytes = sum(r['bytes'] for r in rows)
    require(total_bytes == 43540949, 'all698_literal_Git_total43540949')

    actual_originals = {}

        provider = json.loads(provider_body)
        supplied = provider[spec['member_class']]
        require(len(supplied) == spec['members'], 'whole_provider_member_denominator')
        for raw in supplied:
            expected[(spec['family'], spec['path'], spec['member_class'], canon(raw))] += 1
        expected[(spec['family'], spec['path'], spec['input_kind'], canon(None))] += 1
        providers.append({'family': spec['family'], 'selector': ref(provider_path), 'actual_members': len(supplied), 'schema': provider.get('schema')})
    observed = Counter((r['family'], r['selection_source_path'], r['member_class'], canon(r['selected_record'])) for r in manifest['logical_sources'])
    require(expected == observed and sum(expected.values()) == 1167 and len(providers) == 28, 'independent1167_provider_row_Counter')
    source_rows = []
    for row in manifest['logical_sources']:
        stored_body = (STAGE / 'payload' / row['canonical_destination']).read_bytes()
        verify_body(stored_body, row['stored'])
        require(Path(row['stored']['path']).read_bytes() == stored_body, 'literal_stored_original_available')
        if row['storage_mode'] == 'gzip_literal_archive':
            original_body = gzip.decompress(stored_body)
        else:
            require(row['storage_mode'] == 'literal', 'declared_storage_mode')
            original_body = stored_body
        verify_body(original_body, row['original'])
        require(Path(row['original']['path']).read_bytes() == original_body, 'whole_decoded_original_available_and_equal')
        source_rows.append({'family': row['family'], 'storage_mode': row['storage_mode'], 'original_bytes': len(original_body), 'stored_bytes': len(stored_body)})
    require(sum(r['original_bytes'] for r in source_rows) == manifest['original_logical_bytes'] == 173281765, 'logical_original173281765')
    require(sum(r['stored_bytes'] for r in source_rows) == manifest['stored_logical_bytes'] == 85641576, 'logical_stored85641576')
    require(sum(r['stored']['bytes'] for r in carriers.values()) == manifest['unique_payload_bytes'] == 41151229, 'unique41151229_payload')

    require(qualification['parent_stages']['combined'] == {'total': 39, 'PASS': 1, 'FAIL': 2, 'UNRUN': 36}, 'all39_stage_statuses_retained')

    require(qualification['findings']['formal_G_acceptance'] is False and qualification['findings']['formal_ledger_change'] is False and qualification['findings']['check_status_is_not_finding_outcome'] is True, 'no_finding_authority_promotion')
    require(qualification['future_plan_patch_applied'] is False and qualification['draft_PR_bodies_published_by_carrier'] is False and qualification['source_native_numeric_or_global_gate_reexecuted_by_carrier'] is False, 'unapplied_future_and_no_native_execution')
    require(qualification['source_frames']['different_algorithms_and_input_formats'] is True and qualification['source_frames']['equality_or_drift_shortcut'] is False, 'metadata_vs_raw_frame_distinction')
    require(qualification['environment']['package_METADATA'] == 207 and qualification['environment']['protected_control_refs'] == 3 and qualification['environment']['full_binary_environment_identity'] == 'not_established', 'environment_scoped_custody')
    enums = {'recomputed', 'independently_reconciled', 'consumer_asserted', 'institutionally_supplied', 'not_established'}

            'manifest': ref(STAGE / 'manifest.json'), 'copier': ref(STAGE / 'copy-final-publication.py.txt'), 'qualification': qualification,
