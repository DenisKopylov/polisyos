def need(ok, why):
    if not ok:
        raise ValueError(why)

def sha(data):
    return hashlib.sha256(data).hexdigest()

def ref(path):
    p = Path(path)
    b = p.read_bytes()
    return {'path': str(p), 'bytes': len(b), 'sha256': sha(b)}

def read(path):
    return json.loads(Path(path).read_text())

def state():
    def cmd(args):
        p = subprocess.run(['git', '-C', str(ROOT), *args], capture_output=True, text=True)
        need(p.returncode == 0, 'Git source read failed')
        return p.stdout.strip()
    return {'source_sha': cmd(['rev-parse', 'HEAD']), 'source_tree': cmd(['rev-parse', 'HEAD^{tree}']), 'tracked_status': cmd(['status', '--porcelain=v1', '--untracked-files=no'])}

def typed_equal(a, b):
    if type(a) is not type(b):
        return False
    if isinstance(a, dict):
        return a.keys() == b.keys() and all(typed_equal(a[k], b[k]) for k in a)
    if isinstance(a, list):
        return len(a) == len(b) and all(typed_equal(x, y) for x, y in zip(a, b))
    return a == b

def case_key(c):
    return tuple((k, c.get(k)) for k in ['job_name', 'classname', 'name', 'file', 'seconds', 'outcome'])

def counts(cases):
    c = collections.Counter(r['outcome'] for r in cases)
    return {'cases': len(cases), 'passed': c['passed'], 'failed': c['failed'], 'errors': c['error'], 'skipped': c['skipped']}

def parse_xml(path, job):
    document = ET.parse(path).getroot()
    cases, failures = [], []
    for tc in document.iter('testcase'):
        failure, error, skip = tc.find('failure'), tc.find('error'), tc.find('skipped')
        status = 'error' if error is not None else 'failed' if failure is not None else 'skipped' if skip is not None else 'passed'
        row = {'name': tc.get('name', ''), 'classname': tc.get('classname', ''), 'file': tc.get('file'), 'seconds': tc.get('time', '0'), 'outcome': status, 'job_name': job}
        cases.append(row)
        node = error if error is not None else failure
        if node is not None:
            failures.append({'job': job, 'name': row['name'], 'classname': row['classname'], 'file': row['file'], 'kind': 'error' if error is not None else 'failure', 'message': node.get('message', ''), 'complete_trace': node.text or ''})
    summaries = [dict(s.attrib) for s in document.iter('testsuite')]
    need(sum(int(s.get('tests', 0)) for s in summaries) == len(cases), 'complete JUnit denominator mismatch')
    return cases, failures

def admit_claim(actual, expected):
    need(typed_equal(actual, expected), 'complete typed current numeric/source/owner claim admission')
