def function_hashes(data):
 tree=ast.parse(data);out={}
 def walk(body,parent=''):
  for node in body:
   if isinstance(node,(ast.FunctionDef,ast.AsyncFunctionDef,ast.ClassDef)):
    name=parent+node.name
    # Named lexical body identity, annotations/defaults/decorators included; import runtime is separate.
    out[name]={'ast_sha256':hashlib.sha256(ast.dump(node,include_attributes=False).encode()).hexdigest(),'line':node.lineno}
    if isinstance(node,ast.ClassDef):walk(node.body,name+'.')
 walk(tree.body);return out

def imports(data):
 rows=[]
 for n in ast.walk(ast.parse(data)):
  if isinstance(n,ast.ImportFrom) and n.module and n.module.startswith('polisyos'):
   rows.append({'line':n.lineno,'module':n.module,'names':[a.name for a in n.names]})
  elif isinstance(n,ast.Import):
   rows.extend({'line':n.lineno,'module':a.name,'names':[]} for a in n.names if a.name.startswith('polisyos'))
 return rows
