def probe(name,argv,healthy):
    stdout=io.StringIO();stderr=io.StringIO();calls={'git':0,'mkdir':0}
    def no_git(*args,**kwargs):
        calls['git']+=1
        raise AssertionError('a prepared plan must not execute Git')
    def no_mkdir(*args,**kwargs):
        calls['mkdir']+=1
        raise AssertionError('a prepared plan must not write output directories')
    error=None;code=0
    with patch.object(sys,'argv',[str(SCRIPT),*argv]),patch.object(subprocess,'check_output',no_git),patch.object(Path,'mkdir',no_mkdir),contextlib.redirect_stdout(stdout),contextlib.redirect_stderr(stderr):
        try:
            runpy.run_path(str(SCRIPT),run_name='__main__')
        except SystemExit as exc:
            code=exc.code or 0
        except Exception as exc:
            error={'type':type(exc).__name__,'message':str(exc)}
            code=1
    row={'name':name,'argv':[str(SCRIPT),*argv],'exit_code':code,'error':error,'stdout':stdout.getvalue(),'stderr':stderr.getvalue(),'callbacks':calls}
    assert calls=={'git':0,'mkdir':0},row
    if healthy:
        assert code==0 and error is None,row
        observation=json.loads(stdout.getvalue())
        assert observation['check_status']=='UNRUN' and observation['source_native_gate_executed'] is False,row
    else:
        assert code!=0,row
    assert not OUTPUT.exists()
    row['outcome']='PASS'
    return row
