"""Preserve complete closed moderate inputs/outputs without copying the large raw graph."""
import gzip
import hashlib
import json

OUT.mkdir()
(OUT / 'assets').mkdir()
logical = []
storage = {}

def sha(data):
    return hashlib.sha256(data).hexdigest()

def add(name, path):
    data = path.read_bytes()
    digest = sha(data)
    compressed = len(data) > 180000
    try:
        data.decode('utf-8')
        utf8 = True
    except UnicodeDecodeError:
        utf8 = False
    encoding = 'gzip' if compressed else 'literal'
    encoded = gzip.compress(data, mtime=0) if compressed else data
    key = (digest, encoding)
    if key not in storage:
        portable = 'assets/' + digest + ('.bin' if compressed or not utf8 else '.txt')
        with (OUT / portable).open('xb') as stream:
            stream.write(encoded)
        storage[key] = {'portable_path': portable, 'bytes': len(encoded), 'sha256': sha(encoded),
                        'encoding': encoding, 'decoded_bytes': len(data), 'decoded_sha256': digest}
    logical.append({'logical_name': name, 'original_path': str(path), **storage[key]})
