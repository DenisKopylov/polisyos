def require(value, label):
    if not value:
        raise ValueError(label)


def ref(path):
    data = path.read_bytes()
    return {'path': str(path), 'bytes': len(data), 'sha256': hashlib.sha256(data).hexdigest()}


def guard():
    def git(*args):
        return subprocess.run(['git', '-C', str(ROOT), *args], capture_output=True, text=True, check=True).stdout
    return {'source': git('rev-parse', 'HEAD').strip(), 'tree': git('rev-parse', 'HEAD^{tree}').strip(), 'tracked_status': git('status', '--porcelain', '--untracked-files=no')}


def load_module(path):
    loader = importlib.machinery.SourceFileLoader('_independent_static119', str(path))
    spec = importlib.util.spec_from_loader(loader.name, loader)
    module = importlib.util.module_from_spec(spec)
    loader.exec_module(module)
    return module


def literal_selection(path, pin):
    require(ref(path)['sha256'] == pin, 'closed_selector_pin')
    selection = json.loads(path.read_bytes())
    require(selection['source'] == SOURCE and selection['tree'] == TREE and selection['finding_closure'] is False, 'literal_source_claim')
    unique = {}
    originals = []
    for row in selection['assets']:
        blob = (path.parent / row['portable_path']).read_bytes()
        require(type(row['bytes']) is int and len(blob) == row['bytes'] and hashlib.sha256(blob).hexdigest() == row['sha256'], 'stored_byte_identity')
        data = gzip.decompress(blob) if row['encoding'] == 'gzip' else blob
        require(len(data) == row['decoded_bytes'] and hashlib.sha256(data).hexdigest() == row['decoded_sha256'], 'whole_decoded_byte_identity')
        require(Path(row['original_path']).read_bytes() == data, 'literal_original_identity')
        unique[row['portable_path']] = row['bytes']
        originals.append({key: row[key] for key in ('logical_name', 'original_path', 'portable_path', 'bytes', 'sha256', 'decoded_bytes', 'decoded_sha256')})
    require(selection['logical_count'] == len(originals) and selection['unique_count'] == len(unique) and selection['unique_bytes'] == sum(unique.values()), 'complete_selector_Counter')
    return {'selector': ref(path), 'logical': len(originals), 'unique': len(unique), 'stored_bytes': sum(unique.values()), 'members': originals}


def functions(path):
    return {n.name: n for n in ast.parse(path.read_text()).body if isinstance(n, ast.FunctionDef)}


def main():
    before = guard()
    require(before == {'source': SOURCE, 'tree': TREE, 'tracked_status': ''}, 'Root_frozen_source')
    require(ref(V3)['sha256'] == '8d4009b445e41a23313ca28a50bfe9ac7e9059b7216ff7a1276cfd7e4e0214f6', 'V3_program_pin')
    old, current = functions(V2), functions(V3)
    reused = []
    for name, expected in old.items():
        node = copy.deepcopy(current[name])
        if name == 'validate':
            call = node.body.pop(0)
            require(ast.dump(call) == ast.dump(ast.parse('admit_ancillary_claims(x)').body[0]), 'one_new_admission_inlet')
        require(ast.dump(node) == ast.dump(expected), 'unchanged_derived_control_body:' + name)
        reused.append(name)
    module = load_module(V3)
    original = json.loads(module.MAP.read_bytes())
    healthy = module.validate(original)
    records = json.loads(module.RECORDS.read_bytes())
    cases = json.loads(Path('/tmp/e02-E-final-wave-numeric-independent-b2-20261007/case-index.json').read_bytes())['cases']
    rawrows = {(r['symbol'], r['check_class']): r for r in records['rows']}
    actualtokens = Counter(json.dumps(c, sort_keys=True) for c in cases)
    require(len(original['rows']) == len(rawrows) == 119, 'full119_annotations')
    require(Counter((r['symbol'], r['check_class']) for r in original['rows']) == Counter(rawrows), '119_native_symbol_class_multiset')
    named_rows = role_rows = named_case_bindings = 0
    for row in original['rows']:
        expected = rawrows[(row['symbol'], row['check_class'])]
        require(row['current_writer'] == expected['current_writer'] and row['source_input'] == expected['source_input'], 'complete_native_writer_source_identity')
        require(row['native_static_status'] == expected['native_mechanism_record']['status'], 'static_status_not_promoted')
        require(row['measurement_limit'] and row['remaining_boundary']['owner'] and row['remaining_boundary']['next_result'], 'precise_unmeasured_owner_boundary')
        require('not' in row['measurement_limit'] or 'No new' in row['measurement_limit'], 'annotation_does_not_claim_default_dispatch')
        named_rows += bool(row['current_executed_named_test_callers'])
        role_rows += bool(row['qualified_accepted_role_native_receipt'])
        for binding in row['current_executed_named_test_callers']:
            require(actualtokens[json.dumps(binding['case'], sort_keys=True)] > 0, 'named_case_exists_in_prior_independent_case_index')
            require(set(binding['AST_caller_records']) <= set(row['actual_AST_test_callers']), 'named_AST_association')
            named_case_bindings += 1
    require(named_rows == 48 and role_rows == 35 and original['source_role_actual_Git_equal_checks'] == 13, 'qualified_association_denominators')
    require(original['static_actual'] == {'regressions': 24, 'new_unresolved': 95, 'rows': 119, 'E': 91, 'B': 28, 'native_outcome': 'FAIL', 'coverage': 'partial'}, 'static119_counts_separate_from_execution')
    require(original['measured_default_production_dispatch'] is False and original['finding_closure'] is False and original['new_native_execution'] is False and original['P41'] == 'not_established', 'no_surface_or_finding_authority')
    selectors = [literal_selection(Path('/tmp/e02-E-final3-witness-supplement-b2f2f65e-20261007/publication-selection.json'), 'd99833380c2f3c39bd14cd712afc3ef8c9ca3a714345e2b1c0bdde9f288ecaee'),
                 literal_selection(Path('/tmp/e02-E-final3-witness-claim-correction-b2f2f65e-20261007/publication-selection.json'), 'd43afe828cb30840f7e3ce2c824a700c44a0a60e57544c0d437ffcdac7eb624e')]
    controls = [{'case': 'healthy_full119', 'state': 'GO', 'result': healthy}]
    mutants = []
    changes = [(['schema'], 'unknown'), (['source_role_actual_Git_equal_checks'], True),
               (['case_groups', 'PCL', 'scope'], 'Full served verification and branch coverage'),
               (['rows', 0, 'remaining_boundary', 'owner'], 'Invented authority'),
               (['rows', 0, 'measurement_limit'], 'Every branch and served consumer measured'),
               (['rows', 0, 'actual_closed_case_group'], 'B_CONTROL'),
               (['rows', 0, 'writer_basis'], {'writer': 'A', 'basis': 'invented'})]
    for path, value in changes:
        candidate = copy.deepcopy(original)
        module.setpath(candidate, path, value)
        mutants.append(candidate)
    candidate = copy.deepcopy(original)
    group = candidate['case_groups']['PCL']
    group['actual_cases'].append(copy.deepcopy(group['actual_cases'][0]))
    group['actual_case_count'] = len(group['actual_cases'])
    group['actual_outcomes'] = dict(Counter(c['outcome'] for c in group['actual_cases']))
    mutants.append(candidate)
    for index, candidate in enumerate(mutants):
        try:
            module.validate(candidate)
        except ValueError as exc:
            controls.append({'case': 'complete_semantic_attachment_' + str(index), 'state': 'REFUSED', 'reason': str(exc)})
        else:
            raise RuntimeError('attachment_escape')
    tree = ast.parse(V3.read_text())
    fn = next(n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == 'admit_ancillary_claims')
    hits = 0
    for index, statement in enumerate(fn.body):
        if isinstance(statement, ast.If) and 'typed_equal' in ast.unparse(statement.test):
            fn.body[index] = ast.Pass()
            hits += 1
    require(hits == 1, 'actual_one_ancillary_typed_predicate')
    ns = {'__name__': '_peer_ancillary_guard_removed', '__file__': str(V3)}
    exec(compile(ast.fix_missing_locations(tree), str(V3) + ':actual_typed_If_removed', 'exec'), ns)
    for index, candidate in enumerate(mutants):
        result = ns['validate'](candidate)
        controls.append({'case': 'actual_ancillary_predicate_removed_' + str(index), 'state': 'ADMITTED_PROPERTY_FAIL', 'trusted_source_map_SHA_guard_remains': True, 'result': result})
    after = guard()
    require(before == after, 'Root_read_only_guard_equal')
    return {'schema': 'policyos.e02.independent.static119.annotation.v1', 'source': SOURCE, 'tree': TREE,
            'disposition': 'GO literal complete119 source/case annotation and complete typed semantic ancillary admission; no default-dispatch or finding authority',
            'source_guard_before': before, 'source_guard_after': after, 'V3': ref(V3), 'unchanged_V2_functions_reused': reused,
            'actual119': {'rows': 119, 'current_named_test_association_rows': named_rows, 'accepted_role_association_rows': role_rows, 'named_case_bindings': named_case_bindings, 'independently_previously_verified_roles': 13},
            'selector_custody': selectors, 'controls': controls, 'optimization_mode': not __debug__,
            'native_tests_or_gates_repeated': False, 'P37': {'literal_source_case_conservation': 'independently_reconciled', 'typed_and_Counter_controls': 'recomputed', 'served_branches_and_finding_authority': 'not_established'},
            'limits': ['Static unresolved/uninvoked statuses remain FAIL/partial proxy observations.', 'Named executed-test associations do not prove branches or production dispatch.', '13 prior independent accepted current source-role checks reused; no duplicate source/math review.']}


if __name__ == '__main__':
    print(json.dumps(main(), indent=2, sort_keys=True))
