def require(condition: bool, message: str) -> None:
    if not condition:
        raise ValueError(message)


def digest(body: bytes) -> str:
    return hashlib.sha256(body).hexdigest()


def reference(path: Path) -> dict[str, Any]:
    body = path.read_bytes()
    return {"path": str(path), "bytes": len(body), "sha256": digest(body)}


def same(left: object, right: object) -> bool:
    """JSON strict equality: True, 1 and 1.0 are distinct inputs."""
    if type(left) is not type(right):
        return False
    if isinstance(left, dict):
        return left.keys() == right.keys() and all(same(left[k], right[k]) for k in left)
    if isinstance(left, list):
        return len(left) == len(right) and all(same(a, b) for a, b in zip(left, right, strict=True))
    return left == right


def admit_sha(value: object) -> None:
    require(
        type(value) is str and re.fullmatch(r"[0-9a-f]{40}", value) is not None,
        "full exact SHA required",
    )


def admit_repo_path(value: object) -> None:
    require(type(value) is str and value.startswith("policy-engine/"), "repo path prefix")
    require("\x00" not in value and not any(c.isspace() for c in value), "repo path characters")
    parts = PurePosixPath(value).parts
    require(
        ".." not in parts and "." not in parts and str(PurePosixPath(value)) == value,
        "canonical repo path",
    )


def admit_inputs(packet: dict[str, Any]) -> None:
    require(packet["candidate"] == SOURCE and packet["tree"] == TREE, "immutable source identity")
    require(
        type(packet["inputs"]) is dict and len(packet["inputs"]) == 17,
        "complete fixed input denominator",
    )
    for ref in packet["inputs"].values():
        require(type(ref) is dict and set(ref) == {"path", "bytes", "sha256"}, "typed input ref")
        require(type(ref["path"]) is str and Path(ref["path"]).is_absolute(), "absolute input ref")
        require("\x00" not in ref["path"], "input path NUL")
        require(type(ref["bytes"]) is int and ref["bytes"] >= 0, "input size type")
        require(
            type(ref["sha256"]) is str and re.fullmatch(r"[0-9a-f]{64}", ref["sha256"]) is not None,
            "input digest type",
        )
    # Whole packet preflight precedes any Git callback, then exact physical bytes.
    for ref in packet["inputs"].values():
        require(same(reference(Path(ref["path"])), ref), "input byte drift: " + ref["path"])
    for _role, path, revision, _receipt in ROLES:
        admit_sha(revision)
        admit_repo_path(PREFIX + path)


def git(*args: str) -> bytes:
    allowed = {"show", "rev-parse", "diff", "status"}
    require(args and args[0] in allowed, "read-only Git opcode")
    result = subprocess.run(  # noqa: S603 -- fixed absolute Git, admitted read-only opcode
        ["/usr/bin/git", "-C", str(ROOT), *args],
        check=False,
        capture_output=True,
        shell=False,
    )
    require(
        result.returncode == 0, "read-only Git failed: " + result.stderr.decode(errors="replace")
    )
    return result.stdout


def blob(revision: str, path: str) -> bytes:
    admit_sha(revision)
    admit_repo_path(path)
    return git("show", revision + ":" + path)


def record_blob(revision: str, path: str) -> dict[str, Any]:
    body = blob(revision, path)
    return {"revision": revision, "path": path, "bytes": len(body), "sha256": digest(body)}


def load_inputs(path: Path) -> tuple[dict[str, Any], dict[str, Any]]:
    raw = path.read_bytes()
    require(digest(raw) == INPUT_HASH, "immutable input manifest hash")
    packet = json.loads(raw)
    admit_inputs(packet)
    values = {
        key: json.loads(Path(ref["path"]).read_bytes()) for key, ref in packet["inputs"].items()
    }
    return packet, values


def source_bindings(refs: dict[str, Any]) -> list[dict[str, Any]]:
    result = []
    for role, suffix, accepted_source, evidence in ROLES:
        path = PREFIX + suffix
        current = record_blob(SOURCE, path)
        accepted = record_blob(accepted_source, path)
        equality = current["sha256"] == accepted["sha256"] and current["bytes"] == accepted["bytes"]
        require(equality, "unqualified changed role: " + role)
        require((ROOT / path).read_bytes() == blob(SOURCE, path), "physical source drift: " + path)
        result.append(
            {
                "role": role,
                "current_source": current,
                "accepted_source": accepted,
                "body_equal": equality,
                "accepted_evidence": refs[evidence],
                "basis": "recomputed",
                "qualification": (
                    "This role body equals its named independently accepted source; no "
                    "blanket prior six-file equality or new numeric execution is "
                    "asserted."
                ),
            }
        )
    return result


def original_bindings(owner: dict[str, Any]) -> dict[str, Any]:
    finding_ref = record_blob(SOURCE, TSV + "finding-owners.tsv")
    bundle_ref = record_blob(SOURCE, TSV + "bundle-owners.tsv")
    findings = {
        row["finding_id"]: row
        for row in csv.DictReader(
            io.StringIO(blob(SOURCE, finding_ref["path"]).decode()), delimiter="\t"
        )
        if row["unit"] == "E"
    }
    bundles = {
        row["bundle_id"]: row
        for row in csv.DictReader(
            io.StringIO(blob(SOURCE, bundle_ref["path"]).decode()), delimiter="\t"
        )
        if row["unit"] == "E"
    }
    require(set(findings) == {row["id"] for row in owner["rows"]}, "full owner finding membership")
    require(set(bundles) == {row["id"] for row in owner["bundles"]}, "full owner bundle membership")
    cards = []
    fragments = []
    cache: dict[str, bytes] = {}
    for bundle in owner["bundles"]:
        cards.append({"bundle_id": bundle["id"], **record_blob(SOURCE, bundle["criterion_card"])})
    for row in owner["rows"]:
        require(
            same(row["canonical_ownership"], findings[row["id"]]), "ownership drift: " + row["id"]
        )
        for ref in row["criterion_evidence"]["original_criterion_refs"]:
            path = ref["path"]
            admit_repo_path(path)
            lines = ref["lines"]
            require(
                type(lines) is list
                and len(lines) == 2
                and all(type(v) is int and v > 0 for v in lines),
                "criterion line types",
            )
            raw = cache.setdefault(path, blob(SOURCE, path))
            selected = b"".join(raw.splitlines(keepends=True)[lines[0] - 1 : lines[1]])
            require(digest(selected) == ref["sha256"], "original criterion changed: " + row["id"])
            fragments.append(
                {"finding_id": row["id"], "revision": SOURCE, **ref, "bytes": len(selected)}
            )
    return {
        "finding_owners": finding_ref,
        "bundle_owners": bundle_ref,
        "cards": cards,
        "criterion_fragments": fragments,
    }

def main() -> None:

    parser.add_argument("--inputs", type=Path, required=True)
    parser.add_argument("--output", type=Path, required=True)
    parser.add_argument("--validate", type=Path)
    args = parser.parse_args()

    packet = build(inputs, values)

    )
