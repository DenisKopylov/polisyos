def require(value: bool, message: str) -> None:
    if not value:
        raise ValueError(message)


def ref(path: Path) -> dict[str, Any]:
    body = path.read_bytes()
    return {"path": str(path), "bytes": len(body), "sha256": hashlib.sha256(body).hexdigest()}


def change(original: dict[str, Any], path: list[str | int], value: object) -> dict[str, Any]:
    result = copy.deepcopy(original)
    target: Any = result
    for key in path[:-1]:
        target = target[key]
    require(
        path[-1] in target if isinstance(target, dict) else path[-1] < len(target),
        "existing mutation field",
    )
    target[path[-1]] = value
    return result
