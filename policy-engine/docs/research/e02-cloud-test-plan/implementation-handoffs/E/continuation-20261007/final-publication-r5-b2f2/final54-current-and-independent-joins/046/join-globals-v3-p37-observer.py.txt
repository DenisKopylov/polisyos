def require(value: bool, message: str) -> None:
    if not value:
        raise ValueError(message)


def ref(path: Path) -> dict[str, Any]:
    body = path.read_bytes()
    return {"path": str(path), "bytes": len(body), "sha256": hashlib.sha256(body).hexdigest()}


def read_ref(row: dict[str, Any]) -> dict[str, Any]:
    require(type(row) is dict, "typed input reference")
    require(type(row.get("bytes")) is int and row["bytes"] >= 0, "integer reference size")
    require(
        type(row.get("path")) is str and Path(row["path"]).is_absolute(),
        "absolute existing receipt path",
    )
    require(type(row.get("sha256")) is str and len(row["sha256"]) == 64, "reference SHA256")
    require(
        ref(Path(row["path"])) == {key: row[key] for key in ("path", "bytes", "sha256")},
        "complete byte custody",
    )
    return json.loads(Path(row["path"]).read_bytes())


def check_output(row: dict[str, Any]) -> None:
    require(
        type(row.get("bytes")) is int and row["bytes"] >= 0, "typed complete stdout/stderr size"
    )
    require(ref(Path(row["path"])) == row, "complete literal stdout/stderr custody")


def native_gate(
    name: str, row: dict[str, Any], same: Callable[[object, object], bool]
) -> dict[str, Any]:
    raw = read_ref(row["receipt"])
    require(
        raw["source"] == raw["head_after"] == SOURCE and raw["tree"] == TREE,
        "current native gate identity",
    )
    require(
        raw["source_immutable"] is True and same(raw["source_before"], raw["source_after"]),
        "native source guard",
    )
    require(raw["ROOT_environment_metadata_unchanged"] is True, "native ROOT metadata guard")
    require(
        same(raw["ROOT_environment_metadata_before"], raw["ROOT_environment_metadata_after"]),
        "all saved native environment records unchanged",
    )
    before = raw["ROOT_environment_metadata_before"]
    metadata_rows = before.get("combined_rows", before.get("metadata_rows"))
    require(type(metadata_rows) is list and len(metadata_rows) == 210, "210 complete guard rows")
    package_rows = [item for item in metadata_rows if item["path"].endswith(".dist-info/METADATA")]
    controls = [item for item in metadata_rows if not item["path"].endswith(".dist-info/METADATA")]
    require(
        len(package_rows) == 207
        and {item["path"] for item in controls}
        == {"pyvenv.cfg", "bin/python", "bin/polisyos-tools"},
        "207 package plus3 explicit protected controls",
    )
    require(type(raw["exit_code"]) is int and raw["exit_code"] == 1, "literal nonzero native exit")
    require(raw["outcome"] == raw["canonical_selector_outcome"] == "FAIL", "native terminal FAIL")
    require(raw["finding_closure"] is False, "native check is not finding closure")
    for key in ("stdout", "stderr"):
        check_output(raw[key])
    for key in ("command", "cwd", "source_before", "source_after", "outcome", "exit_code"):
        if key in row:
            require(same(raw[key], row[key]), "native index/raw field: " + key)
    return {
        "selector": name,
        "check_outcome": raw["outcome"],
        "exit_code": raw["exit_code"],
        "execution_state": "terminal",
        "command": raw["command"],
        "cwd": raw["cwd"],
        "receipt": row["receipt"],
        "stdout": raw["stdout"],
        "stderr": raw["stderr"],
        "source": SOURCE,
        "tree": TREE,
        "source_frame": raw["source_before"],
        "source_immutable": raw["source_immutable"],
        "environment": {
            "package_METADATA_count": 207,
            "protected_control_count": 3,
            "saved_before_after_equal": True,
            "whole_binary_identity": raw["ROOT_environment_full_binary_byte_guard"],
        },
        "P41": "not_established",
        "finding_closure": False,
    }


def admit_complete_claims(
    observed: dict[str, Any], expected: dict[str, Any], same: Callable[[object, object], bool]
) -> None:
    require(
        same(observed, expected),
        "whole typed current54/cases/7gate/39stage/recipe/source/owner claims against fixed inputs",
    )


def admit_boundary(packet: dict[str, Any]) -> None:
    require(packet["candidate"] == SOURCE and packet["tree"] == TREE, "final source identity")
    require(
        packet["formal_G_acceptance"] is False and packet["formal_ledger_change"] is False,
        "owner proposals are separate from G intake",
    )
    require(
        packet["current_numeric_wave"]["counts"]
        == {"cases": 1465, "passed": 1457, "failed": 8, "errors": 0, "skipped": 0},
        "unchanged complete native case counts",
    )
    require(packet["current_global_gates"]["check_outcome"] == "FAIL", "no full gate PASS")

    args = parser.parse_args()
    result, _module = build()
    args.output.write_text(json.dumps(result, ensure_ascii=False, indent=2) + "\n")
    sys.stdout.write(
        json.dumps(
            {
                "check_outcome": "PASS",
                "metadata_scope": "current54/source/case/gate metadata; no native test executed",
                "output": ref(args.output),
                "numeric": result["current_numeric_wave"]["counts"],
                "globals": result["current_global_gates"]["gate_outcome_counts"],
                "stages": result["current_global_gates"]["parent_stage_counts"],
                "E_proposals": result["bounded_E_proposals"],
                "G_acceptance": False,
            }
        )
        + "\n"
    )


if __name__ == "__main__":
