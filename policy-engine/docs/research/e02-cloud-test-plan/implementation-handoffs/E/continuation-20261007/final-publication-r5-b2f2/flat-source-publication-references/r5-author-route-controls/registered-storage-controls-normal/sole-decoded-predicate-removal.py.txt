def registered_expand(entry, document, lookup=None):
    schema = 'policyos.e02.final3.moderate_literal_selection.v1'
    if entry.get('member_schema') != schema:
        return original_expand(entry, document, lookup)
    if document.get('schema') != schema or entry['member_class'] != 'assets':
        raise ValueError('registered exact moderate selector schema')
    raw_rows = document.get('assets')
    if not isinstance(raw_rows, list) or len(raw_rows) != entry['members']:
        raise ValueError('complete registered encoded member denominator')
    admitted = []
    for raw in raw_rows:
        if not isinstance(raw, dict) or set(raw) != {'logical_name', 'original_path', 'portable_path', 'bytes', 'sha256', 'encoding', 'decoded_bytes', 'decoded_sha256'}:
            raise ValueError('whole registered encoded member shape')
        route = raw['portable_path']
        if not isinstance(route, str) or not route or Path(route).is_absolute() or any((part in ('', '.', '..') for part in route.split('/'))) or ('\\' in route) or any((ord(c) < 32 for c in route)):
            raise ValueError('pure full relative encoded source context')
        logical_name = raw['logical_name']
        if not isinstance(logical_name, str) or not logical_name or Path(logical_name).is_absolute() or any((part in ('', '.', '..') for part in logical_name.split('/'))) or ('\\' in logical_name) or any((ord(c) < 32 for c in logical_name)):
            raise ValueError('pure full relative logical member context')
        if raw['encoding'] not in ('literal', 'gzip'):
            raise ValueError('unsupported registered evidence encoding')
        original = mechanism.identity({'path': raw['original_path'], 'bytes': raw['decoded_bytes'], 'sha256': raw['decoded_sha256']})
        if '\\' in original['path'] or any((ord(c) < 32 for c in original['path'])) or any((part in ('.', '..') for part in original['path'].split('/'))):
            raise ValueError('pure full absolute original source context')
        stored = mechanism.identity({'path': str(Path(entry['path']).parent / route), 'bytes': raw['bytes'], 'sha256': raw['sha256']})
        admitted.append((raw, route, original, stored))
    result = []
    for raw, route, original, stored in admitted:
        body = mechanism.admit_body(stored, lookup)
        decoded = body if raw['encoding'] == 'literal' else mechanism.gzip.decompress(body)
        if lookup is None:
            mechanism.admit_body(original, None)
        result.append({'family': entry['family'], 'selection_source_path': entry['path'], 'member_class': 'assets', 'original': original, 'stored': stored, 'storage_mode': 'literal' if raw['encoding'] == 'literal' else 'gzip_literal_archive', 'logical_destination': finite_storage_route(entry['family'], route, body), 'selected_record': raw})
    return result
