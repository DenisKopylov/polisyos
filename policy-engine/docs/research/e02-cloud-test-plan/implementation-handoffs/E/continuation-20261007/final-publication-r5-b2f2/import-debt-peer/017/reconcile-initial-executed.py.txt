def require(condition: object, message: str) -> None:
    if not condition:
        raise ValueError(message)


def digest(body: bytes) -> str:
    return hashlib.sha256(body).hexdigest()


def frozen_json(value: object) -> bytes:
    return json.dumps(value, ensure_ascii=False, sort_keys=True, separators=(",", ":")).encode()


def emit(name: str, value: object) -> dict[str, Any]:
    body = (json.dumps(value, ensure_ascii=False, indent=2) + "\n").encode()
    path = OUT / name
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_bytes(body)
    return {"path": str(path), "bytes": len(body), "sha256": digest(body)}


def file_ref(path: Path) -> dict[str, Any]:
    body = path.read_bytes()
    return {"path": str(path), "bytes": len(body), "sha256": digest(body)}


def git(*args: str, input_bytes: bytes | None = None) -> bytes:
    require(all(type(arg) is str and "\x00" not in arg for arg in args), "Git argv types")
    fixed = args in {
        ("rev-parse", "HEAD"),
        ("rev-parse", "HEAD^{tree}"),
        ("rev-parse", f"{SOURCE}^{{tree}}"),
        ("branch", "--show-current"),
        ("status", "-sb"),
        ("diff", "--name-status", "--find-renames", "-z", BASE, SOURCE),
    }
    object_list = (
        len(args) == 3
        and args[:2] == ("ls-tree", "-rz")
        and args[2]
        in {
            BASE,
            OLD,
            SOURCE,
        }
    )
    object_read = args == ("cat-file", "--batch") and isinstance(input_bytes, bytes)
    if object_read:
        require(
            all(re.fullmatch(rb"[0-9a-f]{40}", row) for row in input_bytes.splitlines()),
            "Git batch objects",
        )
    ancestry = (
        len(args) == 4
        and args[:3] == ("merge-base", "--is-ancestor", SOURCE)
        and bool(re.fullmatch(r"[0-9a-f]{40}", args[3]))
    )
    require(fixed or object_list or object_read or ancestry, "exact read-only Git classes")
    argv = ["/usr/bin/git", "-C", str(ROOT), *args]
    result = subprocess.run(  # noqa: S603 - exact read-only Git command classes admitted below.
        argv, input=input_bytes, capture_output=True, check=False, shell=False
    )
    COMMANDS.append(
        {
            "argv": argv,
            "exit_code": result.returncode,
            "stdout_bytes": len(result.stdout),
            "stdout_sha256": digest(result.stdout),
            "stderr": result.stderr.decode(errors="replace"),
            "input_bytes": len(input_bytes) if input_bytes else 0,
            "input_sha256": digest(input_bytes) if input_bytes else None,
        }
    )
    require(result.returncode == 0, f"Git failed: {argv}: {result.stderr!r}")
    return result.stdout

def snapshot() -> dict[str, Any]:
    return {
        "time_utc": dt.datetime.now(dt.UTC).isoformat(),
        "head": git("rev-parse", "HEAD").decode().strip(),
        "tree": git("rev-parse", "HEAD^{tree}").decode().strip(),
        "branch": git("branch", "--show-current").decode().strip(),
        "status_sb": git("status", "-sb").decode(),
        "metadata": metadata(),
    }


def tree_map(revision: str) -> dict[str, str]:
    require(bool(re.fullmatch(r"[0-9a-f]{40}", revision)), "full Git revision required")
    entries: dict[str, str] = {}
    for entry in git("ls-tree", "-rz", revision).split(b"\x00"):
        if not entry:
            continue
        header, path = entry.split(b"\t", 1)
        _mode, kind, blob = header.split(b" ")
        if kind == b"blob":
            entries[path.decode()] = blob.decode()
    return entries


def read_blobs(blobs: set[str]) -> dict[str, bytes]:
    for blob in blobs:
        require(bool(re.fullmatch(r"[0-9a-f]{40}", blob)), "unsafe tree blob")
    request = ("\n".join(sorted(blobs)) + "\n").encode()
    result = git("cat-file", "--batch", input_bytes=request)
    offset = 0
    bodies: dict[str, bytes] = {}
    for requested in sorted(blobs):
        end = result.index(b"\n", offset)
        blob, kind, raw_size = result[offset:end].split(b" ")
        require(blob.decode() == requested and kind == b"blob", "cat-file object mismatch")
        size = int(raw_size)
        body = result[end + 1 : end + 1 + size]
        require(
            len(body) == size and result[end + 1 + size : end + 2 + size] == b"\n", "blob framing"
        )
        bodies[requested] = body
        offset = end + 2 + size
    require(offset == len(result), "unconsumed blob output")
    return bodies


def module_info(path: str) -> tuple[str, bool, str | None]:
    if path.startswith("policy-engine/src/"):
        parts = path.removeprefix("policy-engine/src/").split("/")
    else:
        parts = path.removeprefix("policy-engine/").split("/")
    package = parts[-1] == "__init__.py"
    parts = parts[:-1] if package else [*parts[:-1], parts[-1].removesuffix(".py")]
    module = ".".join(parts)
    root = parts[1] if len(parts) > 1 and parts[0] == "polisyos" else None
    return module, package, root


def resolve_relative(module: str, package: bool, target: str | None, level: int) -> str | None:
    if level == 0:
        return target
    parts = module.split(".") if package else module.split(".")[:-1]
    if level - 1 > len(parts):
        return None
    base = parts[: len(parts) - (level - 1)]
    if target:
        base.extend(target.split("."))
    return ".".join(base)


def literal_text(node: ast.AST) -> str | None:
    if isinstance(node, ast.Constant) and isinstance(node.value, str):
        return node.value
    if isinstance(node, ast.BinOp) and isinstance(node.op, ast.Add):
        left, right = literal_text(node.left), literal_text(node.right)
        return left + right if left is not None and right is not None else None
    return None


def import_rows(path: str, body: bytes) -> dict[str, Any]:
    tree = ast.parse(body, filename=path)
    module, package, root = module_info(path)
    aliases = {"__import__": "__import__"}
    for node in ast.walk(tree):
        if isinstance(node, ast.Import):
            for alias in node.names:
                if alias.name == "importlib":
                    aliases[alias.asname or alias.name] = "importlib"
        elif isinstance(node, ast.ImportFrom) and node.module == "importlib" and node.level == 0:
            for alias in node.names:
                if alias.name == "import_module":
                    aliases[alias.asname or alias.name] = "import_module"
    rows: list[dict[str, Any]] = []
    unresolved: list[dict[str, Any]] = []
    for node in ast.walk(tree):
        targets: list[tuple[str | None, list[str], str, bool]] = []
        if isinstance(node, ast.Import):
            targets = [(alias.name, [], "static-import", True) for alias in node.names]
        elif isinstance(node, ast.ImportFrom):
            targets = [
                (
                    resolve_relative(module, package, node.module, node.level),
                    [alias.name for alias in node.names],
                    "static-from",
                    True,
                )
            ]
        elif isinstance(node, ast.Call):
            func = node.func
            recognized = (
                isinstance(func, ast.Name)
                and aliases.get(func.id) in {"__import__", "import_module"}
            ) or (
                isinstance(func, ast.Attribute)
                and func.attr == "import_module"
                and isinstance(func.value, ast.Name)
                and aliases.get(func.value.id) == "importlib"
            )
            if recognized:
                target = literal_text(node.args[0]) if node.args else None
                if target is None:
                    unresolved.append(
                        {
                            "source_path": path,
                            "line": node.lineno,
                            "kind": "recognized-dynamic-nonliteral",
                            "expression": ast.unparse(node),
                            "status": "not_established",
                        }
                    )
                elif target.startswith("."):
                    package_arg = (
                        node.args[1]
                        if len(node.args) > 1
                        else next((kw.value for kw in node.keywords if kw.arg == "package"), None)
                    )
                    anchor = literal_text(package_arg) if package_arg is not None else None
                    if isinstance(package_arg, ast.Name) and package_arg.id == "__name__":
                        anchor = module
                    if anchor is not None:
                        level = len(target) - len(target.lstrip("."))
                        target = resolve_relative(anchor, True, target.lstrip(".") or None, level)
                    else:
                        unresolved.append(
                            {
                                "source_path": path,
                                "line": node.lineno,
                                "kind": "recognized-dynamic-relative-unresolved",
                                "expression": ast.unparse(node),
                                "status": "not_established",
                            }
                        )
                        target = None
                if target:
                    targets = [(target, [], "literal-dynamic", False)]
        for target, symbols, kind, canonical in targets:
            if not target or not target.startswith("polisyos."):
                continue
            parts = target.split(".")
            target_root = parts[1] if len(parts) > 1 else None
            rows.append(
                {
                    "source_path": path,
                    "source_module": module,
                    "source_root": root,
                    "target_module": target,
                    "target_root": target_root,
                    "line": node.lineno,
                    "symbols": symbols,
                    "kind": kind,
                    "canonical_collect_deep_import_edges_scope": canonical and root is not None,
                    "cross_root": root is not None and root != target_root,
                }
            )
    return {
        "rows": sorted(rows, key=lambda row: (row["line"], row["target_module"])),
        "unresolved": unresolved,
    }


def assignments(body: bytes) -> dict[str, Any]:
    result: dict[str, Any] = {}
    for node in ast.parse(body).body:
        if isinstance(node, ast.Assign):
            names = [target.id for target in node.targets if isinstance(target, ast.Name)]
            value = node.value
        elif isinstance(node, ast.AnnAssign) and isinstance(node.target, ast.Name):
            names = [node.target.id]
            value = node.value
        else:
            continue
        try:
            literal = ast.literal_eval(value) if value is not None else None
        except (ValueError, TypeError):
            continue
        result.update(dict.fromkeys(names, literal))
    return result


def edge_tokens(rows: list[dict[str, Any]], static_only: bool = False) -> set[tuple[str, str]]:
    return {
        (row["source_path"], row["target_module"])
        for row in rows
        if row["cross_root"] and (not static_only or row["kind"] != "literal-dynamic")
    }


def admitted(row: dict[str, Any], allowed: dict[str, set[str]]) -> bool:
    return row["target_module"] == f"polisyos.{row['target_root']}" or row[
        "target_module"
    ] in allowed.get(row["target_root"], set())


def source_record(
    path: str, revision: str, trees: dict[str, dict[str, str]], bodies: dict[str, bytes]
) -> dict[str, Any]:
    blob = trees[revision].get(path)
    if blob is None:
        return {"path": path, "source": revision, "exists": False}
    body = bodies[blob]
    return {
        "path": path,
        "source": revision,
        "exists": True,
        "blob": blob,
        "bytes": len(body),
        "sha256": digest(body),
    }
