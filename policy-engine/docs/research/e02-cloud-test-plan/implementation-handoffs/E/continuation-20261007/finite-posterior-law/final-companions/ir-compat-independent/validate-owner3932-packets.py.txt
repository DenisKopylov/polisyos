def need(condition: bool, reason: str) -> None:
    if not condition:
        raise ValueError(reason)


def read_ref(value: dict[str, object]) -> bytes:
    need(type(value["bytes"]) is int and value["bytes"] >= 0, "ref count type")
    digest = value["sha256"]
    need(isinstance(digest, str) and len(digest) == 64, "ref digest type")
    need(all(c in "0123456789abcdef" for c in digest), "ref digest format")
    path = Path(str(value["path"]))
    need(path.is_absolute() and ".." not in path.parts, "ref path admission")
    need(str(path).startswith(str(WORK) + "/"), "ref source scope")
    body = path.read_bytes()
    need(len(body) == value["bytes"], "ref size drift")
    need(hashlib.sha256(body).hexdigest() == digest, "ref hash drift")
    return body
