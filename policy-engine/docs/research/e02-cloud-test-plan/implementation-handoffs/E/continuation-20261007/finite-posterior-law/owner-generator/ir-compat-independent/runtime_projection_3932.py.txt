def require(value: object, message: str) -> None:
    if not value:
        raise ValueError(message)


def record(path: Path) -> dict[str, object]:
    raw = path.read_bytes()
    return {"path": str(path), "bytes": len(raw), "sha256": hashlib.sha256(raw).hexdigest()}


def git(*args: str) -> bytes:
    require(GIT is not None, "resolved Git executable required")
    return subprocess.run(  # noqa: S603 -- resolved executable and fixed exact-SHA/read-only argv below
        [GIT, *args], cwd=REPO, check=True, capture_output=True, shell=False
    ).stdout


def source_guard() -> dict[str, object]:
    status = git("status", "--porcelain", "--untracked-files=all").decode()
    require(
        all(line.startswith("?? " + ALLOWED_DOCS) for line in status.splitlines()),
        "only author-owned untracked handoff docs allowed",
    )
    live_head = git("rev-parse", "HEAD").decode().strip()
    head_delta = git("diff", "--name-only", CANDIDATE, live_head).decode().splitlines()
    require(all(path.startswith(ALLOWED_DOCS) for path in head_delta), "tracked product drift")
    require(git("rev-parse", CANDIDATE + "^{tree}").decode().strip() == TREE, "tree drift")
    return {
        "status": status,
        "live_head": live_head,
        "candidate_to_live_head_paths": head_delta,
        "tracked_source_clean": True,
    }
