def require(condition: bool, reason: str) -> None:
    if not condition:
        raise ValueError(reason)


def typed_equal(actual: object, expected: object, path: str = "$") -> None:
    require(type(actual) is type(expected), "JSON field type mismatch at " + path)
    if isinstance(actual, dict) and isinstance(expected, dict):
        require(set(actual) == set(expected), "JSON field denominator mismatch at " + path)
        for key in expected:
            typed_equal(actual[key], expected[key], path + "/" + str(key))
    elif isinstance(actual, list) and isinstance(expected, list):
        require(len(actual) == len(expected), "JSON array denominator mismatch at " + path)
        for i, value in enumerate(expected):
            typed_equal(actual[i], value, path + "/" + str(i))
    else:
        require(actual == expected, "JSON field value mismatch at " + path)

def git_blob(path: str, *, absent_allowed: bool = False) -> bytes | None:
    paths = {r["path"] for r in CANONICAL_PROJECTION["changed_generated_companions"]}
    paths.update(r["path"] for r in CANONICAL_OWNER["property_paths_and_native_consumers"])
    footprint = json.loads(legacy_namespace["read_ref"](CANONICAL_PROJECTION["source_footprint"]))
    paths.update(r["path"] for r in footprint["paths"])
    paths.update(r["original_criterion"]["path"] for r in CANONICAL_OWNER["rows"])
    require(
        path in paths and ".." not in Path(path).parts, "Git source path outside bound footprint"
    )
    # Exact immutable revision, canonical admitted repository path, shell=False.
    result = subprocess.run(  # noqa: S603
        ["/usr/bin/git", "-C", str(REPO), "show", SOURCE + ":" + path],
        capture_output=True,
        check=False,
    )
    if result.returncode != 0:
        require(
            absent_allowed
            and path
            in {
                "policy-engine/schemas/snapshots/ir/posterior_summary_profile.schema.json",
                "policy-engine/schemas/snapshots/ir/posterior_summary_profile_v2.schema.json",
            },
            "missing bound Git source",
        )
        return None
    return result.stdout


def admit_current_claims(projection: dict[str, Any], owner: dict[str, Any]) -> None:
    runtime = json.loads(
        legacy_namespace["read_ref"](projection["new_affected_runtime_projection"])
    )
    native_runtime = json.loads(legacy_namespace["read_ref"](runtime["runtime_output"]))
    typed_equal(
        projection["runtime_schema_components"],
        len(native_runtime["components"]["schemas"]),
        "$/runtime_schema_components",
    )
    actual_environment = runtime["existing_DoE184_all_METADATA_unchanged"]
    typed_equal(
        projection["existing_DoE184_environment_metadata_unchanged"],
        actual_environment,
        "$/existing_DoE184_environment_metadata_unchanged",
    )
    for row in projection["changed_generated_companions"]:
        path = row["path"]
        require(path in legacy_namespace["EXPECTED_PATHS"], "unadmitted generated companion")
        old = git_blob(path, absent_allowed=True)
        expected_old_bytes = 0 if old is None else len(old)
        expected_old_hash = None if old is None else hashlib.sha256(old).hexdigest()
        typed_equal(row["old_bytes"], expected_old_bytes, path + "/old_bytes")
        typed_equal(row["old_sha256"], expected_old_hash, path + "/old_sha256")
        if path == "policy-engine/schemas/runtime_api_v1.openapi.json":
            actual = legacy_namespace["read_ref"](runtime["runtime_output"])
        else:
            actual = (PROJECTED / Path(path).relative_to("policy-engine")).read_bytes()
        typed_equal(row["projected_bytes"], len(actual), path + "/projected_bytes")
        typed_equal(
            row["projected_sha256"], hashlib.sha256(actual).hexdigest(), path + "/projected_sha256"
        )
        typed_equal(
            row["same_as_source_candidate"], old == actual, path + "/same_as_source_candidate"
        )
    footprint = json.loads(legacy_namespace["read_ref"](projection["source_footprint"]))
    for row in footprint["paths"]:
        raw = git_blob(row["path"])
        typed_equal(row["bytes"], len(raw), row["path"] + "/bytes")
        typed_equal(row["sha256"], hashlib.sha256(raw).hexdigest(), row["path"] + "/sha256")
    for row in owner["property_paths_and_native_consumers"]:
        require(row["revision"] == SOURCE, "native consumer source revision")
        body = git_blob(row["path"])
        typed_equal(row["bytes"], len(body), row["path"] + "/bytes")
        typed_equal(row["sha256"], hashlib.sha256(body).hexdigest(), row["path"] + "/sha256")
    for row in owner["rows"]:
        fragment = row["original_criterion"]
        body = git_blob(fragment["path"]).decode().splitlines(keepends=True)
        start, end = fragment["lines"]
        require(type(start) is type(end) is int and start <= end, "criterion typed lines")
        text = "".join(body[start - 1 : end])
        typed_equal(fragment["text"], text, row["id"] + "/original_criterion")
        typed_equal(
            fragment["sha256"],
            hashlib.sha256(text.encode()).hexdigest(),
            row["id"] + "/original_criterion_sha256",
        )
    # Preserve the explicit owner/source-observation record, including prose and extra fields.
    # This conservation check follows independent raw/Git recomputation; it does not ratify science.
    typed_equal(owner, CANONICAL_OWNER, "$/frozen_E_owner_decision")
    typed_equal(projection, CANONICAL_PROJECTION, "$/closed_projection_claims")


def validate(projection: dict[str, Any], owner: dict[str, Any]) -> None:
    legacy_namespace["validate"](projection, owner)
    admit_current_claims(projection, owner)
