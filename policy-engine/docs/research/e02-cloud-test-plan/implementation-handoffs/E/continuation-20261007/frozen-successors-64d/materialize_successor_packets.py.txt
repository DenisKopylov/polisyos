def copy(source: Path, relative: str, role: str, expected: dict | None = None) -> None:
    path = PurePosixPath(relative)
    if path.is_absolute() or '..' in path.parts or relative != path.as_posix() or not path.parts:
        raise ValueError('Invalid bounded publication destination')
    if source.is_symlink() or not source.is_file():
        raise ValueError('Source must be closed regular input')
    data = source.read_bytes()
    digest = hashlib.sha256(data).hexdigest()
    if expected and (len(data) != expected['bytes'] or digest != expected['sha256']):
        raise ValueError('Selected closed source bytes changed: ' + str(source))
    target = DEST / relative
    target.parent.mkdir(parents=True, exist_ok=True)
    with target.open('xb') as handle:
        handle.write(data)
    copies.append({'source_path': str(source), 'copied_path': PREFIX + '/' + relative, 'bytes': len(data), 'sha256': digest, 'role': role})
