def digest(data):
    return hashlib.sha256(data).hexdigest()


def git(*args):

def req(value, message):
    if not value:

def write(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)

def source(path, ref=CAND):
    data = git('show', ref + ':' + path)
    return dict(path=path, source_sha=ref, git_blob=git('rev-parse', ref + ':' + path).decode().strip(), bytes=len(data), sha256=digest(data))

    data = path.read_bytes()
