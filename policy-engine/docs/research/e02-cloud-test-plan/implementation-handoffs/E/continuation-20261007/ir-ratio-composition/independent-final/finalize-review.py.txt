def typed_equal(left, right):
    if type(left) is not type(right):
        return False
    if isinstance(left, dict):
        return left.keys() == right.keys() and all(typed_equal(left[key], right[key]) for key in left)
    if isinstance(left, list):
        return len(left) == len(right) and all(typed_equal(a, b) for a, b in zip(left, right))
    return left == right


def git(*args):
    return subprocess.check_output(['git', *args], cwd=ROOT)


def admit_bound_inputs(value):
    if type(value) is not dict or set(value) != {'source', 'tree', 'base', 'paths', 'counts', 'finding_closure', 'type_scope'}:
        raise ValueError('exact bound input fields required')
    if (value['source'], value['tree'], value['base']) != (SHA, TREE, BASE):
        raise ValueError('wrong immutable source identity')
    if not typed_equal(value['counts'], {'pytest': 54, 'pytest_PASS': 54, 'runtime': 31, 'runtime_PASS': 31}):
        raise ValueError('counts must match exact closed native receipts')
    if value['finding_closure'] is not False or value['type_scope'] != 'scoped_original_baseline_relative':
        raise ValueError('scope/authority promotion')
    if type(value['paths']) is not list or len(value['paths']) != 5:
        raise ValueError('complete five-path source footprint required')
    seen = set()
    for row in value['paths']:
        if type(row) is not dict or set(row) != {'path', 'bytes', 'sha256'}:
            raise ValueError('exact source row required')
        if type(row['path']) is not str or row['path'] not in PATHS or row['path'] in seen:
            raise ValueError('source footprint mismatch')
        seen.add(row['path'])
        body = git('show', f'{SHA}:{row["path"]}')
        if type(row['bytes']) is not int or row['bytes'] != len(body) or row['sha256'] != hashlib.sha256(body).hexdigest():
            raise ValueError('source body hash/size mismatch')
        if (ROOT / row['path']).read_bytes() != body:
            raise ValueError('actual source body differs from immutable Git')
    if seen != PATHS:
        raise ValueError('full path denominator mismatch')
