def source(values, weights=None, mass=0.9, context=None):
    return adapter.summarize_bayesian_calibration_posterior(
        {'x': values}, weights=weights, credible_mass=mass,
        draw_ids=[f'oracle-row:{i}' for i in range(len(values))],
        sample_axis='oracle-chain:draw', context=context,
    ).parameter_envelopes['x']


def fresh(name, value):
    path = OUT / 'cas' / name
    ref = u.persist_uncertainty_envelope(artifacts.FileSystemCAS(path), value)
    result = u.load_uncertainty_envelope(artifacts.FileSystemCAS(path), ref)
    assert result.model_dump(mode='json') == value.model_dump(mode='json')
    return result, ref.model_dump(mode='json')


def run(name, fn):

    row = {'case': name, 'check_status': 'PASS', **value}
    ROWS.append(row)

def summary(value):
    profile = ps.read_posterior_summary_profile(value)
    return {'mean': None if profile is None else profile.posterior_mean,
            'point': value.point_estimate, 'interval': list(value.confidence_interval),
            'gate_eligible': value.gate_eligible,
            'profile_id': None if profile is None else profile.profile_id,
            'schema_version': value.schema_version}

def healthy(name, values, weights, mass, transform, expected_mean, expected_point, expected_interval):
    original = source(values, weights, mass)
    calls = []
    result = u.push_forward_envelope(lambda x: calls.append(x) or transform(x), original, map_name='oracle-map')
    result, ref = fresh(name, result)
    observed = summary(result)
    assert observed['mean'] == expected_mean
    assert observed['point'] == expected_point
    assert observed['interval'] == list(expected_interval)
    assert observed['gate_eligible'] is False
    profile = ps.read_posterior_summary_profile(result)
    assert profile.draw_ids == tuple(f'oracle-row:{i}' for i in range(len(values)))
    assert profile.sample_axis == 'oracle-chain:draw'
    assert len(calls) == len(values)
    return {'actual_callbacks': len(calls), 'fresh_ref': ref, 'observed': observed,
            'oracle_basis': 'independently_reconciled', 'law_mean_oracle': expected_mean,
            'law_point_oracle': expected_point, 'law_interval_oracle': list(expected_interval)}

def binding_control():
    context = ps.PosteriorSummaryContext(parameters={'x': ps.PosteriorParameterBinding(
        estimand_id='actual-input:score', unit='score', scale='linear')},
        purpose='predictive', time_roles={'fit_time': '2026-10-01'})
    original = source([0., 1., 4.], [1., 1., 2.], .5, context)
    original = original.model_copy(update={'metadata': {**original.metadata, 'fit_authority': True}})
    old = ps.read_posterior_summary_profile(original)
    result, ref = fresh('binding', u.push_forward_envelope(lambda x: x * 2., original))
    new = ps.read_posterior_summary_profile(result)
    assert new.binding is None and new.context.parameters == {}
    assert new.context.time_roles == context.time_roles and new.context.purpose == context.purpose
    assert result.metadata['inherited_parent_lineage_authoritative'] is False
    assert result.metadata['derived_parameter_binding'] == 'not_established'
    assert 'fit_authority' not in result.metadata
    assert new.carrier_content_hash != old.carrier_content_hash and new.joint_law_sha256 != old.joint_law_sha256
    forged, forged_ref = fresh('binding-forged', result.model_copy(update={'metadata': original.metadata}))
    try:
        ps.read_posterior_summary_profile(forged)
    except ValueError as exc:
        error = str(exc)
    else:
        raise AssertionError('stale profile/carrier substitution admitted')
    return {'fresh_ref': ref, 'forged_ref': forged_ref, 'forged_refusal': error,
            'derived_binding': None, 'context_parameters': {}, 'inherited_authority': False}


run('arbitrary_map_reissue_and_stale_content_refusal', binding_control)


def refused_declaration(variant, operation):
    original = source([0., 1.], [1., 2.])
    md = copy.deepcopy(original.metadata)
    for key in ('posterior_summary_profile_id', 'posterior_summary_profile_version'):
        md.pop(key)
    if variant == 'missing':
        md.pop(ps.PROFILE_KEY)
    else:
        md[ps.PROFILE_KEY]['profile_id'] = 'urn:unsupported-posterior-law:9'
        md[ps.PROFILE_KEY]['profile_version'] = '9.0'
    forged = original.model_copy(update={'metadata': md})
    calls = []
    func = lambda x: calls.append(x) or x
    try:
        if operation == 'push':
            u.push_forward_envelope(func, forged)
        elif operation == 'compress':
            u.compress_envelope(forged, target='interval')
        else:
            u.pull_back_envelope(func, forged, upstream_particles=(0., 1.))
    except ValueError as exc:
        error = str(exc)
    else:
        raise AssertionError('unknown declared law was emitted')
    assert calls == [] and md['posterior_summary_profile_required'] is True
    return {'actual_callbacks': 0, 'refusal': error, 'required_marker_retained': True,
            'new_outer_discriminators_removed': 2, 'result_emitted': False}


for variant in ('missing', 'unknown'):
    for operation in ('push', 'compress', 'pull'):
        run(f'{variant}_declared_profile_{operation}_precallback', lambda v=variant, o=operation: refused_declaration(v, o))

def lossy(target):
    original = source([0.] * 99 + [100.])
    result, ref = fresh('loss-' + target, u.compress_envelope(original, target=target))
    assert not any(k in result.metadata for k in DROP_KEYS)
    assert ps.read_posterior_summary_profile(result) is None and result.gate_eligible is False
    if target == 'moments':
        assert result.point_estimate == 1.
        assert result.distribution_payload.parameters == {'mean': 1., 'std': math.sqrt(99.)}
        z = NormalDist().inv_cdf((1. + .9) / 2.)
        assert result.confidence_interval == (1. - z * math.sqrt(99.), 1. + z * math.sqrt(99.))
    forged, forged_ref = fresh('loss-forged-' + target, result.model_copy(update={'metadata': original.metadata}))
    try:
        ps.read_posterior_summary_profile(forged)
    except ValueError as exc:
        error = str(exc)
    else:
        raise AssertionError('lawless compressed object accepted old profile')
    return {'fresh_ref': ref, 'named_profile': None, 'gate_eligible': False,
            'actual_point': result.point_estimate, 'actual_interval': list(result.confidence_interval),
            'forged_ref': forged_ref, 'forged_refusal': error, 'all_ten_law_markers_cleared': True}

def compile_function(body, name, namespace=None):
    tree = ast.parse(body)
    node = next(n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == name)
    segment = ast.get_source_segment(body, node)
    ns = dict(u.__dict__) if namespace is None else namespace
    exec(compile(segment, '<independent-exact-function-' + name + '>', 'exec', flags=__future__.annotations.compiler_flag), ns)
    return ns[name], segment

def legacy_full(schema, operation):
    env = u.UncertaintyEnvelope(schema_version=schema, point_estimate=.5, confidence_interval=(0., 1.),
        confidence_level=None, interval_semantics='deterministic_bounds', source='ensemble', distribution_family='bootstrap',
        distribution_payload=u.PosteriorSamplesCarrier(samples=(0., 1.), weights=(.5, .5)), gate_eligible=False)
    old_name = {'push': 'push_forward_envelope', 'pull': 'pull_back_envelope', 'compress': 'compress_envelope'}[operation]
    old_fn, old_segment = LEGACY_FUNCS[old_name]
    results = []
    for label, fn in [('exact_base', old_fn), ('candidate', getattr(u, old_name))]:
        if operation == 'push':
            result = fn(lambda x: x, env, map_name='identity')
        elif operation == 'pull':
            result = fn(lambda x: x, env, local_inverse=lambda x: x, map_name='identity')
        else:
            result = fn(env, target='particles')
        results.append({'source': label, 'body': result.model_dump(mode='json')})
    assert results[0]['body'] == results[1]['body']
    result, ref = fresh('legacy-' + schema + '-' + operation, result)
    assert result.model_dump(mode='json') == results[0]['body']
    return {'input_schema': schema, 'output_schema': result.schema_version, 'full_output_equal_to_exact_base': True,
            'legacy_function_sha256': hashlib.sha256(old_segment.encode()).hexdigest(), 'fresh_ref': ref,
            'output_sha256': hashlib.sha256(json.dumps(results[0]['body'], sort_keys=True, separators=(',', ':')).encode()).hexdigest()}

def admission_removal():
    segment = inspect.getsource(u._ratio_composition_profile)
    needle = 'profile = read_posterior_summary_profile(envelope)'
    assert segment.count(needle) == 1
    removed, altered = compile_function(segment.replace(needle, 'profile = None'), '_ratio_composition_profile')
    original = u._ratio_composition_profile
    env = source([0., 1.], [1., 2.])
    md = copy.deepcopy(env.metadata)
    md.pop('posterior_summary_profile_id')
    md.pop('posterior_summary_profile_version')
    md[ps.PROFILE_KEY]['profile_id'] = 'urn:unsupported-posterior-law:9'
    md[ps.PROFILE_KEY]['profile_version'] = '9.0'
    forged = env.model_copy(update={'metadata': md})
    calls = []
    try:
        u._ratio_composition_profile = removed
        result = u.push_forward_envelope(lambda x: calls.append(x) or x, forged)
    finally:
        u._ratio_composition_profile = original
    result, ref = fresh('removal-common-inlet', result)
    try:
        ps.read_posterior_summary_profile(result)
    except ValueError as exc:
        late = str(exc)
    else:
        raise AssertionError('unknown law fresh reader unexpectedly admitted')
    assert len(calls) == 2 and result.metadata['posterior_summary_profile_required'] is True
    return {'defining_property_outcome': 'FAIL', 'actual_callbacks': 2, 'fresh_ref': ref,
            'late_reader_refusal': late, 'required_marker_and_unknown_profile_retained': True,
            'source_edit': {'function': '_ratio_composition_profile', 'replacement': 'strict reader call -> None',
                            'original_sha256': hashlib.sha256(segment.encode()).hexdigest(),
                            'altered_sha256': hashlib.sha256(altered.encode()).hexdigest()}, 'source_files_written': False}


run('sole_common_admission_removal_restores_unknown_callback_escape', admission_removal)
