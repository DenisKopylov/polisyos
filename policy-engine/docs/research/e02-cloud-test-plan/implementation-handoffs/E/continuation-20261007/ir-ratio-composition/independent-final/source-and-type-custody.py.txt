# A complete bounded static census, not grep-zero and not evidence of arbitrary dynamic dispatch.
targets = {'push_forward_envelope', 'pull_back_envelope', 'compress_envelope'}
files = git('ls-files', 'policy-engine/src/**/*.py', 'policy-engine/tests/**/*.py').decode().splitlines()
assert len(files) == len(set(files))
inputs, references, errors = [], [], []
for path in files:
    body = (ROOT / path).read_bytes()
    inputs.append({'path': path, 'bytes': len(body), 'sha256': hashlib.sha256(body).hexdigest()})
    try:
        tree = ast.parse(body, filename=path)
    except SyntaxError as exc:
        errors.append({'path': path, 'error': str(exc)})
        continue
    imported = {}
    for node in ast.walk(tree):
        if isinstance(node, ast.ImportFrom):
            for alias in node.names:
                imported[alias.asname or alias.name] = (node.module or '') + '.' + alias.name
        elif isinstance(node, ast.Import):
            for alias in node.names:
                imported[alias.asname or alias.name] = alias.name
    for node in ast.walk(tree):
        if not isinstance(node, ast.Call):
            continue
        if isinstance(node.func, ast.Name):
            target = node.func.id
            resolved = imported.get(target, target)
        elif isinstance(node.func, ast.Attribute):
            target = node.func.attr
            try:
                expr = ast.unparse(node.func)
            except Exception:
                expr = target
            first, _, tail = expr.partition('.')
            resolved = imported.get(first, first) + ('.' + tail if tail else '')
        else:
            continue
        if target in targets:
            references.append({'path': path, 'line': node.lineno, 'target': target, 'resolved_static_spelling': resolved,
                               'surface': 'test' if path.startswith('policy-engine/tests/') else 'source'})
assert errors == []
product = [row for row in references if row['surface'] == 'source']
full = {'schema': 'e02-independent-composition-static-census:1', 'source': SHA, 'tree': TREE,
        'input_files': inputs, 'input_count': len(inputs), 'references': references,
        'source_references': product, 'errors': errors,
        'limitation': 'Complete tracked Python source/tests denominator, static call spellings only; dynamic/reflection/injected external caller reachability unmeasured.'}
(OUT / 'caller-census.json').write_text(json.dumps(full, indent=2) + '\n')
assert git('rev-parse', 'HEAD').decode().strip() == SHA
assert git('rev-parse', 'HEAD^{tree}').decode().strip() == TREE

          'type_reexecuted': False, 'actual_author_receipt': receipt,
          'original_baseline_bytes': len(baseline), 'original_baseline_sha256': hashlib.sha256(baseline).hexdigest(),
          'supported_config_settings_equal_except_scratch_baseline': True,
          'complete_static_census_inputs': len(inputs), 'static_source_references': product,
          'census_bytes': (OUT / 'caller-census.json').stat().st_size,
          'census_sha256': hashlib.sha256((OUT / 'caller-census.json').read_bytes()).hexdigest(),
          'source_written': False,
          'limitations': ['Unsupported .json.tmp 9-errors/209-warnings observer uses fallback config, retained historically; not current canonical type outcome.',
                          'Earlier actual tracked baseline prune/restoration observations retained, not overwritten by isolated supported run.',
                          'No dynamic production caller, generated surface, scientific authority, finding closure or whole gate PASS inferred.']}
(OUT / 'source-and-type-custody.json').write_text(json.dumps(result, indent=2) + '\n')
