def require(ok,why):
 if not ok:raise ValueError(why)

def read(repo,ref,path):
 require(isinstance(ref,str) and re.fullmatch('[0-9a-f]{40}',ref) is not None,'exact immutable source SHA')
 require(isinstance(path,str) and path and not Path(path).is_absolute() and '..' not in Path(path).parts and not path.startswith('-'),'Git source path')
 return subprocess.run(['git','-C',str(repo),'show',ref+':'+path],check=True,capture_output=True).stdout

def asset(repo,record):
 require(type(record['bytes']) is int and record['bytes']>=0,'typed bytes')
 require(isinstance(record['sha256'],str) and re.fullmatch('[0-9a-f]{64}',record['sha256']) is not None,'typed digest')
 p=Path(record['path'])
 data=p.read_bytes() if p.is_absolute() else read(repo,record.get('source_sha',PRIOR),record['path'])
 require(len(data)==record['bytes'] and hashlib.sha256(data).hexdigest()==record['sha256'],'exact evidence bytes')
 if 'git_blob' in record:
  require(subprocess.run(['git','-C',str(repo),'rev-parse',record['source_sha']+':'+record['path']],check=True,capture_output=True).stdout.decode().strip()==record['git_blob'],'Git blob binding')
 return data
