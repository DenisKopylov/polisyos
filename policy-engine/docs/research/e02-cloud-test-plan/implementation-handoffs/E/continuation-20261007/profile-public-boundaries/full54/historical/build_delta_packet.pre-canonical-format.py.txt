def git(*args: str) -> bytes:
    return subprocess.check_output(["/usr/bin/git", *args], cwd=ROOT)  # noqa: S603 - argv is read-only Git, never shell text


def body(sha: str, path: str) -> bytes:
    return git("show", f"{sha}:{path}")


def digest(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def identity(sha: str, path: str) -> dict:
    data = body(sha, path)
    return {"revision": sha, "path": path, "bytes": len(data), "sha256": digest(data)}


def ref(input_name: str, fragment: str) -> dict:
    return {"input": input_name, "fragment": fragment}


def source_tree(sha: str) -> str:
    return git("rev-parse", f"{sha}^{{tree}}").decode().strip()


def unchanged_paths(old: str, new: str, paths: list[str]) -> list[dict]:
    rows = []
    for path in paths:
        left = body(old, path)
        try:
            right = body(new, path)
        except subprocess.CalledProcessError:
            right = None
        rows.append(
            {
                "path": path,
                "base_sha256": digest(left),
                "candidate_sha256": digest(right) if right is not None else None,
                "body_equal": left == right,
            }
        )
    return rows


def validate(packet: dict) -> None:
    """Recompute criteria, owners, unique cases and status-layer assertions."""
    old = json.loads(body(OLD_HEAD, OLD_PATH))
    g = json.loads(body(G_HEAD, G_PATH))
    rows = packet["rows"]
    if [r["id"] for r in rows] != [r["id"] for r in old["rows"]]:
        raise ValueError("exact 54 original row identities/order")
    original_links = {(r["id"], b) for r in old["rows"] for b in r["bundle_ids"]}
    links = [(r["id"], b) for r in rows for b in r["bundle_ids"]]
    if len(links) != 55 or set(links) != original_links:
        raise ValueError("exact 55 ID-to-bundle associations")
    if len({b for _, b in links}) != 22:

    histories = Counter(r["historical_ledger_status"] for r in rows)
    proposals = Counter(r["retained_E_author_proposal"] for r in rows)
    if histories != {"partial": 49, "held": 4, "closed": 1}:
        raise ValueError("historical ledger layer")
    if proposals != {"closed": 43, "limited": 9, "held": 2}:
        raise ValueError("author proposals layer")
    if packet["status_layers"]["formal_G_acceptance"] != 0:
        raise ValueError("formal G acceptance is separate")
    if packet["status_layers"]["formal_ledger_change"] is not False:
        raise ValueError("no automatic historical ledger migration")
    source_bindings = packet["original_sources"]
    for source in (
        source_bindings["finding_owners"],
        source_bindings["bundle_owners"],
        *source_bindings["cards"],
    ):
        exact = identity(source["revision"], source["path"])
        if any(source[key] != exact[key] for key in ("bytes", "sha256")):
            raise ValueError("original full cards and owner TSV content identity")
        if body(source["revision"], source["path"]) != body(
            packet["candidate"]["commit"], source["path"]
        ):
            raise ValueError("unexpected original ownership/card source change")
    for fragment in source_bindings["criterion_fragments"]:
        lines = body(fragment["revision"], fragment["path"]).splitlines(keepends=True)
        start, end = fragment["lines"]
        data = b"".join(lines[start - 1 : end])
        if len(data) != fragment["bytes"] or digest(data) != fragment["sha256"]:
            raise ValueError("exact original criterion line-fragment identity")
    for index, row in enumerate(rows):
        before = old["rows"][index]
        review = g["per_finding_rows"][index]
        for key in ("id", "bundle_ids", "historical_ledger_status"):
            if row[key] != before[key]:
                raise ValueError(f"original row {row['id']} {key}")
        if row["retained_E_author_proposal"] != review["current_E_proposed_outcome"]:
            raise ValueError(f"retained proposal {row['id']}")
        if row["historical_property"] != before["criterion_evidence"]["property"]:
            raise ValueError(f"exact original property {row['id']}")
        if row["original_criteria"] != before["criterion_evidence"]["original_criterion_refs"]:
            raise ValueError(f"criterion citation {row['id']}")
        if row["formal_G_acceptance"] or row["formal_ledger_change"]:
            raise ValueError("author finding proposal does not change G intake")
        if row["retained_E_author_proposal"] in ("limited", "held"):
            if not row["external_owner_input"]:
                raise ValueError(f"missing exact external input {row['id']}")
        elif row["external_owner_input"] is not None:
            raise ValueError("unrelated production inputs do not gate closed proposals")
    old_cases = json.loads(body(OLD_HEAD, CASE_PATH))["cases"]
    expected = [c for c in old_cases if c["outcome"] == "failed"]
    failures = packet["shared_A_reason_cases"]["historical_unique_failures"]
    tokens = {(c["job_name"], c["classname"], c["name"]) for c in failures}
    if len(failures) != 8 or len(tokens) != 8:
        raise ValueError("eight unique historical A reason failures")
    if failures != expected:
        raise ValueError("immutable b2 exact case tokens, no removed/doubled failure")
    if packet["shared_A_reason_cases"]["associated_findings"] != ["B32", "LA-051"]:
        raise ValueError("both affected IDs navigate the same eight cases")
    if packet["shared_A_reason_cases"]["current_A_native_check"] != "UNRUN":
        raise ValueError("A source-level review is not a current native run")
    if packet["shared_A_reason_cases"]["capability_state"] != "verification_missing":
        raise ValueError("existing consumer cannot be claimed wholly absent")
    if packet["historical_checks"]["candidate"] != OLD_SOURCE:
        raise ValueError("b2 wave does not belong to new candidate")
    if packet["historical_checks"]["counts"] != {
        "cases": 1465,
        "passed": 1457,
        "failed": 8,
        "errors": 0,
        "skipped": 0,
    }:
        raise ValueError("exact b2 observation retained")
    if packet["new_boundary"]["affected_original_ids"] != ["B201", "B202"]:
        raise ValueError("raw/join repairs not global closure by count")
    if packet["new_boundary"]["gate_eligible"] is not False:
        raise ValueError("finite law does not confer gate authority")
    if packet["new_boundary"]["production_or_institutional_authority"] != "not_established":
        raise ValueError("synthetic fixture not factual authority")
    if packet["new_boundary"]["mechanism_completed"]:
        binding = packet["new_boundary"]["source_and_receipt_binding"]
        if (
            not binding
            or not binding.get("independent_review")
            or not binding.get("deciding_receipts")
        ):
            raise ValueError("new mechanism requires immutable source/review/receipts")
        if binding["candidate"] != packet["candidate"]["commit"]:
            raise ValueError("binding source identity")
        if binding["tree"] != source_tree(binding["candidate"]):
            raise ValueError("binding source tree")
    for dep in packet["criterion_dependency_reconciliation"]:
        recomputed = unchanged_paths(OLD_SOURCE, packet["candidate"]["commit"], [dep["path"]])[0]
        if recomputed != dep:
            raise ValueError("dependency bytes recomputed from exact Git source")

def main() -> None:

    parser.add_argument("--output-stem", required=True)
    parser.add_argument("--binding", type=Path)
    args = parser.parse_args()
    if Path(args.output_stem).name != args.output_stem or not args.output_stem:
        raise ValueError("output stem must stay in this packet directory")
    target = OUT / f"{args.output_stem}.json"
    if args.verify is not None:
        packet = json.loads(args.verify.read_text())
        target = args.verify
