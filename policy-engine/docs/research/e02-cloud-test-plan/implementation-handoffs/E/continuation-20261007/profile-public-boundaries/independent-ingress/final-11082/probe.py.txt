def sha(data):
    return hashlib.sha256(data).hexdigest()

def git(*args):
    return subprocess.check_output(['git', *args], cwd=ROOT)

def source_binding(candidate):
    entries = git('ls-tree', '-r', '-z', candidate, 'policy-engine/src').split(b'\0')
    errors=[];count=0;total_bytes=0;framed=hashlib.sha256()
    for entry in entries:
        if not entry: continue
        header, name=entry.split(b'\t',1)
        mode, kind, expected_oid=header.decode().split()
        assert kind=='blob'
        path=name.decode();file=ROOT/path
        actual=os.readlink(file).encode() if mode=='120000' else file.read_bytes()
        actual_oid=hashlib.sha1(b'blob '+str(len(actual)).encode()+b'\0'+actual).hexdigest()
        if actual_oid != expected_oid: errors.append(path)
        count+=1;total_bytes+=len(actual)
        framed.update(name+b'\0'+expected_oid.encode()+b'\0'+str(len(actual)).encode()+b'\0')
    return {'paths':count,'bytes':total_bytes,'full_path_gitblob_size_frame_sha256':framed.hexdigest(),
            'all_git_blobs_bound':not errors,'errors':errors}
