def git(*args, missing=False):
    q = subprocess.run(['git', *args], cwd=R, capture_output=True)
    assert missing or q.returncode == 0, (args, q.stderr)
    return q

def source_ref(path, source=H):
    data = git('show', source + ':' + path).stdout
    return {'path': path, 'source': source, 'blob': git('rev-parse', source + ':' + path).stdout.decode().strip(), 'bytes': len(data), 'sha256': hashlib.sha256(data).hexdigest()}

def write(name, data):
    path = S / name
    path.write_text(json.dumps(data, ensure_ascii=False, indent=2) + '\n')
    return {'path': str(path), 'bytes': path.stat().st_size, 'sha256': hashlib.sha256(path.read_bytes()).hexdigest()}

def edges(path, commit):
    q = git('show', commit + ':' + path, missing=True)
    if q.returncode:
        return []
    root = path.removeprefix('policy-engine/src/polisyos/').split('/')[0] if path.startswith('policy-engine/src/polisyos/') else None
    rows = []
    for n in ast.walk(ast.parse(q.stdout)):
        pairs = [(n.module, [(a.name, a.asname) for a in n.names])] if isinstance(n, ast.ImportFrom) and n.level == 0 and n.module else ([(a.name, [(a.name, a.asname)]) for a in n.names] if isinstance(n, ast.Import) else [])
        for module, names in pairs:
            if module.startswith('polisyos.'):
                rows.append({'path': path, 'line': n.lineno, 'module': module, 'names': names, 'cross_root': root is not None and module.split('.')[1] != root, 'supported_entrypoint': module in supported})
    return rows
