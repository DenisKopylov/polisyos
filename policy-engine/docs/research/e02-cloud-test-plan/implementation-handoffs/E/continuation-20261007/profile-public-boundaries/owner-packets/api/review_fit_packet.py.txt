def sha(data):
    return hashlib.sha256(data).hexdigest()

def file_ref(path):
    data = Path(path).read_bytes()
    return {'path': str(path), 'bytes': len(data), 'sha256': sha(data)}

def git(*args, allowed_error=False):
    q = subprocess.run(['git', *args], cwd=R, capture_output=True)
    if not allowed_error:
        assert q.returncode == 0, (args, q.stderr)
    return q

def validate_assets(selection):
    for row in selection['assets']:
        actual = file_ref(row['source_path'])
        assert actual['bytes'] == row['bytes'], row['source_path']
        assert actual['sha256'] == row['sha256'], row['source_path']
    assert len({row['portable_path'] for row in selection['assets']}) == len(selection['assets'])
    assert sum(row['bytes'] for row in selection['assets']) == selection['bytes']

        q = git('show', commit + ':' + row['path'], allowed_error=True)

            assert q.returncode != 0
