ROOT = Path('/workspace/e02-E-continuation-20261006')
OUT = Path('/tmp/e02-E-forecast-packet-thirteenth-20261007')

REVIEW = PLAN + 'integration/reviews/2026-10-07-thirteenth-wave/'
assets = []


def run(*args: str) -> bytes:
    return subprocess.check_output(args, cwd=ROOT)


def source(ref: str, path: str) -> bytes:
    return run('git', 'show', ref + ':' + path)


def sha(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def capture(ref: str, path: str, portable: str) -> bytes:
    data = source(ref, path)
    target = OUT / portable
    target.parent.mkdir(parents=True, exist_ok=True)
    if target.exists() and target.read_bytes() != data:
        raise ValueError('refuse overwriting distinct existing evidence: ' + portable)
    target.write_bytes(data)
    assets.append({'source_ref': ref, 'source_path': path,
                   'source_blob': run('git', 'rev-parse', ref + ':' + path).decode().strip(),
                   'portable_path': portable, 'bytes': len(data), 'sha256': sha(data)})
    return data


def write(name: str, data: object) -> None:
    body = json.dumps(data, indent=2, ensure_ascii=False, sort_keys=True) + '\n'
    (OUT / name).write_text(body)


assert run('git', 'rev-parse', SOURCE + '^{tree}').decode().strip() == TREE
xml_paths = [p for p in run('git', 'ls-tree', '-r', '--name-only', DOC, R5).decode().splitlines()
             if p.startswith(R5 + 'final-numeric/canonical-collection/checks/')
             and p.endswith('/pytest.xml')]

all_cases = []
failures = []
old_selector = re.compile(r'test_frc_01|test_frc_02|test_forecast_owner|::test_(actual_ets_cas|actual_resolver|missing_ref_tier|resolved_limited)')
new_selector = re.compile(old_selector.pattern + r'|::test_native_gateway_tier_s6_and_disposition_share_resolver_reason\[')
for path in xml_paths:
    job = Path(path).parent.name
    data = capture(DOC, path, 'literal-JUnit/' + job + '.xml')
    for ordinal, case in enumerate(ET.fromstring(data).iter('testcase')):
        outcome = ('ERROR' if case.find('error') is not None else
                   'FAIL' if case.find('failure') is not None else
                   'SKIP' if case.find('skipped') is not None else 'PASS')
        row = {'job': job, 'ordinal_zero_based': ordinal,
               'classname': case.get('classname', ''), 'name': case.get('name', ''),
               'outcome': outcome, 'observed_source': SOURCE, 'observed_tree': TREE,
               'JUnit_path': path, 'JUnit_carrier': DOC, 'JUnit_sha256': sha(data)}
        row['identity'] = job + ':' + str(ordinal) + ':' + row['classname'] + '::' + row['name']
        all_cases.append(row)
        if outcome != 'PASS':
            row['failure_message'] = next(iter(case)).get('message', '')
            row['failure_text'] = next(iter(case)).text or ''
            failures.append(row)
counts = collections.Counter(c['outcome'] for c in all_cases)
