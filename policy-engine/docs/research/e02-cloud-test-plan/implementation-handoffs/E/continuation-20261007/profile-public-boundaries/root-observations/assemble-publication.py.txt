def fingerprint(path):
    data = path.read_bytes()
    return {'bytes': len(data), 'sha256': hashlib.sha256(data).hexdigest()}


def copy(source, relative, expected=None, role='literal command output or immutable receipt', selector=None):
    source = Path(source)
    relative = Path(relative)
    if relative.is_absolute() or '..' in relative.parts:
        raise ValueError(f'unsafe publication path: {relative}')
    actual = fingerprint(source)
    if expected is not None and any(actual[k] != expected[k] for k in actual):
        raise ValueError(f'changed deciding input: {source}')
    target = DEST / relative
    if target.exists() and target.read_bytes() != source.read_bytes():
        raise ValueError(f'publication conflict: {target}')
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_bytes(source.read_bytes())
    if fingerprint(target) != actual:
        raise ValueError(f'copy divergence: {target}')
    item = {'path': relative.as_posix(), 'original_path': str(source), **actual, 'role': role}
    if selector:
        item['original_selection'] = selector
    existing = next((x for x in rows if x['path'] == item['path']), None)
    if existing and existing != item:
        raise ValueError(f'index collision: {relative}')
    if not existing:
        rows.append(item)


def select(selector, prefix):
    selector = Path(selector)
    record = json.loads(selector.read_text())
    identity = {'path': str(selector), **fingerprint(selector)}
    assets = record.get('assets', record.get('selected_moderate_files', record.get('literal_files', [])))
    if not assets:
        raise ValueError(f'empty selector: {selector}')
    for asset in assets:
        source = Path(asset.get('source_path', asset.get('path', selector.parent / asset['name'] if 'name' in asset else '')))
        if not source.is_absolute():
            source = selector.parent / source
        name = asset.get('portable_path', asset.get('name', asset.get('path')))
        role = asset.get('role', 'literal historical output/input or immutable receipt')
        # Historical overlay inputs and executed scripts are preserved as documents,
        # not installed as active publication utilities. Their original path/hash
        # and original executed argv remain in the copy index and literal receipts.
        if selector.name == 'evidence-index.json' and str(name).endswith('.py'):
            name += '.txt'
            role = 'literal historical executed script or overlay source snapshot; not an active utility'
        copy(source, Path(prefix) / name, asset, role, identity)
    copy(selector, Path(prefix) / ('input-selection-' + selector.name), role='immutable selection; its original paths map through copy-index.json')
