# Remove only physical current/pinned-Git byte equality; marker fields stay unchanged.
class RemovePhysicalEquality(ast.NodeTransformer):
    count=0
    def visit_If(self,node):
        if any(isinstance(child,ast.Constant) and isinstance(child.value,str) and 'Review current copy differs from pinned Git' in child.value for child in ast.walk(node)):
            self.count+=1
            return ast.copy_location(ast.Pass(),node)
        return self.generic_visit(node)
