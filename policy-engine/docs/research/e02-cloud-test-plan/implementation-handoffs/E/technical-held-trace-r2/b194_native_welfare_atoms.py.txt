if outcome.status=="ok":
 bundle=load_welfare_bundle(FileSystemCAS(store.root),outcome.state.artifacts_index[ARTIFACT_WELFARE_BUNDLE_REF])
 report=from_canonical_bytes(store.get_bytes(bundle.diagnostics["propagation_report_ref"]))
 outcomes=from_canonical_bytes(store.get_bytes(report["draw_outcomes_ref"]["artifact_id"]))
 sampled=[row["sampled_input"]["x"] for row in outcomes["sampled_inputs"]]
 print(json.dumps({"independent_declared_support":[0.,1.],"expected_failure_mass":.5,"actual_out_of_support_draw_count":sum(v not in (0.,1.) for v in sampled),"sampled_min":min(sampled),"sampled_max":max(sampled)},indent=2))
 print(json.dumps({"bundle_status":bundle.status.value,"credible_interval":bundle.credible_interval,"report":report},indent=2))
