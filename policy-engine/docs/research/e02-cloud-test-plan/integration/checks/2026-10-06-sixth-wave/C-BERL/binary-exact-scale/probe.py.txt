from __future__ import annotations

import hashlib
import json
import math
import sys
from pathlib import Path

import numpy as np
import polisyos.berl.adapters.conditional_shapley as source
from polisyos.berl.adapters.conditional_shapley import GaussianJointLaw, gaussian_conditional_moments

EXPECTED = {'conditional_shapley.py': '78611ed4de5662c9994d648e6654be4df89176ef02e3f698cb5060a3b55168e4', '_coalition.py': '931416a01a4fc3e8870c79c86c9fece0002f58aa2a6d931757061c83dfa08f67', 'protocol.py': '4117e55ba77d3466090fb18ae52c6301f876b2129617086eafa5334a003525ec'}
EXPECTED_SOURCE_ROOT = Path('/Users/deniskopylov/.codex/worktrees/e02-integration-6971/polisyos/policy-engine/_build/e02-g-continuation-20261006/1548-C-berl-falsifier/selected-source').resolve()
module_path = Path(source.__file__).resolve()
loaded = {
    'conditional_shapley.py': module_path,
    '_coalition.py': Path(sys.modules['polisyos.berl.adapters._coalition'].__file__).resolve(),
    'protocol.py': Path(sys.modules['polisyos.berl.adapters.protocol'].__file__).resolve(),
}
actual_hashes = {name: hashlib.sha256(path.read_bytes()).hexdigest() for name, path in loaded.items()}
expected_paths = {
    'conditional_shapley.py': EXPECTED_SOURCE_ROOT / 'polisyos/berl/adapters/conditional_shapley.py',
    '_coalition.py': EXPECTED_SOURCE_ROOT / 'polisyos/berl/adapters/_coalition.py',
    'protocol.py': EXPECTED_SOURCE_ROOT / 'polisyos/berl/adapters/protocol.py',
}
source_qualified = all(
    loaded[name] == expected_paths[name].resolve()
    and actual_hashes[name] == EXPECTED[name]
    for name in EXPECTED
)

# Exact powers of two make every covariance entry and observation binary-representable.
scale = math.ldexp(1.0, -35)
scale_sq = math.ldexp(1.0, -70)
observed_x1 = math.ldexp(1.0, -38)
assert scale * scale == scale_sq
assert observed_x1 / scale == 0.125
law = GaussianJointLaw(
    mean=(0.0, 0.0, 0.0),
    covariance=(
        (scale_sq, 0.0, scale),
        (0.0, 1.0, 0.0),
        (scale, 0.0, 1.0),
    ),
)
result = gaussian_conditional_moments(
    law,
    feature_order=('x1', 'x2', 'x3'),
    coalition=frozenset({'x1', 'x2'}),
    observed={'x1': observed_x1, 'x2': 0.0},
)
expected_moments = {'x3_mean': 0.125, 'x3_variance': 0.0}
actual_moments = {'x3_mean': result.mean[2], 'x3_variance': result.covariance[2][2]}
matched = (
    math.isclose(actual_moments['x3_mean'], expected_moments['x3_mean'], rel_tol=1e-12, abs_tol=1e-12)
    and math.isclose(actual_moments['x3_variance'], expected_moments['x3_variance'], rel_tol=1e-12, abs_tol=1e-12)
)
report = {
    'source_qualification': {
        'selected_source_root': str(EXPECTED_SOURCE_ROOT),
        'loaded_paths': {name: str(path) for name, path in loaded.items()},
        'actual_sha256': actual_hashes,
        'expected_sha256': EXPECTED,
        'all_three_git_modules_source_qualified': source_qualified,
        'numpy_version': np.__version__,
        'numpy_path': str(Path(np.__file__).resolve()),
    },
    'input': {
        'scale_definition': '2**-35',
        'scale_hex': scale.hex(),
        'scale_squared_definition': '2**-70',
        'scale_squared_hex': scale_sq.hex(),
        'observed_x1_definition': 'scale/8 = 2**-38',
        'observed_x1_hex': observed_x1.hex(),
        'covariance': [[scale_sq, 0.0, scale], [0.0, 1.0, 0.0], [scale, 0.0, 1.0]],
        'observed_block': [[scale_sq, 0.0], [0.0, 1.0]],
        'observed_block_full_rank_spd_by_positive_diagonal': scale_sq > 0.0,
        'full_covariance_psd_witness': 'the (x1,x3) block equals [scale,1] outer-product [scale,1]; x2 is independent with variance 1',
    },
    'target': {
        'status': 'FAIL_REPRODUCED' if not matched else 'NOT_REPRODUCED',
        'analytic_expected': expected_moments,
        'runtime_returned': actual_moments,
        'reason': 'x1=scale*x3 exactly; observing scale/8 fixes x3=1/8 with variance 0',
    },
}
print(json.dumps(report, sort_keys=True, separators=(',', ':')))
if not source_qualified:
    raise SystemExit(1)
if matched:
    raise SystemExit(2)
