from __future__ import annotations
import hashlib, json, os, platform, resource, subprocess, sys, time
from datetime import datetime, timezone
from pathlib import Path
root=Path(__file__).resolve().parent
probe=root/'probe.py'

env={
 'PATH':os.environ.get('PATH','/usr/bin:/bin'),
 'PYTHONPATH':str(source),
 'PYTHONDONTWRITEBYTECODE':'1',
 'PYTHONNOUSERSITE':'1',
 'OMP_NUM_THREADS':'1','OPENBLAS_NUM_THREADS':'1','MKL_NUM_THREADS':'1','VECLIB_MAXIMUM_THREADS':'1',

 out,err=p.communicate(timeout=15.0)

 timed_out=True;p.terminate()

 'command':command,'cwd':str(root),'interpreter':sys.executable,'interpreter_version':sys.version,'interpreter_prefix':sys.prefix,

raise SystemExit(p.returncode if not timed_out else 124)
